def async_test(function):
    @wraps(function)
    def wrapper(*args, **kwargs):
        return asyncio.run(function(*args, **kwargs))

    return wrapper

def _producer() -> JournalProducer:
    return JournalProducer(
        pid=os.getpid(), pid_create_time=1.0, owner_kind="process", ownership_kind="none"
    )


def _append(paths, effects, *, kind: str = "test"):
    return append_journal_transaction(
        paths,
        operation_kind=kind,
        effects=effects,
        producer=_producer(),
    )


def _setup_base(paths, tmp_path: Path, nodes, *, watermark: int = 0):
    """Write a base snapshot for the CURRENT manifest epoch and point the
    manifest at it (transition mode).  Returns the updated manifest."""

    snapshot_dir = tmp_path / "snapshots"
    snapshot_dir.mkdir(parents=True, exist_ok=True)
    manifest = read_manifest(paths, required=True)
    candidate = write_snapshot_candidate(
        new_snapshot_candidate_path(snapshot_dir),
        {"nodes": nodes},
        account_scope_id=paths.account_root.name,
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        incorporated_through_seq=watermark,
        publication_kind="full_refresh",
    )
    updated = manifest_with_updates(
        manifest,
        base_snapshot_path=candidate,
        coherence_mode="transition",
        base_incorporated_through_seq=watermark,
        journal_floor_seq=watermark + 1,
    )
    write_manifest(
        paths,
        updated,
        expected_epoch_id=manifest.epoch_id,
        expected_head_seq=manifest.journal_head_seq,
    )
    return read_manifest(paths, required=True)


def _view_for(paths, payload, *, seq: int | None = None) -> CacheView:
    manifest = read_manifest(paths, required=True)
    return CacheView.build(
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        adopted_sequence=manifest.journal_head_seq if seq is None else seq,
        nodes_export=payload,
        account_scope_id=manifest.account_scope_id,
    )


def _publisher(paths, payload=None, *, seq: int | None = None) -> CacheViewPublisher:
    if payload is None:
        return CacheViewPublisher(unpopulated_cache_view(account_scope_id=ACCOUNT_A))
    return CacheViewPublisher(_view_for(paths, payload, seq=seq))


def _adopt(paths, publisher, tmp_path):
    return adopt_manifest_head(
        paths=paths, publisher=publisher, snapshot_dir=tmp_path / "snapshots"
    )


def _write_raw_record(paths, sequence: int, effects, *, account: str = ACCOUNT_A, epoch_id: str | None = None):
    manifest = read_manifest(paths, required=True)
    record = JournalTransaction(
        schema_version=1,
        account_scope_id=account,
        sequence=sequence,
        previous_sequence=sequence - 1,
        transaction_id=f"raw{sequence:027d}",
        published_epoch_id=epoch_id or manifest.epoch_id,
        operation_kind="test",
        committed_at=NOW,
        producer=_producer(),
        verification=(),
        effects=tuple(effects),
    )
    atomic_write_json_durable(
        paths.journal_dir / journal_record_filename(sequence), record.to_dict()
    )
    return record

@async_test
async def test_incremental_failure_triggers_one_clean_base_recovery(tmp_path, monkeypatch):
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)
    publisher = _publisher(paths, {"nodes": list(BASE_NODES)}, seq=0)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])

    real_apply = adoption_module.apply_journal_transaction
    calls = {"count": 0}

    def _fail_once(*args, **kwargs):
        calls["count"] += 1
        if calls["count"] == 1:
            raise EffectApplicationError("injected local divergence")
        return real_apply(*args, **kwargs)

    monkeypatch.setattr(adoption_module, "apply_journal_transaction", _fail_once)

    result = await _adopt(paths, publisher, tmp_path)

    assert result.disposition == "recovered_from_base"
    assert result.view.adopted_sequence == 1
    assert result.view.nodes_by_id["C"]["name"] == "gamma"
    assert publisher.capture() is result.view
    # Seal item 6: the recovery diagnostics name the incremental failure,
    # its exception class, and the successful clean-base recovery.
    recovery_diags = [d for d in result.diagnostics if d["kind"] == "incremental_replay_failed"]
    assert len(recovery_diags) == 1
    assert recovery_diags[0]["exception"] == "EffectApplicationError"
    assert recovery_diags[0]["recovery"] == "clean_base_succeeded"

@async_test
async def test_cow_patch_racing_adoption_is_rebased_never_discarded(tmp_path):
    """§5.3/§12.4: a transition COW patch at the SAME markers must survive a
    racing journal adoption — the tail rebases onto the patched view."""

    from workflowy_mcp.coherence.operation_cache import publish_local_cache_view

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)
    publisher = _publisher(paths, {"nodes": list(BASE_NODES)}, seq=0)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])

    # Local legacy patch lands FIRST (same markers, new object).
    def _rename(working):
        for index, node in enumerate(working["nodes"]):
            if node["id"] == "A":
                patched = dict(node)
                patched["name"] = "alpha-local"
                working["nodes"][index] = patched
                return

    publish_local_cache_view(publisher, _rename, reason="test-patch")
    assert publisher.capture().adopted_sequence == 0  # markers unchanged

    result = await _adopt(paths, publisher, tmp_path)

    # The tail replay rebased onto the patched view: BOTH effects present.
    assert result.disposition == "tail_replayed"
    assert result.view.adopted_sequence == 1
    assert result.view.nodes_by_id["A"]["name"] == "alpha-local"
    assert result.view.nodes_by_id["C"]["name"] == "gamma"

@async_test
async def test_older_captured_target_never_regresses_newer_adoption(tmp_path, monkeypatch):
    """A captures head N; the shared head advances to N+1; B publishes N+1
    locally; A's older completion must return the N+1 view without
    publishing, with to_* describing the N+1 view."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)
    publisher = _publisher(paths, {"nodes": list(BASE_NODES)}, seq=0)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])

    real_replay = adoption_module._replay_tail_onto
    fired = {"done": False}

    def _advance_head_and_publish_n_plus_1(payload, records, **kwargs):
        if not fired["done"]:
            fired["done"] = True
            # Head moves to N+1 AFTER A captured N...
            _append(paths, [SemanticEffect.node_create(s02_node("D", name="delta"))])
            # ...and adopter B completes the full N+1 adoption locally.
            manifest = read_manifest(paths, required=True)
            all_records = [
                _read_record_file(
                    paths.journal_dir / journal_record_filename(seq), paths
                )
                for seq in (1, 2)
            ]
            full = real_replay(payload, all_records, **kwargs)
            competitor = CacheView.build(
                epoch_number=manifest.epoch_number,
                epoch_id=manifest.epoch_id,
                adopted_sequence=2,
                nodes_export=full,
                account_scope_id=ACCOUNT_A,
            )
            publisher.publish(competitor, expected_current=publisher.capture())
        return real_replay(payload, records, **kwargs)

    monkeypatch.setattr(
        adoption_module, "_replay_tail_onto", _advance_head_and_publish_n_plus_1
    )

    result = await _adopt(paths, publisher, tmp_path)

    # A's older N=1 attempt never published; the N+1 view stands.
    assert publisher.capture().adopted_sequence == 2
    assert result.view.adopted_sequence == 2
    assert result.to_sequence == 2
    assert result.to_epoch_id == result.view.epoch_id
    assert result.disposition == "already_current"
    assert result.view.nodes_by_id["D"]["name"] == "delta"  # N+1's effect present

@async_test
async def test_apparent_local_ahead_resolves_after_locked_reread(tmp_path, monkeypatch):
    """Seal item 1 rule 5: a stale lock-free manifest read that makes the
    local view look newer must NOT fail — the locked re-read resolves it."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)
    _append(paths, [SemanticEffect.node_create(s02_node("C"))])
    manifest = read_manifest(paths, required=True)
    stale = manifest_with_updates(manifest, journal_head_seq=0)  # the "old complete artifact"
    publisher = _publisher(paths, {"nodes": list(BASE_NODES) + [s02_node("C")]}, seq=1)

    real_read = adoption_module.read_manifest
    calls = {"count": 0}

    def _stale_first_read(paths_arg, *, required=True):
        calls["count"] += 1
        if calls["count"] == 1:
            return stale  # lock-free read observes the PRIOR artifact
        return real_read(paths_arg, required=required)

    monkeypatch.setattr(adoption_module, "read_manifest", _stale_first_read)
    lock_entries = {"count": 0}
    real_lock = adoption_module.coherence_lock_async

    def _counting_lock(*args, **kwargs):
        lock_entries["count"] += 1
        return real_lock(*args, **kwargs)

    monkeypatch.setattr(adoption_module, "coherence_lock_async", _counting_lock)

    result = await _adopt(paths, publisher, tmp_path)

    # The locked re-read saw the real head (1): current, no error, no fail.
    assert result.disposition == "already_current"
    assert result.view.adopted_sequence == 1
    assert lock_entries["count"] == 1  # the slow path DID re-read under the mutex

@async_test
async def test_result_targets_describe_returned_view_after_lost_cas(tmp_path, monkeypatch):
    """Another local adopter publishes the head-N view while this adopter
    builds an older candidate: the returned result must describe the NEWER
    view actually returned (to_* == view markers)."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)
    publisher = _publisher(paths, {"nodes": list(BASE_NODES)}, seq=0)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])
    _append(paths, [SemanticEffect.node_create(s02_node("D", name="delta"))])

    real_replay = adoption_module._replay_tail_onto
    fired = {"done": False}

    def _competing_adoption(payload, records, **kwargs):
        if not fired["done"]:
            fired["done"] = True
            # A competing local adopter completes the FULL tail first.
            full = real_replay(payload, records, **kwargs)
            manifest = read_manifest(paths, required=True)
            competitor = CacheView.build(
                epoch_number=manifest.epoch_number,
                epoch_id=manifest.epoch_id,
                adopted_sequence=2,
                nodes_export=full,
                account_scope_id=ACCOUNT_A,
            )
            publisher.publish(competitor, expected_current=publisher.capture())
        return real_replay(payload, records, **kwargs)

    monkeypatch.setattr(adoption_module, "_replay_tail_onto", _competing_adoption)

    result = await _adopt(paths, publisher, tmp_path)

    assert result.view.adopted_sequence == 2
    assert result.to_sequence == 2  # describes the RETURNED view, never the stale target
    assert result.to_epoch_id == result.view.epoch_id
    assert result.disposition in ("already_current", "tail_replayed")
    assert result.view.nodes_by_id["D"]["name"] == "delta"


@async_test
async def test_epoch_reload_never_overwrites_newer_published_epoch(tmp_path, monkeypatch):
    """An epoch-E reload completing after E+1 was already published locally
    must return the E+1 view untouched (no regression, consistent result)."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    manifest = _setup_base(paths, tmp_path, BASE_NODES)
    publisher = CacheViewPublisher(unpopulated_cache_view(account_scope_id=ACCOUNT_A))

    real_materialize = adoption_module._materialize_reload
    fired = {"done": False}

    def _competing_epoch_publish(paths_arg, manifest_arg, **kwargs):
        if not fired["done"]:
            fired["done"] = True
            newer = CacheView.build(
                epoch_number=manifest.epoch_number + 1,
                epoch_id="f" * 32,
                adopted_sequence=0,
                nodes_export={"nodes": [s02_node("NEW", name="newer")]},
                account_scope_id=ACCOUNT_A,
            )
            publisher.publish(newer, expected_current=publisher.capture())
        return real_materialize(paths_arg, manifest_arg, **kwargs)

    monkeypatch.setattr(adoption_module, "_materialize_reload", _competing_epoch_publish)

    result = await _adopt(paths, publisher, tmp_path)

    assert result.view.epoch_id == "f" * 32
    assert result.to_epoch_id == "f" * 32
    assert result.view.nodes_by_id["NEW"]["name"] == "newer"
    assert publisher.capture().epoch_id == "f" * 32
