# Retired worker-surface arguments (scoped to the worker + its launchers;
# nexus_map_codebase's standalone historical CLI legitimately keeps its own).
RETIRED_WORKER_ARGS = ["--nexus-tag", "--json-file", "--parent-id", "--import-policy", "--dry-run"]


def _production_py_files() -> list[Path]:
    files = [p for p in SRC.rglob("*.py") if p.is_file()]
    files.extend(p for p in ROOT_MODULES if p.is_file())
    return files


def _scan_repo_wide(paths: list[Path] | None = None) -> list[str]:
    violations: list[str] = []
    for path in paths if paths is not None else _production_py_files():
        text = path.read_text(encoding="utf-8", errors="replace")
        for token in RETIRED_REPO_WIDE:
            if token in text:
                violations.append(f"{path.name}: {token}")
    return violations


def test_production_source_has_zero_retired_identifiers() -> None:
    assert _scan_repo_wide() == []


def test_injected_violations_are_detected(tmp_path: Path) -> None:
    for token in RETIRED_REPO_WIDE:
        fixture = tmp_path / f"injected_{abs(hash(token))}.py"
        fixture.write_text(f"# synthetic S06 fence fixture\nX = {token!r}\n", encoding="utf-8")
        found = _scan_repo_wide([fixture])
        assert found, f"scanner failed to detect injected {token!r}"


def test_worker_modes_are_exactly_the_governed_pair() -> None:
    source = WORKER.read_text(encoding="utf-8")
    assert 'choices=["carto_refresh", "markdown_generate"]' in source
    assert '"enchanted"' not in source
    assert '"direct"' not in source
    assert '"carto_refresh"' in source
    assert '"markdown_generate"' in source


def test_worker_and_launchers_have_no_retired_arguments() -> None:
    for path in (WORKER, GOVERNANCE, SERVER):
        text = path.read_text(encoding="utf-8")
        for arg in RETIRED_WORKER_ARGS:
            assert arg not in text, f"{arg} still accepted/emitted by {path.name}"


def test_no_nexus_runs_base_worker_plumbing() -> None:
    worker_source = WORKER.read_text(encoding="utf-8")
    server_source = SERVER.read_text(encoding="utf-8")
    assert "NEXUS_RUNS_BASE" not in worker_source
    assert 'setdefault("NEXUS_RUNS_BASE"' not in server_source


def test_reconcile_and_nexus_have_no_jewel_word() -> None:
    for path in (RECONCILE, NEXUS):
        assert "JEWEL" not in path.read_text(encoding="utf-8")


def test_etch_logger_has_no_jewelstorm() -> None:
    assert "jewelstorm" not in ETCH.read_text(encoding="utf-8")


def test_zero_ra_notes_salvage_beacon_labels() -> None:
    import re

    for path in _production_py_files():
        text = path.read_text(encoding="utf-8", errors="replace")
        assert "ra-notes-salvage" not in text, f"{path.name} retains ra-notes-salvage"
        # Belt-and-suspenders: no beacon block mentions the retired label even
        # with separator variations.
        for match in re.finditer(r"slice_labels=([^\n]*)", text):
            labels = [lbl.strip() for lbl in match.group(1).split(",")]
            assert "ra-notes-salvage" not in labels, f"{path.name}: {match.group(0)}"
