_KILL_CHECKPOINT_DRIVER = r"""
import asyncio
import sys

sys.path.insert(0, sys.argv[3] + "/tests")
from pathlib import Path

from coherence_harness import (
    ACCOUNT_A,
    make_coherence_root,
    s02_exit_ops,
    s02_node,
    s04_append_n,
    s04_checkpoint_config,
    s04_install_base_manifest,
    s04_write_base_snapshot,
)
from workflowy_mcp.coherence import DurableFileOps, run_checkpoint_rotation


def main() -> None:
    seam = sys.argv[1]
    root = Path(sys.argv[2])
    paths = make_coherence_root(root, ACCOUNT_A)
    snapshot_dir = root / "nodes_export_cache"
    base = s04_write_base_snapshot(
        paths, snapshot_dir, [s02_node("a")], epoch_number=0, epoch_id="b" * 32
    )
    s04_install_base_manifest(paths, base_snapshot_path=base, epoch_id="b" * 32)
    s04_append_n(paths, 3)

    if seam in ("unlink", "retire_after_first"):
        from workflowy_mcp.carto_job_governance import (
            _default_mkdir,
            _default_replace,
            _default_tmp_name,
            _default_unlink,
            _default_write_fsync,
        )

        import os

        state = {"journal_unlinks": 0}
        limit = 1 if seam == "unlink" else 2

        def _exit_unlink(path):
            # Exit ONLY on the Nth journal-record unlink (retirement); the
            # atomic writer's own-temp cleanups pass through to the real
            # unlink.  "unlink" kills at the FIRST attempt (retirement
            # start); "retire_after_first" lets the first incorporated
            # record retire and kills at the second (partial retirement).
            if getattr(path, "parent", None) is not None and path.parent.name == "journal":
                state["journal_unlinks"] += 1
                if state["journal_unlinks"] >= limit:
                    os._exit(42)
            _default_unlink(path)

        fs_ops = DurableFileOps(
            write_fsync=_default_write_fsync,
            replace=_default_replace,
            mkdir=_default_mkdir,
            unlink=_exit_unlink,
            tmp_name=_default_tmp_name,
        )
    elif seam in ("basis_capture", "private_replay"):
        import os

        import workflowy_mcp.coherence.checkpoint as _checkpoint_module

        if seam == "basis_capture":
            # Kill INSIDE the basis-capture locked region (claim acquired,
            # due revalidated, no durable lineage mutation yet) — §11.6
            # 'before/after basis capture' (and the state-equivalent 'after
            # checkpoint claim' point).
            def _exit_read_records(*args, **kwargs):
                os._exit(42)

            _checkpoint_module.read_journal_records = _exit_read_records
        else:
            # Kill during private materialization (after basis capture,
            # before the candidate write) — §11.6 'during private replay'.
            def _exit_materialize(*args, **kwargs):
                os._exit(42)

            _checkpoint_module.materialize_payload_with_records = _exit_materialize
        fs_ops = None
    else:
        times = 1
        point = seam
        if seam.startswith("claim_replace:"):
            point, _, raw = seam.partition(":")
            times = int(raw)
        fs_ops = s02_exit_ops(point, times=times)
    asyncio.run(
        run_checkpoint_rotation(
            paths,
            snapshot_dir=snapshot_dir,
            config=s04_checkpoint_config(record_threshold=2),
            fs_ops=fs_ops,
        )
    )


main()
"""

def _snapshot_dir(root: Path) -> Path:
    return root / "nodes_export_cache"


def _based_manifest(paths, root: Path) -> None:
    base = s04_write_base_snapshot(
        paths, _snapshot_dir(root), [s02_node("a")], epoch_number=0, epoch_id="b" * 32
    )
    s04_install_base_manifest(paths, base_snapshot_path=base, epoch_id="b" * 32)


def _journal_sequences(paths) -> list[int]:
    if not paths.journal_dir.exists():
        return []
    return sorted(
        seq
        for entry in paths.journal_dir.iterdir()
        if (seq := parse_journal_record_filename(entry.name)) is not None
    )


def _assert_snapshot_envelopes_parse(root: Path) -> None:
    """No torn candidate: every snapshot envelope on disk parses as JSON."""
    for entry in _snapshot_dir(root).glob("*_nodes_export_snapshot.json"):
        json.loads(entry.read_text(encoding="utf-8"))


def _rotate(paths, root: Path, *, threshold: int = 2):
    return asyncio.run(
        run_checkpoint_rotation(
            paths,
            snapshot_dir=_snapshot_dir(root),
            config=s04_checkpoint_config(record_threshold=threshold),
        )
    )

@pytest.mark.parametrize(
    "seam",
    [
        "candidate_temp_write",
        "candidate_fsynced",
        "candidate_replace",
        "manifest_temp_write",
        "manifest_replace",
    ],
    ids=[
        "before_candidate_write",
        "after_candidate_fsync",
        "before_candidate_replace",
        "after_candidate_replace",
        "before_epoch_publish",
    ],
)
def test_kill_before_epoch_publish_leaves_old_lineage(tmp_path, seam):
    """Kills from the first candidate touch through the manifest publish
    attempt: the OLD epoch is wholly authoritative at every point.  Honest
    id↔seam map (S04 Remediation Chain 3): ``candidate_replace`` trips
    BEFORE the replace (id ``before_candidate_replace`` — previously
    mislabeled 'after'); the true post-candidate-replace/pre-publication
    point is the ``manifest_temp_write`` seam (id
    ``after_candidate_replace``); ``candidate_fsynced`` is the true
    post-temp-fsync/pre-replace point."""
    root = tmp_path / "kill"
    root.mkdir()
    completed = run_subprocess(
        _KILL_CHECKPOINT_DRIVER, seam, str(root), str(Path.cwd()), timeout=120.0
    )
    assert completed.returncode != 0  # the os._exit kill landed
    paths = make_coherence_root(root, ACCOUNT_A, bootstrap=False)
    manifest = read_manifest(paths, required=True)
    # Old epoch wholly authoritative — never a half epoch, never torn.
    assert manifest.epoch_number == 0
    assert manifest.epoch_id == "b" * 32
    assert manifest.base_incorporated_through_seq == 0
    assert manifest.journal_head_seq == 3
    assert _journal_sequences(paths) == [1, 2, 3]
    _assert_snapshot_envelopes_parse(root)
    # Dead-owner recovery on the next rotation: the failed claim is recovered
    # and a successor publishes E+1 cleanly.
    result = _rotate(paths, root)
    assert result.status == "published", result.warnings
    manifest = read_manifest(paths, required=True)
    assert manifest.epoch_number == 1
    assert manifest.base_incorporated_through_seq == 3

def test_kill_after_publish_before_terminal_recovers_completed(tmp_path):
    root = tmp_path / "kill"
    root.mkdir()
    completed = run_subprocess(
        _KILL_CHECKPOINT_DRIVER,
        "claim_replace:3",  # the post-publish 'completed' terminal write
        str(root),

    )
    assert completed.returncode != 0
    paths = make_coherence_root(root, ACCOUNT_A, bootstrap=False)
    manifest = read_manifest(paths, required=True)
    # The NEW epoch is already wholly authoritative (publish completed).
    assert manifest.epoch_number == 1
    assert manifest.base_incorporated_through_seq == 3
    claim = read_refresh_claim(paths)

    # Dead-owner recovery classifies by manifest target identity: completed.
    from coherence_harness import s04_producer

    from workflowy_mcp.coherence import SemanticEffect, seal_verified_journal_transaction

    result = _rotate(paths, root, threshold=1)

def test_kill_mid_retirement_partial_residue_retired_by_next_maintenance(tmp_path):
    """§11.6 mid-retirement / §12.2 during_journal_retirement: the FIRST
    incorporated record retires, then the owner dies before the remainder —
    a genuine PARTIAL retirement.  The next maintenance pass removes the
    residue and NEVER deletes the active tail (seq >= floor)."""
    root = tmp_path / "kill"
    root.mkdir()
    completed = run_subprocess(
        _KILL_CHECKPOINT_DRIVER,
        "retire_after_first",
        str(root),
        str(Path.cwd()),
        timeout=120.0,
    )
    assert completed.returncode != 0
    paths = make_coherence_root(root, ACCOUNT_A, bootstrap=False)
    manifest = read_manifest(paths, required=True)
    # New epoch authoritative; retirement died after exactly ONE unlink:
    # record 1 retired, records 2 and 3 survive as partial residue.
    assert manifest.epoch_number == 1
    assert manifest.base_incorporated_through_seq == 3
    assert manifest.journal_floor_seq == 4
    assert _journal_sequences(paths) == [2, 3]
    # A new active-tail record (seq 4, AT the floor) must never be deleted.
    # RAW append (not the sealer): the sealer's own §10.7 'open' maintenance
    # would quietly retire the residue at seal time — the point HERE is to
    # prove the NEXT maintenance pass removes it.
    append_journal_transaction(
        paths,
        operation_kind="test",
        effects=[SemanticEffect.tombstone(["tail"])],
        producer=s04_producer(),
        transaction_id="tx-tail",
    )
    assert _journal_sequences(paths) == [2, 3, 4]
    # The next maintenance pass retires ONLY the below-floor residue.
    result = _rotate(paths, root, threshold=500)
    assert result.status == "not_due"
    assert _journal_sequences(paths) == [4]
    manifest = read_manifest(paths, required=True)
    assert manifest.journal_head_seq == 4

def test_barrier_and_append_serialize_without_lost_state(tmp_path):
    """IN-PROCESS THREAD serialization (honest label, S04 Remediation Chain 2):
    useful thread-level coverage of the mutex discipline; the GENUINE
    cross-process twin (barrier raiser + journal appender as separate live
    subprocesses) lives in tests/test_coherence_s04_multiprocess_acceptance.py."""
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    errors: list[BaseException] = []
    from coherence_harness import s04_append

    def _appends() -> None:
        try:
            for _ in range(5):
                s04_append(paths)
        except BaseException as exc:  # noqa: BLE001
            errors.append(exc)

    def _barriers() -> None:
        try:
            for index in range(3):
                s04_raise_barrier(paths, transaction_id=f"tx-concurrent-{index}")
        except BaseException as exc:  # noqa: BLE001
            errors.append(exc)

    threads = [threading.Thread(target=_appends), threading.Thread(target=_barriers)]
    for thread in threads:
        thread.start()
    for thread in threads:
        thread.join()
    assert errors == []
    manifest = read_manifest(paths, required=True)
    assert manifest.journal_head_seq == 5
    assert manifest.dirty_status.active is True
    assert manifest.dirty_status.generation == 3
    # The journal tail remains contiguous (no torn interleaving).
    assert _journal_sequences(paths) == [1, 2, 3, 4, 5]


def test_concurrent_rotations_publish_exactly_once(tmp_path):
    """IN-PROCESS THREAD one-flight coverage (honest label, S04 Remediation
    Chain 2): the GENUINE cross-process twin (two checkpoint requesters as
    separate live subprocesses sharing the fixed claim) lives in
    tests/test_coherence_s04_multiprocess_acceptance.py."""
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _based_manifest(paths, tmp_path)
    s04_append_n(paths, 3)
    outcomes: list = []
    errors: list[BaseException] = []

    def _rotate_sync() -> None:
        try:
            outcomes.append(_rotate(paths, tmp_path))
        except BaseException as exc:  # noqa: BLE001
            errors.append(exc)

    threads = [threading.Thread(target=_rotate_sync) for _ in range(2)]
    for thread in threads:
        thread.start()
    for thread in threads:
        thread.join()
    assert errors == []
    statuses = sorted(result.status for result in outcomes)
    # Exactly ONE publication (the other racer either defers on the live
    # claim or finds the tail already compacted — both are correct one-flight
    # outcomes; the epoch advances exactly once either way).
    assert statuses.count("published") == 1
    assert statuses[0] in ("deferred_publisher_busy", "not_due")
    assert statuses[1] == "published"
    manifest = read_manifest(paths, required=True)
    assert manifest.epoch_number == 1  # epoch advanced EXACTLY once


def test_lock_free_readers_observe_only_complete_manifests(tmp_path):
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _based_manifest(paths, tmp_path)
    s04_append_n(paths, 3)
    stop = threading.Event()
    result_holder: dict = {}

    def _reader() -> None:
        result_holder["result"] = run_reader_loop(paths.manifest_path, stop, poll=0.0005)

    reader = threading.Thread(target=_reader)
    reader.start()
    rotate_result = _rotate(paths, tmp_path)
    stop.set()
    reader.join()
    assert rotate_result.status == "published", rotate_result.warnings
    reader_result = result_holder["result"]
    assert reader_result.torn == 0
    assert reader_result.unexpected == []
    assert reader_result.reads > 0
