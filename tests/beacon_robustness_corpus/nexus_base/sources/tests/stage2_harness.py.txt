"""Waypoint A — Stage 2 (Chunk 1) shared multi-process harness primitives.

Every Stage-2 test spawns REAL OS subprocesses contending on REAL
filesystem governance artifacts (``.mutation_leases/``, ``.refresh_locks/``
(CARTO claims), ``.coherence/`` (manifest/journal/refresh-publication)).
Synchronization is deterministic — sentinel files and injected seams —
never probability-based ``time.sleep`` races as the load-bearing proof.

Verdict-laundering guard: every actor writes its REAL ``os.getpid()`` into
a marker file, and the parent asserts the actors' PIDs are pairwise
distinct.  A result attributed to the wrong process is no evidence at all;
distinct-PID logging is what keeps each verdict attached to its actor.

Actor code strings import harness modules via ``sys.path`` insertion of the
repo's ``tests`` directory (the same shape as the S04/S05 acceptance
files); the installed editable package provides ``workflowy_mcp``.
"""

from __future__ import annotations

import json
import os
import subprocess
import sys
from collections.abc import Mapping
from pathlib import Path
from typing import Any

from coherence_harness import (  # noqa: F401 — re-exported conveniences
    launch_subprocess,
    touch,
    wait_for_file,
)

REPO = Path(__file__).resolve().parents[1]
TESTS_DIR = Path(__file__).resolve().parent


# ---------------------------------------------------------------------------
# Actor launching
# ---------------------------------------------------------------------------


# @beacon[
#   id=auto-beacon@launch_actor-p1c2,
#   role=launch_actor,
#   slice_labels=mci-harness-process,
#   kind=ast,
# ]
def launch_actor(code: str, *args: str, env: Mapping[str, str] | None = None) -> subprocess.Popen[str]:
    """Launch one long-running actor subprocess on the ACTIVE interpreter."""

    return launch_subprocess(code, *args, env=env)


def run_actor(code: str, *args: str, timeout: float = 60.0) -> subprocess.CompletedProcess[str]:
    """Launch one short actor subprocess and wait for it (bounded)."""

    full_env = dict(os.environ)
    process = subprocess.Popen(
        [sys.executable, "-c", code, *args],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        env=full_env,
    )
    out, err = process.communicate(timeout=timeout)
    return subprocess.CompletedProcess(process.args, process.returncode, out, err)


# ---------------------------------------------------------------------------
# PID markers (the verdict-laundering guard)
# ---------------------------------------------------------------------------


def pid_marker_code(marker_expr: str) -> str:
    """Actor-side one-liner: write this process's real PID to a marker file."""

    return f'touch({marker_expr}, str(os.getpid()))\n'


def read_pid(marker: Path, *, timeout: float = 60.0) -> int:
    """Read one actor PID marker (waits for it deterministically)."""

    assert wait_for_file(marker, timeout=timeout), f"PID marker never appeared: {marker}"
    text = marker.read_text(encoding="utf-8").strip()
    pid = int(text)
    assert pid > 0
    return pid


# @beacon[
#   id=auto-beacon@collect_distinct_pids-j87p,
#   role=collect_distinct_pids,
#   slice_labels=mci-harness-process,
#   kind=ast,
# ]
def collect_distinct_pids(root: Path, actor_names: list[str], *, timeout: float = 60.0) -> dict[str, int]:
    """Read every actor's PID marker and assert pairwise distinctness.

    Verdict-laundering guard: a test whose actors somehow shared a PID
    (fork reuse, a launcher shim mistake, an accidental in-process double)
    could silently attribute one process's evidence to another.  Refuse
    that here.  NOTE: the venv ``python.exe`` on this machine is a launcher
    shim — ``Popen.pid`` is the shim, not the interpreter — which is
    exactly why actors self-report ``os.getpid()`` through marker files.
    """

    pids: dict[str, int] = {}
    for name in actor_names:
        pids[name] = read_pid(root / f"pid_{name}", timeout=timeout)
    values = list(pids.values())
    assert len(set(values)) == len(values), (
        f"actor PIDs are not pairwise distinct (verdict-laundering risk): {pids}"
    )
    return pids


# ---------------------------------------------------------------------------
# Result payloads
# ---------------------------------------------------------------------------


def write_result(path: Path, payload: Mapping[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(dict(payload)), encoding="utf-8")


def read_result(path: Path, *, timeout: float = 60.0) -> dict[str, Any]:
    assert wait_for_file(path, timeout=timeout), f"result file never appeared: {path}"
    return json.loads(path.read_text(encoding="utf-8"))


# @beacon[
#   id=auto-beacon@finish_actor-x2zl,
#   role=finish_actor,
#   slice_labels=mci-harness-process,
#   kind=ast,
# ]
def finish_actor(process: subprocess.Popen[str], *, timeout: float = 120.0, label: str = "actor") -> str:
    """Collect an actor's exit and assert a clean status; returns stdout."""

    out, err = process.communicate(timeout=timeout)
    assert process.returncode == 0, f"{label} failed (rc={process.returncode}):\n{out}\n{err}"
    return out
