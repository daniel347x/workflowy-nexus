def _cache() -> dict[str, dict]:
    return {
        "vault": {"id": "vault"},
        "repo": {"id": "repo", "parent_id": "vault"},
        "file": {"id": "file", "parent_id": "repo"},
    }


def _footprint(tmp_path: Path, name: str = "a.py") -> dict:
    target = tmp_path / "repo" / name
    target.parent.mkdir(exist_ok=True)
    target.write_text("x = 1\n", encoding="utf-8")
    return gov.compute_footprint(target, "file", _cache(), kind="file")


def _spawning_claim(tmp_path: Path, jobs: Path, *, job_id: str) -> tuple[dict, Path]:
    process = psutil.Process(os.getpid())
    claim = gov.build_claim(
        job_id=job_id,
        job_type="CARTO_REFRESH",
        dedupe_key=hashlib.sha256(job_id.encode("utf-8")).hexdigest(),
        account_scope_id=ACCOUNT,
        footprint=_footprint(tmp_path, f"{job_id}.py"),
        launcher="system",
        parent_job_id=None,
        launcher_pid=os.getpid(),
        pid=os.getpid(),
        pid_create_time=process.create_time(),
        handoff_state="spawning",
        spawn_started_at=gov.utc_now_iso(),
        execution_owner="launcher",
    )
    claim_path = gov.write_claim(claim, jobs)
    assert claim_path.exists()
    job_file = jobs / f"{job_id}.json"
    gov.atomic_write_json(
        job_file,
        {
            "id": job_id,
            "type": "CARTO_REFRESH",
            "status": "spawning",
            "claim_id": claim["claim_id"],
            "pid": claim["pid"],
            "pid_create_time": claim["pid_create_time"],
            "execution_owner": "launcher",
            "handoff_state": "spawning",
        },
    )
    return claim, job_file


async def _spawn(claim: dict, job_file: Path, jobs: Path, popen_factory) -> dict:
    return await gov._spawn_governed_worker(
        claim=claim,
        job_file=job_file,
        log_file=jobs / f"{claim['job_id']}.log",
        jobs_root=jobs,
        worker_script=jobs / "worker.py",
        worker_mode="carto_refresh",
        env={},
        popen_factory=popen_factory,
    )


class _AdmissionWriterGate:
    def __init__(self, job_file: Path, *, replace_job_then_raise: bool = False):
        self.job_file = job_file
        self.replace_job_then_raise = replace_job_then_raise
        self.block_tombstone = True
        self.blocked_attempt = asyncio.Event()
        self.recovered = asyncio.Event()

    def __call__(self, path, payload):
        target = Path(path)
        if target == self.job_file:
            if self.replace_job_then_raise:
                gov.atomic_write_json(target, payload)
            raise OSError("initial job JSON write failed")
        if target.parent.name == ".refresh_tombstones":
            if self.block_tombstone:
                self.blocked_attempt.set()
                raise OSError("tombstone publication unavailable")
            gov.atomic_write_json(target, payload)
            self.recovered.set()
            return
        gov.atomic_write_json(target, payload)


class _TombstoneGate:
    def __init__(self, real_writer):
        self.real_writer = real_writer
        self.blocked = True
        self.blocked_attempt = asyncio.Event()
        self.recovered = asyncio.Event()
        self.first_attempt_entered = threading.Event()
        self.allow_first_attempt = threading.Event()
        self.wait_on_first_attempt = False
        self.calls = 0

    def __call__(self, tombstone, jobs_root, *, atomic_writer=gov.atomic_write_json):
        self.calls += 1
        if self.blocked:
            self.blocked_attempt.set()
            if self.wait_on_first_attempt and self.calls == 1:
                self.first_attempt_entered.set()
                assert self.allow_first_attempt.wait(timeout=2)
            raise OSError("tombstone publication unavailable")
        result = self.real_writer(tombstone, jobs_root, atomic_writer=atomic_writer)
        self.recovered.set()
        return result


async def _finish_supervised_claim_recovery(gate, claim_id: str) -> None:
    key = f"claim:{claim_id}"
    supervisor = gov._CLEANUP_TASKS[key]
    gate.blocked = False
    if hasattr(gate, "block_tombstone"):
        gate.block_tombstone = False
    await asyncio.wait_for(gate.recovered.wait(), timeout=2)
    await asyncio.wait_for(asyncio.shield(supervisor), timeout=2)
    assert key not in gov._CLEANUP_OBLIGATIONS
    assert key not in gov._CLEANUP_TASKS


def test_admission_job_write_and_tombstone_failure_registers_supervisor_then_recovers(tmp_path):
    async def scenario() -> None:
        jobs = tmp_path / "jobs"
        job_file = jobs / "new.json"
        writer = _AdmissionWriterGate(job_file)

        with pytest.raises(gov.GovernanceCleanupError) as caught:
            await gov.admit_carto_job(
                job_file=job_file,
                job_payload={"id": "new", "type": "CARTO_REFRESH"},
                job_type="CARTO_REFRESH",
                dedupe_key="a" * 64,
                account_scope_id=ACCOUNT,
                footprint=_footprint(tmp_path),
                launcher="system",
                parent_job_id=None,
                prepared=False,
                jobs_root=jobs,
                atomic_writer=writer,
            )

        claims = list(gov.governance_paths(jobs).claims_dir.glob("*.lock.json"))
        assert len(claims) == 1
        claim = gov.read_claim(claims[0])
        key = f"claim:{claim['claim_id']}"
        assert not job_file.exists()
        assert not list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))
        assert caught.value.code == "governance_cleanup_required"
        assert isinstance(caught.value.__cause__, OSError)
        assert "initial job JSON write failed" in str(caught.value)
        assert claim["claim_id"] in str(caught.value)
        assert str(claims[0]) in str(caught.value)
        assert "affected region=" in str(caught.value)
        assert [row["key"] for row in gov.pending_cleanup_obligations()] == [key]
        assert list(gov._CLEANUP_TASKS) == [key]
        assert not gov._CLEANUP_TASKS[key].done()

        await _finish_supervised_claim_recovery(writer, claim["claim_id"])
        assert not claims[0].exists()
        assert len(list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))) == 1

    asyncio.run(scenario())

def test_adopted_worker_preserved_without_launcher_obligation(tmp_path, monkeypatch):
    async def scenario() -> None:
        jobs = tmp_path / "jobs"
        jobs.mkdir()
        claim, job_file = _spawning_claim(tmp_path, jobs, job_id="worker-wins")

        real_async_os_lock = gov.async_os_lock
        lock_calls = 0

        @asynccontextmanager
        async def fail_post_popen_bookkeeping_lock(path, *args, **kwargs):
            nonlocal lock_calls
            lock_calls += 1
            if lock_calls == 2:
                raise OSError("launcher bookkeeping lock failed")
            async with real_async_os_lock(path, *args, **kwargs):
                yield

        monkeypatch.setattr(gov, "async_os_lock", fail_post_popen_bookkeeping_lock)

        def popen_factory(*_args, **_kwargs):
            gov.adopt_carto_worker_claim(job_file, pid=4242, pid_create_time=42.0)
            return SimpleNamespace(pid=4242)

        result = await _spawn(claim, job_file, jobs, popen_factory)
        current = gov.read_claim(gov.claim_path_for(claim, jobs))
        job = json.loads(job_file.read_text(encoding="utf-8"))
        assert result["success"] is True
        assert result["rollback_outcome"]["ownership"] == "worker_owned"
        assert current["execution_owner"] == "detached_worker"
        assert current["handoff_state"] == "running"
        assert job["status"] == "running"
        assert not gov.pending_cleanup_obligations()
        assert not gov._CLEANUP_TASKS
        assert not list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))

    asyncio.run(scenario())


def test_blocked_rollback_seals_out_contending_adoption_and_uses_one_supervisor(
    tmp_path, monkeypatch
):
    async def scenario() -> None:
        jobs = tmp_path / "jobs"
        jobs.mkdir()
        claim, job_file = _spawning_claim(tmp_path, jobs, job_id="contended-blocked")
        gate = _TombstoneGate(gov.write_tombstone_max_merge)
        gate.wait_on_first_attempt = True
        monkeypatch.setattr(gov, "write_tombstone_max_merge", gate)

        real_process = gov.psutil.Process

        class FailingChildProcessLookup:
            def create_time(self):
                raise OSError("bookkeeping failed")

        def process_lookup(pid):
            if int(pid) == 4343:
                return FailingChildProcessLookup()
            return real_process(pid)

        monkeypatch.setattr(gov.psutil, "Process", process_lookup)
        adoption: dict[str, object] = {}
        adopter_done = threading.Event()

        def adopter() -> None:
            assert gate.first_attempt_entered.wait(timeout=2)
            gate.allow_first_attempt.set()
            try:
                adoption["result"] = gov.adopt_carto_worker_claim(
                    job_file, pid=4343, pid_create_time=43.0
                )
            except BaseException as exc:
                adoption["error"] = exc
            finally:
                adopter_done.set()

        adopter_thread = threading.Thread(target=adopter)

        def popen_factory(*_args, **_kwargs):
            adopter_thread.start()
            return SimpleNamespace(pid=4343)

        result = await _spawn(claim, job_file, jobs, popen_factory)
        assert adopter_done.wait(timeout=2)
        adopter_thread.join(timeout=2)
        assert not adopter_thread.is_alive()

        claim_path = gov.claim_path_for(claim, jobs)
        key = f"claim:{claim['claim_id']}"
        assert result["error_kind"] == "governance_cleanup_required"
        assert isinstance(adoption.get("error"), gov.GovernanceConflictError)
        assert "result" not in adoption
        assert gov.read_claim(claim_path)["execution_owner"] == "launcher"
        assert gov.read_claim(claim_path)["handoff_state"] == "running"
        assert [row["key"] for row in gov.pending_cleanup_obligations()] == [key]
        assert list(gov._CLEANUP_TASKS) == [key]
        assert not list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))

        await _finish_supervised_claim_recovery(gate, claim["claim_id"])
        assert not claim_path.exists()
        assert len(list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))) == 1

    asyncio.run(scenario())

def _snapshot(path: Path) -> dict[str, str]:
    # S02: governed snapshots carry the v1 lineage wrapper (legacy wrappers
    # are bootstrap-only and no longer accepted at this seam).
    payload = {
        "snapshot_schema_version": 1,
        "snapshot_timestamp": gov.utc_now_iso(),
        "account_scope_id": ACCOUNT,
        "epoch_number": 0,
        "epoch_id": "e" * 32,
        "incorporated_through_seq": 0,
        "publication_kind": "full_refresh",
        "nodes_export": {"nodes": []},
    }
    path.write_text(json.dumps(payload), encoding="utf-8")
    return {"snapshot_path": str(path), "freshness_disposition": "performed"}


def _cleanup_failure() -> gov.GovernanceCleanupError:
    try:
        raise OSError("initial job JSON write failed")
    except OSError as primary:
        try:
            raise gov.GovernanceCleanupError(
                "claim cleanup remains blocked",
                artifact_paths=("A.lock.json", "B.lock.json"),
            ) from primary
        except gov.GovernanceCleanupError as cleanup:
            return cleanup


def test_public_refresh_launcher_preserves_cleanup_required_plain_and_prepared(
    tmp_path, monkeypatch
):
    worker = tmp_path / "worker.py"
    worker.write_text("# worker\n", encoding="utf-8")
    snapshot = _snapshot(tmp_path / "snapshot.json")

    async def fail_admission(**_kwargs):
        raise _cleanup_failure()

    monkeypatch.setattr(gov, "admit_carto_job", fail_admission)

    async def launch(*, prepared: bool) -> dict:
        kwargs = {
            "root_uuid": "file",
            "mode": "file",
            "account_scope_id": ACCOUNT,
            "footprint_factory": lambda _selection: _footprint(tmp_path),
            "jobs_root": tmp_path / ("prepared-jobs" if prepared else "plain-jobs"),
            "worker_script": worker,
            "env": {},
            "freshness_callback": lambda: snapshot,
        }
        if prepared:
            kwargs.update(
                preparation=lambda *_args: {"success": True},
                initial_footprint=_footprint(tmp_path, "prepared.py"),
            )

    async def scenario() -> None:

            result = await launch(prepared=prepared)
            assert result == {
                "success": False,
                "error_kind": "governance_cleanup_required",
                "error": "claim cleanup remains blocked",
                "blocking_artifacts": ["A.lock.json", "B.lock.json"],
                "root_uuid": "file",
                "mode": "file",
                "primary_error": "initial job JSON write failed",
            }

    asyncio.run(scenario())


def test_public_plain_launch_preserves_existing_cleanup_supervisor(tmp_path, monkeypatch):
    worker = tmp_path / "worker.py"
    worker.write_text("# worker\n", encoding="utf-8")
    snapshot = _snapshot(tmp_path / "snapshot.json")
    jobs = tmp_path / "jobs"
    real_admit = gov.admit_carto_job

    class Writer:
        def __init__(self):
            self.block_tombstone = True
            self.blocked_attempt = asyncio.Event()
            self.recovered = asyncio.Event()

        def __call__(self, path, payload):
            target = Path(path)
            if target.parent == jobs and target.suffix == ".json":
                raise OSError("initial job JSON write failed")
            if target.parent.name == ".refresh_tombstones":
                if self.block_tombstone:
                    self.blocked_attempt.set()
                    raise OSError("tombstone publication unavailable")
                gov.atomic_write_json(target, payload)
                self.recovered.set()
                return
            gov.atomic_write_json(target, payload)

    writer = Writer()

    async def injected_admit(**kwargs):
        return await real_admit(**kwargs, atomic_writer=writer)

    monkeypatch.setattr(gov, "admit_carto_job", injected_admit)

    async def scenario() -> None:
        result = await gov.launch_carto_refresh_job(
            root_uuid="file",
            mode="file",
            account_scope_id=ACCOUNT,
            footprint_factory=lambda _selection: _footprint(tmp_path),
            jobs_root=jobs,
            worker_script=worker,
            env={},
            freshness_callback=lambda: snapshot,
        )
        claims = list(gov.governance_paths(jobs).claims_dir.glob("*.lock.json"))
        assert len(claims) == 1
        claim = gov.read_claim(claims[0])
        key = f"claim:{claim['claim_id']}"
        assert result["error_kind"] == "governance_cleanup_required"
        assert result["blocking_artifacts"] == [str(claims[0])]
        assert result["primary_error"] == "initial job JSON write failed"
        assert [row["key"] for row in gov.pending_cleanup_obligations()] == [key]
        assert list(gov._CLEANUP_TASKS) == [key]

        await _finish_supervised_claim_recovery(writer, claim["claim_id"])
        assert not claims[0].exists()
        assert len(list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))) == 1

    asyncio.run(scenario())

def test_admission_job_write_failure_with_already_absent_claim_keeps_primary(tmp_path):
    async def scenario() -> None:
        jobs = tmp_path / "jobs"
        job_file = jobs / "absent.json"

        def writer(path, payload):
            target = Path(path)
            if target == job_file:
                claims = list(gov.governance_paths(jobs).claims_dir.glob("*.lock.json"))
                assert len(claims) == 1
                claims[0].unlink()
                raise OSError("initial job JSON write failed")
            gov.atomic_write_json(target, payload)

        with pytest.raises(OSError, match="initial job JSON write failed"):

        assert not job_file.exists()
        assert not list(gov.governance_paths(jobs).claims_dir.glob("*.json"))
        assert not list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))
        assert not gov.pending_cleanup_obligations()
        assert not gov._CLEANUP_TASKS

    asyncio.run(scenario())


def test_admission_job_write_failure_preserves_replacement_token(tmp_path, monkeypatch):
    async def scenario() -> None:
        jobs = tmp_path / "jobs"
        job_file = jobs / "replacement.json"
        real_read_claim = gov.read_claim
        replaced: dict[str, object] = {}

        def read_then_replace(path):
            claim = real_read_claim(path)
            if not replaced:
                replacement = {**claim, "claim_id": "b" * 32}
                gov.atomic_write_json(path, replacement)
                replaced.update(path=Path(path), claim=replacement)
            return claim

        def writer(path, payload):
            target = Path(path)
            if target == job_file:
                raise OSError("initial job JSON write failed")
            gov.atomic_write_json(target, payload)

        monkeypatch.setattr(gov, "read_claim", read_then_replace)
        with pytest.raises(gov.GovernanceConflictError) as caught:
            await gov.admit_carto_job(**_admission_kwargs(tmp_path, jobs, job_file, writer))

        replacement_path = replaced["path"]
        assert isinstance(replacement_path, Path)
        assert real_read_claim(replacement_path)["claim_id"] == "b" * 32
        assert "initial job JSON write failed" in str(caught.value)
        assert "token-mismatched or ambiguous" in str(caught.value)
        assert not job_file.exists()
        assert not list(gov.governance_paths(jobs).tombstones_dir.glob("*.json"))
        assert not gov.pending_cleanup_obligations()
        assert not gov._CLEANUP_TASKS

    asyncio.run(scenario())

def test_spawn_rollback_absent_replacement_and_ambiguous_are_safe_noops(tmp_path):
    async def scenario() -> None:
        # Already absent: ordinary primary failure, no retry authority.
        absent_jobs = tmp_path / "absent-jobs"
        absent_jobs.mkdir()
        absent_claim, absent_job = _spawning_claim(tmp_path, absent_jobs, job_id="absent")
        gov.claim_path_for(absent_claim, absent_jobs).unlink()
        absent = await _spawn(
            absent_claim,
            absent_job,
            absent_jobs,
            lambda *_args, **_kwargs: (_ for _ in ()).throw(AssertionError("not reached")),
        )
        assert absent["success"] is False
        assert absent["rollback_outcome"]["ownership"] == "already_absent"
        assert not gov.pending_cleanup_obligations()
        assert not list(gov.governance_paths(absent_jobs).tombstones_dir.glob("*.json"))

        # Replacement token at the expected job suffix is preserved loudly.
        replacement_jobs = tmp_path / "replacement-jobs"
        replacement_jobs.mkdir()
        original, replacement_job = _spawning_claim(
            tmp_path, replacement_jobs, job_id="replacement"
        )
        replacement_path = gov.claim_path_for(original, replacement_jobs)
        replacement = {**original, "claim_id": "b" * 32}
        gov.atomic_write_json(replacement_path, replacement)
        mismatch = await _spawn(
            original,
            replacement_job,
            replacement_jobs,
            lambda *_args, **_kwargs: (_ for _ in ()).throw(AssertionError("not reached")),
        )
        assert mismatch["error_kind"] == "governance_state_unclassifiable"
        assert mismatch["rollback_outcome"]["ownership"] == "token_mismatch_or_ambiguous"
        assert gov.read_claim(replacement_path)["claim_id"] == "b" * 32
        assert not gov.pending_cleanup_obligations()
        assert not list(gov.governance_paths(replacement_jobs).tombstones_dir.glob("*.json"))

        # Duplicate exact tokens are ambiguous and neither copy is touched.
        ambiguous_jobs = tmp_path / "ambiguous-jobs"
        ambiguous_jobs.mkdir()
        ambiguous_claim, ambiguous_job = _spawning_claim(
            tmp_path, ambiguous_jobs, job_id="ambiguous"
        )
        duplicate = gov.governance_paths(ambiguous_jobs).claims_dir / "duplicate.lock.json"
        gov.atomic_write_json(duplicate, ambiguous_claim)
        ambiguous = await _spawn(
            ambiguous_claim,
            ambiguous_job,
            ambiguous_jobs,
            lambda *_args, **_kwargs: (_ for _ in ()).throw(AssertionError("not reached")),
        )
        assert ambiguous["error_kind"] == "governance_state_unclassifiable"
        assert ambiguous["rollback_outcome"]["ownership"] == "token_mismatch_or_ambiguous"
        assert gov.claim_path_for(ambiguous_claim, ambiguous_jobs).exists()
        assert duplicate.exists()
        assert not gov.pending_cleanup_obligations()
        assert not list(gov.governance_paths(ambiguous_jobs).tombstones_dir.glob("*.json"))

    asyncio.run(scenario())
