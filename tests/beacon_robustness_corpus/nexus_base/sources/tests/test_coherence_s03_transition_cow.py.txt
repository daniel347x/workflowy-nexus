def async_test(function):
    @wraps(function)
    def wrapper(*args, **kwargs):
        return asyncio.run(function(*args, **kwargs))

    return wrapper

def _make_client(tmp_path: Path, monkeypatch):
    from pydantic import SecretStr

    from workflowy_mcp.client.api_client_core import WorkFlowyClientCore
    from workflowy_mcp.models import APIConfiguration

    jobs_root = tmp_path / "cartographer_jobs"
    jobs_root.mkdir(parents=True, exist_ok=True)
    snapshot_dir = tmp_path / "snapshots"
    snapshot_dir.mkdir(parents=True, exist_ok=True)

    monkeypatch.setattr(
        "workflowy_mcp.carto_job_governance.get_cartographer_jobs_dir",
        lambda: jobs_root,
    )

    def _runtime_subdir(*parts):
        if parts == ("nodes_export_cache",):
            snapshot_dir.mkdir(parents=True, exist_ok=True)
            return snapshot_dir
        return tmp_path.joinpath(*parts)

    monkeypatch.setattr(
        "workflowy_mcp.client.api_client_core.get_runtime_subdir", _runtime_subdir
    )
    config = APIConfiguration(
        api_key=SecretStr("test-key-1234"), base_url="https://workflowy.com/api/v1"
    )
    return WorkFlowyClientCore(config), jobs_root, snapshot_dir


def _populate(client, nodes=BASE_NODES, *, epoch_id: str = "e" * 32, seq: int = 0) -> CacheView:
    return client._publish_lineage_view(
        {"nodes": [dict(n) for n in nodes]},
        epoch_number=1,
        epoch_id=epoch_id,
        adopted_sequence=seq,
        snapshot_timestamp=NOW,
        account_scope_id=client._safe_account_scope_id(),
        clear_dirty=True,
    )


def _node_model(node_id: str, **fields):
    from workflowy_mcp.models import WorkFlowyNode

    return WorkFlowyNode(id=node_id, **fields)


def _seed_lineage(probe, snapshot_dir: Path, nodes=BASE_NODES):
    """Seed a manifest + base candidate so operation boundaries adopt cleanly."""

    paths = probe._coherence_paths()
    manifest = initialize_manifest_if_absent(
        paths, new_bootstrap_manifest(paths.account_root.name, now=NOW)
    )
    candidate = write_snapshot_candidate(
        new_snapshot_candidate_path(snapshot_dir),
        {"nodes": [dict(n) for n in nodes]},
        account_scope_id=paths.account_root.name,
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        incorporated_through_seq=0,
        publication_kind="full_refresh",
    )
    write_manifest(
        paths,
        manifest_with_updates(
            manifest, base_snapshot_path=candidate, coherence_mode="transition"
        ),
        expected_epoch_id=manifest.epoch_id,
        expected_head_seq=0,
    )
    return paths


def _make_lineage_client(tmp_path: Path, monkeypatch, nodes=BASE_NODES):
    probe, _jr, snapshot_dir = _make_client(tmp_path, monkeypatch)
    paths = _seed_lineage(probe, snapshot_dir, nodes)
    client, _jr2, _sd2 = _make_client(tmp_path, monkeypatch)
    return client, paths

def test_stale_expected_view_patch_rebases_onto_current(tmp_path, monkeypatch):
    client, _jr, _sd = _make_client(tmp_path, monkeypatch)
    _populate(client)
    basis = client._cache_view_publisher.capture()

    # Another patch lands first, moving the publisher past the basis.
    client._apply_update_to_nodes_export_cache(_node_model("B", name="beta-2"))

    def _rename_a(working):
        for index, node in enumerate(working["nodes"]):
            if node["id"] == "A":
                patched = dict(node)
                patched["name"] = "alpha-stale-basis"
                working["nodes"][index] = patched
                return

    published = publish_local_cache_view(
        client._cache_view_publisher,
        _rename_a,
        expected_view=basis,  # stale — must rebase, never fail or overwrite B's patch
        reason="stale-basis-test",
    )

    assert published.nodes_by_id["A"]["name"] == "alpha-stale-basis"
    assert published.nodes_by_id["B"]["name"] == "beta-2"  # the racing patch survived

@async_test
async def test_adoption_rebases_tail_onto_racing_cow_patch(tmp_path, monkeypatch):
    """§5.3: a COW patch landing mid-adoption must survive — the journal tail
    rebases onto the patched same-generation view (never discarded)."""

    import os

    from workflowy_mcp.coherence import JournalProducer
    from workflowy_mcp.coherence import adoption as adoption_module

    client, jobs_root, _sd = _make_client(tmp_path, monkeypatch)
    paths = client._coherence_paths()
    manifest = read_manifest(paths, required=True)

    # Lineage: base snapshot + one committed record at seq 1.
    candidate = write_snapshot_candidate(
        new_snapshot_candidate_path(tmp_path / "snapshots"),
        {"nodes": BASE_NODES},
        account_scope_id=manifest.account_scope_id,
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        incorporated_through_seq=0,
        publication_kind="full_refresh",
    )
    write_manifest(
        paths,
        manifest_with_updates(
            manifest,
            base_snapshot_path=candidate,
            coherence_mode="transition",
        ),
        expected_epoch_id=manifest.epoch_id,
        expected_head_seq=0,
    )
    manifest = read_manifest(paths, required=True)
    append_journal_transaction(
        paths,
        operation_kind="test",
        effects=[SemanticEffect.node_create(s02_node("C", name="gamma"))],
        producer=JournalProducer(
            pid=os.getpid(), pid_create_time=1.0, owner_kind="process", ownership_kind="none"
        ),
    )
    client._publish_lineage_view(
        {"nodes": [dict(n) for n in BASE_NODES]},
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        adopted_sequence=0,
        snapshot_timestamp=NOW,
        account_scope_id=manifest.account_scope_id,
    )

    # Inject a COW patch precisely BETWEEN the adoption's basis capture and
    # its CAS attempt (inside the tail replay).  Fires EXACTLY ONCE: the
    # adoption's CAS then fails, retries, and rebases onto the patched view.
    real_replay = adoption_module._replay_tail_onto
    fired = {"done": False}

    def _patch_then_replay(payload, records, **kwargs):
        if not fired["done"]:
            fired["done"] = True

            def _rename(working):
                for index, node in enumerate(working["nodes"]):
                    if node["id"] == "A":
                        patched = dict(node)
                        patched["name"] = "alpha-mid-flight"
                        working["nodes"][index] = patched
                        return

            publish_local_cache_view(
                client._cache_view_publisher, _rename, reason="mid-flight"
            )
        return real_replay(payload, records, **kwargs)

    monkeypatch.setattr(adoption_module, "_replay_tail_onto", _patch_then_replay)

    result = await client._adopt_coherence_head()

    assert result.disposition == "tail_replayed"
    view = client._cache_view_publisher.capture()
    assert view.adopted_sequence == 1
    # BOTH the mid-flight local patch AND the journal effect survived.
    assert view.nodes_by_id["A"]["name"] == "alpha-mid-flight"
    assert view.nodes_by_id["C"]["name"] == "gamma"

def _race_once_at_cas(client, monkeypatch, competing_transform):
    """Fire ``competing_transform`` as a rival COW publication exactly once,
    precisely at the first CAS attempt of the NEXT bridge publication."""

    from workflowy_mcp.coherence import CacheView as _CacheView
    from workflowy_mcp.coherence.cache_view import CacheViewPublisher

    publisher = client._cache_view_publisher
    real_publish = CacheViewPublisher.publish_with_commit
    fired = {"done": False}

    def _racing_publish(new_view, *, expected_current, commit=None):
        if not fired["done"]:
            fired["done"] = True
            current = expected_current
            working = dict(current.nodes_export)
            working["nodes"] = list(current.nodes_export["nodes"])
            competing_transform(working)
            competitor = _CacheView.build(
                epoch_number=current.epoch_number,
                epoch_id=current.epoch_id,
                adopted_sequence=current.adopted_sequence,
                nodes_export=working,
                account_scope_id=current.account_scope_id,
            )
            # The rival wins the race.
            real_publish(publisher, competitor, expected_current=current)
        return real_publish(
            publisher, new_view, expected_current=expected_current, commit=commit
        )

    monkeypatch.setattr(publisher, "publish_with_commit", _racing_publish)


@async_test
@pytest.mark.parametrize("field", ["name", "note"])
async def test_patcher_outcome_false_when_rival_removes_target(tmp_path, monkeypatch, field):
    """Attempt 1 finds the node and loses CAS; the rival removed it; the
    retry sees it absent; the final return must be False and the claimed
    patch must NOT appear in the current view."""

    client, _jr, _sd = _make_client(tmp_path, monkeypatch)
    _populate(client)

    def _remove_a(working):
        working["nodes"] = [n for n in working["nodes"] if n.get("id") != "A"]

    _race_once_at_cas(client, monkeypatch, _remove_a)

    if field == "name":
        result = await client.update_cached_node_name("A", "claimed-patch")
    else:
        result = await client.update_cached_node_note("A", "claimed-patch")

    assert result is False
    view = client._cache_view_publisher.capture()
    assert "A" not in view.nodes_by_id  # no phantom patch

def _rename(node_id: str, new_name: str):
    def _transform(working):
        for index, node in enumerate(working["nodes"]):
            if node.get("id") == node_id:
                patched = dict(node)
                patched["name"] = new_name
                working["nodes"][index] = patched
                return
        raise AssertionError(f"target {node_id} missing")

    return _transform


@async_test
async def test_a1_dirty_sample_taken_under_publisher_mutex(tmp_path, monkeypatch):
    """A1: the post-commit dirty sample must run INSIDE the publisher
    critical section (view swap + dirty commit + dirty sample = ONE atomic
    publication).  The probe sampler records whether the publisher mutex is
    HELD at sample time — under the pre-amendment bridge (sample after
    ``publish_with_commit`` returned) the mutex is FREE and this test fails.
    """

    from workflowy_mcp.coherence.operation_cache import (
        cache_operation_boundary,
        current_operation_cache,
        publish_local_cache_view,
    )

    client, _paths = _make_lineage_client(tmp_path, monkeypatch)
    client._mark_nodes_export_dirty(["A"])
    publisher = client._cache_view_publisher
    probe = {"sampled_outside_mutex": None}

    def _sampler():
        # White-box scheduling probe: a non-blocking acquire SUCCEEDS only
        # when the mutex is free — i.e. when the sample happens outside the
        # publication critical section (the A1 defect).
        acquired = publisher._mutex.acquire(blocking=False)
        if acquired:
            probe["sampled_outside_mutex"] = True
            publisher._mutex.release()
        else:
            probe["sampled_outside_mutex"] = False
        return frozenset(client._nodes_export_dirty_ids)

    async with cache_operation_boundary(client, "op-a1"):
        published = publish_local_cache_view(
            publisher,
            _rename("A", "alpha-a1"),
            reason="a1-probe",
            commit=lambda touched: client._nodes_export_dirty_ids.discard("A"),
            snapshot_dirty_ids=_sampler,
        )
        assert probe["sampled_outside_mutex"] is False
        verifier = current_operation_cache()
        assert verifier.nodes_by_id["A"]["name"] == "alpha-a1"
        # The welded dirty state is the POST-COMMIT one (A discarded).
        assert verifier.dirty_ids == frozenset()
        first_successor_snapshot = verifier

        # An interleaved LATER publication with its own dirty mutation must
        # not rewrite the first successor's captured dirty state.
        publish_local_cache_view(
            publisher,
            _rename("A", "alpha-interleaved"),
            reason="interleaved",
            commit=lambda touched: client._nodes_export_dirty_ids.add("Z"),
            snapshot_dirty_ids=lambda: frozenset(client._nodes_export_dirty_ids),
        )
        assert first_successor_snapshot.dirty_ids == frozenset()
        assert first_successor_snapshot.nodes_by_id["A"]["name"] == "alpha-a1"
        # The task then advances to the interleaved successor carrying ITS
        # own atomically captured dirty state (sequential §6.3 advance).
        later = current_operation_cache()
        assert later.nodes_by_id["A"]["name"] == "alpha-interleaved"
        assert later.dirty_ids == frozenset({"Z"})
        assert published.nodes_by_id["A"]["name"] == "alpha-a1"

@async_test
async def test_a2_foreign_publisher_publication_never_advances_snapshot(
    tmp_path, monkeypatch
):
    """A2/B1.2: two same-account clients carry identical (epoch, sequence)
    markers from the shared manifest but own SEPARATE COW chains.  A COW
    publication through client B must never advance the task-local snapshot
    pinned to client A — equal markers do not make a foreign chain a valid
    successor.  Fails if the owner check in _advance_task_snapshot is
    removed.
    """

    from workflowy_mcp.coherence.operation_cache import (
        cache_operation_boundary,
        current_operation_cache,
    )

    # Two clients, same credentials -> same account scope, DISTINCT
    # publishers, both warm-started from the SAME seeded manifest/base
    # (identical markers, separate in-memory COW chains).
    probe, _jr, snapshot_dir = _make_client(tmp_path, monkeypatch)
    _seed_lineage(probe, snapshot_dir)
    client_a, _jr2, _sd2 = _make_client(tmp_path, monkeypatch)
    client_b, _jr3, _sd3 = _make_client(tmp_path, monkeypatch)
    assert client_a._cache_view_publisher is not client_b._cache_view_publisher
    markers_a = (
        client_a._cache_view_publisher.capture().epoch_id,
        client_a._cache_view_publisher.capture().adopted_sequence,
    )
    markers_b = (
        client_b._cache_view_publisher.capture().epoch_id,
        client_b._cache_view_publisher.capture().adopted_sequence,
    )
    assert markers_a == markers_b  # identical markers, separate chains

    from workflowy_mcp.coherence.operation_cache import OperationCacheSnapshot

    async with cache_operation_boundary(client_a, "op-a"):
        snapshot_a = current_operation_cache()
        assert isinstance(snapshot_a, OperationCacheSnapshot)
        view_a_before = snapshot_a.view
        # Publish a COW successor through client B (same markers).
        client_b.publish_local_cache_view(
            _rename("A", "b-write"), reason="b-publication"
        )
        # Client A's task-local snapshot is UNCHANGED: no advance, and its
        # pinned view/content/owner are internally consistent.
        verifier = current_operation_cache()
        assert verifier is snapshot_a
        assert verifier.view is view_a_before
        assert verifier.nodes_by_id["A"]["name"] == "alpha"
        assert verifier.owner is snapshot_a.owner
        # Client B published normally on its own authority.
        assert (
            client_b._cache_view_publisher.capture().nodes_by_id["A"]["name"]
            == "b-write"
        )
        # Client A's publisher never received B's write.
        assert (
            client_a._cache_view_publisher.capture().nodes_by_id["A"]["name"]
            == "alpha"
        )
