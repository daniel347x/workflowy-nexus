def _git(cwd: Path, *args: str) -> None:
    subprocess.run(
        ["git", *args], cwd=str(cwd), check=True, capture_output=True, text=True
    )


@pytest.fixture
def temp_repo(tmp_path: Path) -> Path:
    repo = tmp_path / "repo"
    repo.mkdir()
    _git(repo, "init", "-q")
    _git(repo, "config", "user.email", "test@example.com")
    _git(repo, "config", "user.name", "Test")
    (repo / "a.txt").write_text("hello\n", encoding="utf-8")
    (repo / ".gitignore").write_text(
        "ignored.txt\ntests/.last_run.json\ntests/.runs/\ntests/.last_run.log\n",
        encoding="utf-8",
    )
    _git(repo, "add", "-A")
    _git(repo, "commit", "-q", "-m", "init")
    return repo


def _green_receipt(repo: Path) -> dict:
    commit = rt.compute_commit(repo)
    tree = rt.compute_tree_hash(repo)
    return {
        "schema": rt.SCHEMA,
        "commit": commit,
        "tree_hash": tree,
        "dirty": tree != rt.head_tree_hash(repo),
        "env": "wsl-ext4",
        "python": "3.12.3",
        "platform": "Linux-test",
        "n": 8,
        "dist": "loadfile",
        "tier": "full",
        "files": [],
        "command": "python -m pytest",
        "passed": 10,
        "failed": 0,
        "skipped": 0,
        "errors": 0,
        "xfailed": 0,
        "duration_s": 1.0,
        "timestamp_utc": "2026-01-01T00:00:00+00:00",
        "pytest_exit": 0,
        "failed_tests": [],
        "duration_parallel_s": 1.0,
        "duration_tail_s": 0.0,
    }

# --- receipt schema + signing ----------------------------------------------
def test_receipt_has_all_schema_fields(temp_repo: Path) -> None:
    rt.write_receipt(temp_repo, _green_receipt(temp_repo))
    loaded = rt.load_receipt(temp_repo)
    for field in RECEIPT_FIELDS:
        assert field in loaded, field
    assert loaded["schema"] == 1
    assert loaded["receipt_sha256"] == rt.receipt_digest(loaded)

# --- dirty-tree receipts: refused by default, admitted only on request -------
def _dirty_green_receipt(repo: Path) -> dict:
    """A green receipt written from an UNCOMMITTED tree (a.txt edited in place)."""
    (repo / "a.txt").write_text("edited but not committed\n", encoding="utf-8")
    receipt = _green_receipt(repo)  # recomputes commit + tree_hash + dirty
    assert receipt["dirty"] is True
    return receipt


def test_verify_refuses_dirty_receipt_by_default(temp_repo: Path) -> None:
    rt.write_receipt(temp_repo, _dirty_green_receipt(temp_repo))
    ok, reason = rt.verify(temp_repo, required_tier="full")
    assert not ok and "dirty receipt" in reason and "UNCOMMITTED" in reason


def test_verify_admits_dirty_receipt_only_with_allow_dirty(temp_repo: Path) -> None:
    rt.write_receipt(temp_repo, _dirty_green_receipt(temp_repo))
    ok, reason = rt.verify(temp_repo, required_tier="full", allow_dirty=True)
    assert ok, reason
    assert "DIRTY tree" in reason  # the PASS line says which claim it is making


def test_verify_pass_line_names_tree_and_clean_state(temp_repo: Path) -> None:
    rt.write_receipt(temp_repo, _green_receipt(temp_repo))
    ok, reason = rt.verify(temp_repo, required_tier="full")
    assert ok, reason
    assert "clean tree" in reason
    assert rt.compute_tree_hash(temp_repo)[:8] in reason


def test_tree_hash_is_a_git_tree_object_and_survives_commit(temp_repo: Path) -> None:
    """The bridge the doctrine relies on: committing the same bytes leaves
    `tree_hash` unchanged and equal to `git rev-parse HEAD^{tree}`."""
    (temp_repo / "a.txt").write_text("edited\n", encoding="utf-8")
    dirty_tree = rt.compute_tree_hash(temp_repo)
    assert dirty_tree != rt.head_tree_hash(temp_repo)
    _git(temp_repo, "add", "-A")
    _git(temp_repo, "commit", "-q", "-m", "edit")
    assert rt.compute_tree_hash(temp_repo) == dirty_tree
    assert rt.head_tree_hash(temp_repo) == dirty_tree

def test_window_keeps_the_newest_n_and_mirrors_the_newest(temp_repo: Path) -> None:
    for i in range(12):
        receipt = _green_receipt(temp_repo)
        receipt["timestamp_utc"] = f"2026-01-01T00:00:{i:02d}+00:00"
        rt.archive_run(temp_repo, receipt, f"log {i}\n", keep=10)
    window = rt.runs_dir(temp_repo)
    names = sorted(p.name for p in window.glob("*.json"))
    assert len(names) == 10, names
    assert names[0].startswith("20260101T000002"), names[0]  # oldest two pruned
    assert len(sorted(window.glob("*.log"))) == 10  # logs pruned with them
    mirror = rt.load_receipt(temp_repo)
    assert mirror is not None and mirror["timestamp_utc"].endswith("00:11+00:00")


def test_verify_finds_newest_qualifying_not_merely_the_last(temp_repo: Path) -> None:
    """A later --files run must not destroy the citable full-tier receipt."""
    full = _green_receipt(temp_repo)
    full["timestamp_utc"] = "2026-01-01T00:00:01+00:00"
    rt.archive_run(temp_repo, full)
    later = _green_receipt(temp_repo)
    later["tier"] = "files"
    later["timestamp_utc"] = "2026-01-01T00:00:02+00:00"
    rt.archive_run(temp_repo, later)
    assert rt.load_receipt(temp_repo)["tier"] == "files"  # mirror is the newest
    ok, reason = rt.verify(temp_repo, required_tier="full")
    assert ok, reason
    assert "tier full" in reason
    assert "20260101T000001" in reason  # the full-tier window entry, not the newest


def test_verify_reports_the_newest_failure_when_nothing_qualifies(
    temp_repo: Path,
) -> None:
    stale = _green_receipt(temp_repo)
    stale["tier"] = "files"
    stale["timestamp_utc"] = "2026-01-01T00:00:01+00:00"
    rt.archive_run(temp_repo, stale)
    red = _green_receipt(temp_repo)

    red["pytest_exit"] = 1

    red["timestamp_utc"] = "2026-01-01T00:00:02+00:00"

    ok, reason = rt.verify(temp_repo, required_tier="full")
    assert not ok
    assert reason.startswith("red receipt — failed=3")
    assert "tests/test_c.py::test_three" in reason  # a red receipt names its own failures
    assert "checked" in reason


# --- Windows -> WSL routing: the composed command must carry every arg -----
def test_compose_wsl_command_carries_every_passthrough_arg() -> None:
    """Regression guard for the class of bug that lost every flag once: the
    command is built from a list, and each passthrough token must survive."""
    argv = ["--files", "tests\\test_a.py", "tests/test_b.py", "-n", "4", "--keep", "3"]
    cmd = rt.compose_wsl_command("/mnt/e/repo", argv)
    assert cmd.startswith("cd /mnt/e/repo && python3 scripts/run_tests.py ")
    for token in ["--files", "tests/test_a.py", "tests/test_b.py", "-n", "4", "--keep", "3"]:
        assert token in cmd.split(), (token, cmd)
    assert "\\" not in cmd  # Windows path separators normalised for bash


def test_compose_wsl_command_with_no_args_is_the_bare_full_run() -> None:
    assert rt.compose_wsl_command("/mnt/e/repo", []) == (

# --- clone file list: clone == hashed tree by construction ------------------
@pytest.mark.skipif(
    shutil.which("rsync") is None,
    reason="rsync not on PATH — the ext4 clone path is WSL-only and never runs here",
)
def test_sync_prunes_stale_and_seeds_lockfile(tmp_path: Path) -> None:
    repo = tmp_path / "repo"
    repo.mkdir()
    _git(repo, "init", "-q")
    _git(repo, "config", "user.email", "test@example.com")
    _git(repo, "config", "user.name", "Test")
    (repo / "a.txt").write_text("a\n", encoding="utf-8")
    (repo / ".gitignore").write_text("uv.lock\n", encoding="utf-8")
    (repo / "uv.lock").write_text("LOCK\n", encoding="utf-8")
    (repo / "sub").mkdir()
    (repo / "sub" / "b.py").write_text("b = 1\n", encoding="utf-8")
    _git(repo, "add", "-A")
    _git(repo, "commit", "-q", "-m", "init")

    clone = tmp_path / "clone"
    clone.mkdir()
    (clone / ".venv").mkdir()  # clone-local infra, must survive
    (clone / ".venv" / "python").write_text("py", encoding="utf-8")
    (clone / "stale.txt").write_text("stale", encoding="utf-8")
    (clone / "sub").mkdir()
    (clone / "sub" / "stale.py").write_text("stale", encoding="utf-8")

    rt.sync_to_clone(repo, clone)
    assert (clone / "a.txt").is_file()
    assert (clone / "sub" / "b.py").is_file()
    assert (clone / "uv.lock").read_text(encoding="utf-8") == "LOCK\n"  # seeded
    assert (clone / ".venv" / "python").is_file()  # infra preserved
    assert not (clone / "stale.txt").exists()  # extraneous pruned
    assert not (clone / "sub" / "stale.py").exists()

    # A file removed in the repo disappears from the clone on the next sync.
    (repo / "a.txt").unlink()
    _git(repo, "add", "-A")
    _git(repo, "commit", "-q", "-m", "rm")
    rt.sync_to_clone(repo, clone)
    assert not (clone / "a.txt").exists()
    assert (clone / ".venv" / "python").is_file()

# --- A2: flock on the clone -------------------------------------------------
def test_clone_lock_is_exclusive(tmp_path: Path) -> None:
    if rt.fcntl is None:
        pytest.skip("fcntl unavailable on this platform")
    clone = tmp_path / "clone"
    clone.mkdir()
    first = rt.acquire_clone_lock(clone)
    assert (clone / ".lock").is_file()
    second = os.open(str(clone / ".lock"), os.O_CREAT | os.O_RDWR)
    try:
        with pytest.raises(OSError) as exc:
            rt.fcntl.flock(second, rt.fcntl.LOCK_EX | rt.fcntl.LOCK_NB)
        assert exc.value.errno in (errno.EAGAIN, errno.EACCES)
    finally:
        os.close(second)
        rt.fcntl.flock(first, rt.fcntl.LOCK_UN)
        os.close(first)


def test_clone_lock_second_caller_waits_then_proceeds(tmp_path: Path, capsys) -> None:
    if rt.fcntl is None:
        pytest.skip("fcntl unavailable on this platform")
    clone = tmp_path / "clone"
    clone.mkdir()
    first = rt.acquire_clone_lock(clone)
    acquired: dict[str, int] = {}

    def second() -> None:
        acquired["fd"] = rt.acquire_clone_lock(clone)

    worker = threading.Thread(target=second)
    worker.start()
    try:
        time.sleep(0.3)
        assert worker.is_alive(), "the second caller must block while held"
        rt.fcntl.flock(first, rt.fcntl.LOCK_UN)
        os.close(first)
        worker.join(timeout=5)
        assert not worker.is_alive()
        assert "waiting" in capsys.readouterr().err
    finally:
        if "fd" in acquired:
            rt.fcntl.flock(acquired["fd"], rt.fcntl.LOCK_UN)
            os.close(acquired["fd"])


# --- the `--files` .js branch ([ap:SFK1C9]): partition, node, exit-code fold ---
def _write_js_harness(repo: Path, name: str, exit_code: int) -> str:
    """A minimal dependency-free harness under tests/js/; returns its rel path."""
    js_dir = repo / "tests" / "js"
    js_dir.mkdir(parents=True, exist_ok=True)
    rel = f"tests/js/{name}"
    (js_dir / name).write_text(
        f"console.log('harness {name}');\nprocess.exit({exit_code});\n",
        encoding="utf-8",
    )
    return rel

def test_fold_js_results_counts_exit_codes_not_summary_lines() -> None:
    """JS results are exit-code accounting: one harness = one passed/failed
    unit, failing harnesses NAMED in failed_tests, first nonzero exit wins."""
    counts = {"passed": 5, "failed": 0, "skipped": 2, "errors": 0, "xfailed": 0}
    failed_tests = ["tests/test_a.py::test_x"]  # a pytest failure stays first
    js_exit = rt.fold_js_results(
        counts,
        failed_tests,
        [
            ("tests/js/h1.js", 0),
            ("tests/js/h2.js", 1),
            ("tests/js/h3.js", 2),
        ],
    )
    assert counts == {"passed": 6, "failed": 2, "skipped": 2, "errors": 0, "xfailed": 0}
    assert js_exit == 1  # the FIRST nonzero code, not the last
    assert failed_tests == [
        "tests/test_a.py::test_x",
        "tests/js/h2.js (node exit 1)",
        "tests/js/h3.js (node exit 2)",
    ]


def test_fold_js_results_all_green_leaves_everything_quiet() -> None:
    counts = {"passed": 0, "failed": 0, "skipped": 0, "errors": 0, "xfailed": 0}
    failed_tests: list[str] = []
    assert rt.fold_js_results(counts, failed_tests, [("tests/js/h1.js", 0)]) == 0
    assert counts["passed"] == 1 and counts["failed"] == 0
    assert failed_tests == []


@pytest.mark.skipif(shutil.which("node") is None, reason="node not on PATH")
def test_run_js_harness_captures_exit_code_and_output(tmp_path: Path) -> None:
    node = shutil.which("node")
    (tmp_path / "ok.js").write_text(
        "console.log('ALL PASS');process.exit(0);\n", encoding="utf-8"
    )
    (tmp_path / "bad.js").write_text(
        "console.error('1 FAILURES');process.exit(3);\n", encoding="utf-8"
    )
    code, out, dur = rt.run_js_harness(node, "ok.js", tmp_path)
    assert code == 0 and "ALL PASS" in out and dur >= 0.0
    code, out, _ = rt.run_js_harness(node, "bad.js", tmp_path)
    assert code == 3 and "1 FAILURES" in out  # the REAL node exit code survives

    log2 = rt.format_run_log(
        "wsl-ext4", "files", "pytest cmd", "3 passed in 0.1s\n", None, ""
    )
    assert "# pass 1 (parallel): pytest cmd" in log2
    assert "3 passed in 0.1s" in log2
    assert "js harness" not in log2


@pytest.mark.skipif(shutil.which("node") is None, reason="node not on PATH")
def test_main_files_js_only_never_starts_pytest(
    temp_repo: Path, monkeypatch, capsys
) -> None:
    """The trap this branch must not fall into: an empty pytest target list
    means 'collect the WHOLE suite', so a .js-only --files run must skip the
    pytest pass entirely and still write one folded receipt."""
    # main() derives the repo from its own file location (inside the ext4
    # clone, which is not a git repo) — point it at the temp repo instead.
    monkeypatch.setattr(rt, "repo_root", lambda: temp_repo)
    rel = _write_js_harness(temp_repo, "ok.js", 0)

    def _no_pytest(*args, **kwargs):
        raise AssertionError("pytest must not run for a .js-only --files set")

    monkeypatch.setattr(rt, "run_pytest", _no_pytest)
    assert rt.main(["--in-place", "--files", rel]) == 0
    receipt = rt.load_receipt(temp_repo)
    assert receipt["tier"] == "files"
    assert receipt["files"] == [rel]
    assert receipt["pytest_exit"] == 0
    assert receipt["passed"] == 1 and receipt["failed"] == 0  # exit-code fold
    assert receipt["failed_tests"] == []
    assert rel in receipt["command"] and "pytest" not in receipt["command"]
    log = (temp_repo / rt.LOG_REL).read_text(encoding="utf-8")
    assert "# js harness:" in log and "pass 1" not in log

@pytest.mark.skipif(shutil.which("node") is None, reason="node not on PATH")
def test_main_files_mixed_py_and_js_partitions_targets(
    temp_repo: Path, monkeypatch, capsys
) -> None:
    """Mixed set: pytest sees ONLY the .py side; node runs the .js side; one
    receipt accounts for both."""
    monkeypatch.setattr(rt, "repo_root", lambda: temp_repo)
    rel = _write_js_harness(temp_repo, "ok.js", 0)
    calls: list[list[str]] = []

    def fake_run_pytest(py, args, cwd, extra_env=None):
        calls.append(list(args))
        return 0, "2 passed in 0.01s\n", 0.01

    monkeypatch.setattr(rt, "run_pytest", fake_run_pytest)
    assert rt.main(["--in-place", "--files", "tests/test_a.py", rel]) == 0
    assert len(calls) == 1
    assert "tests/test_a.py" in calls[0]
    assert rel not in calls[0]  # the .js target never reaches pytest
    receipt = rt.load_receipt(temp_repo)
    assert receipt["passed"] == 3  # 2 pytest + 1 node harness
    assert receipt["failed"] == 0 and receipt["pytest_exit"] == 0
    assert "pytest" in receipt["command"] and rel in receipt["command"]
    assert " ; " in receipt["command"]  # both invocations named


def test_main_files_js_without_node_fails_closed(
    temp_repo: Path, monkeypatch, capsys
) -> None:
    """No node on PATH is a loud red receipt (exit 127, harness named), never
    a silent skip — the JS mutation check must not vanish quietly."""
    monkeypatch.setattr(rt, "repo_root", lambda: temp_repo)
    rel = _write_js_harness(temp_repo, "ok.js", 0)
    monkeypatch.setattr(rt, "shutil", types.SimpleNamespace(which=lambda name: None))
    assert rt.main(["--in-place", "--files", rel]) == 1
    assert "not found" in capsys.readouterr().err.lower()
    receipt = rt.load_receipt(temp_repo)
    assert receipt["failed"] == 1 and receipt["passed"] == 0
    assert receipt["pytest_exit"] == rt.NODE_NOT_FOUND_EXIT == 127
    assert receipt["failed_tests"] == [f"{rel} (node exit 127)"]
