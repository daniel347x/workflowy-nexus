"""Stage2-07 (Locked Work Order §6.7) — REAL checkpoint↔compat-patch
contention on the single RefreshPublicationClaim authority
(PAIR-014 / CF-S04A4-03).

REAL OS processes on one coherence root, both claim-first orders, plus the
exact claim-kind verification CF-S04A4-03 demands be OBSERVED (never
inferred from a unit test):

  ORDER A (checkpoint first): the checkpoint owner parks mid-rotation
  holding the claim; the compat-patch actor NEVER dedupes with it — it
  waits through the contended acquisition path (acknowledged by its
  injected sleep seam), then publishes E+2 after the checkpoint's E+1.
  Claim-kind transitions on the fixed artifact are observed at each stage.

  ORDER B (compat first): the compat owner parks at its candidate write
  holding the claim; the checkpoint requester DEFERS WITHOUT WAITING
  (``deferred_publisher_busy`` — defer-not-queue), and a later checkpoint
  publishes E+2 once the compat's E+1 is durable.

  KIND BINDING (CF-S04A4-03): two REAL subprocess observations on the
  single authority — (a) a checkpoint-claim owner attempting a
  ``compat_patch`` publish is rejected with ``CoherenceSchemaError``
  ("compat_patch publication requires a compat_patch claim"); (b) a
  compat-claim owner attempting a ``checkpoint`` publish is rejected with
  ``CoherenceSchemaError`` ("checkpoint publication requires a checkpoint
  claim").  The epoch identity never advances in either case.

Every actor logs its real PID; the parent asserts pairwise distinctness.
"""

def _based_root(root, *, records: int = 3):
    paths = make_coherence_root(root, ACCOUNT_A)
    snapshot_dir = root / "nodes_export_cache"
    base = s04_write_base_snapshot(
        paths,
        snapshot_dir,
        [s02_node("base")],
        epoch_number=0,
        epoch_id="b" * 32,
    )
    s04_install_base_manifest(paths, base_snapshot_path=base, epoch_id="b" * 32)
    if records:
        s04_append_n(paths, records)
    return paths, snapshot_dir

_COMPAT_WAITER = (
    _COMMON
    + """

from workflowy_mcp.coherence import run_compat_patch

touch(root / "pid_compat", str(os.getpid()))

_contended = {"done": False}

def test_checkpoint_first_compat_waits_then_publishes(tmp_path):
    """ORDER A: the checkpoint owner holds the claim (kind observed:
    ``checkpoint``); the compat patch waits through the contended path and
    publishes E+2 after the checkpoint's E+1 (kind observed:
    ``compat_patch``)."""
    root = tmp_path / "share"
    paths, _sd = _based_root(root, records=3)

    checkpoint = launch_actor(_CHECKPOINT_HOLDER, str(root), str(REPO))
    compat = None
    try:
        assert wait_for_file(root / "ck_holding", timeout=120)
        # The checkpoint provably holds the claim before the compat launches.
        claim = read_refresh_claim(paths)
        assert claim is not None and not claim.is_terminal
        assert claim.refresh_kind == "checkpoint"
        assert claim.owner_pid == int((root / "ck_holding").read_text().strip())
        compat = launch_actor(_COMPAT_WAITER, str(root), str(REPO))
        assert wait_for_file(root / "compat_contending", timeout=120)
        touch(root / "release_ck")

        r_ck = read_result(root / "ck_result.json", timeout=120)
        assert r_ck == {"status": "published", "incorporated": 3}, r_ck
        r_compat = read_result(root / "compat_result.json", timeout=180)
        assert r_compat["status"] == "published"
        assert r_compat["epoch_number"] == 2

        manifest = read_manifest(paths, required=True)
        assert manifest.epoch_number == 2
        # Claim-kind transition on the fixed artifact, observed.
        claim = read_refresh_claim(paths)
        assert claim is not None and claim.refresh_kind == "compat_patch"
        # The compat patch really landed.
        payload = materialize_current_payload(paths, manifest).payload
        by_id = {n["id"]: n for n in payload["nodes"]}
        assert by_id["base"]["name"] == "patched-by-compat"

        pids = collect_distinct_pids(root, ["checkpoint", "compat"])
        assert len(set(pids.values())) == 2
    finally:
        finish_actor(checkpoint, label="checkpoint holder")
        if compat is not None:
            finish_actor(compat, label="compat waiter")

def test_compat_first_checkpoint_defers_then_publishes(tmp_path):
    """ORDER B: the compat owner holds the claim (kind observed:
    ``compat_patch``); the checkpoint requester defers WITHOUT waiting
    (defer-not-queue); the compat publishes E+1 and a later checkpoint
    publishes E+2."""
    root = tmp_path / "share"
    paths, snapshot_dir = _based_root(root, records=3)

    compat = launch_actor(_COMPAT_HOLDER, str(root), str(REPO))
    checkpoint = None
    try:
        assert wait_for_file(root / "compat_holding", timeout=120)
        # The compat provably holds the claim before the checkpoint launches.
        claim = read_refresh_claim(paths)
        assert claim is not None and not claim.is_terminal
        assert claim.refresh_kind == "compat_patch"
        assert claim.owner_pid == int((root / "compat_holding").read_text().strip())
        checkpoint = launch_actor(_CHECKPOINT_DEFER, str(root), str(REPO))
        r_defer = read_result(root / "ck_defer_result.json", timeout=120)
        # Defer-not-queue: the checkpoint refuses to contend and does NOT publish.
        assert r_defer == {"status": "deferred_publisher_busy"}, r_defer
        touch(root / "release_compat")

        r_compat = read_result(root / "compat_result.json", timeout=120)
        assert r_compat["status"] == "published"
        assert r_compat["epoch_number"] == 1

        # A later checkpoint over the post-compat tail publishes E+2.
        s04_append_n(paths, 2)
        import asyncio

        followup = asyncio.run(
            run_checkpoint_rotation(
                paths,
                snapshot_dir=snapshot_dir,
                config=s04_checkpoint_config(record_threshold=2),
            )
        )
        assert followup.status == "published", followup.warnings
        manifest = read_manifest(paths, required=True)
        assert manifest.epoch_number == 2

        pids = collect_distinct_pids(root, ["compat", "checkpoint"])
        assert len(set(pids.values())) == 2
    finally:
        finish_actor(compat, label="compat holder")
        if checkpoint is not None:
            finish_actor(checkpoint, label="checkpoint defer actor")

_KIND_BINDING = (
    _COMMON
    + """
from stage2_harness import write_result
from coherence_harness import s02_node
from workflowy_mcp.coherence.checkpoint import _acquire_checkpoint_claim
from workflowy_mcp.coherence.epoch_publisher import publish_epoch
from workflowy_mcp.coherence.errors import CoherenceSchemaError
from workflowy_mcp.coherence.refresh_authority import _acquire_or_join
from workflowy_mcp.coherence.schema import read_manifest
from workflowy_mcp.coherence.snapshot_store import (
    new_snapshot_candidate_path,
    write_snapshot_candidate,
)

mode = sys.argv[3]  # "checkpoint" | "compat"
touch(root / f"pid_kind_{mode}", str(os.getpid()))
ACCOUNT = 'a' * 32
manifest = read_manifest(paths, required=True)
wrong_kind = "compat_patch" if mode == "checkpoint" else "checkpoint"

if os.environ.get("STAGE2_NEUTRALIZE_KIND_BINDING"):
    # SENSITIVITY-ONLY (never set in the shipped test): simulate the absent
    # kind binding — the persisted claim reads back with the PUBLICATION's
    # kind, so the borrowed-claim publish lands and the rejected assertion
    # fails.
    import workflowy_mcp.coherence.epoch_publisher as ep
    from workflowy_mcp.coherence.schema import claim_with_updates
    _real_read = ep.read_refresh_claim
    def _as_kind(paths):
        claim = _real_read(paths)
        if claim is not None:
            claim = claim_with_updates(claim, refresh_kind=wrong_kind)
        return claim
    ep.read_refresh_claim = _as_kind

# Acquire a genuine claim of THIS mode's kind through the real authority.
if mode == "checkpoint":
    decision, claim = asyncio.run(_acquire_checkpoint_claim(
        paths, due_head=manifest.journal_head_seq, fs_ops=None))
    assert decision == "owner"
else:
    async def _sleep(seconds):
        await asyncio.sleep(seconds)
    acquisition = asyncio.run(_acquire_or_join(
        paths, refresh_kind="compat_patch",
        snapshot_dir=root / "nodes_export_cache", sleep=_sleep,
        poll_seconds=0.05, max_wait_seconds=30, cancel_check=None, fs_ops=None))
    assert acquisition.decision == "owner"
    claim = acquisition.claim

# A durable candidate carrying the PUBLICATION kind (only reached when the
# kind binding is neutralized — in the shipped test the kind check fires
# first and the candidate is never validated).
candidate = write_snapshot_candidate(
    new_snapshot_candidate_path(root / "nodes_export_cache"),
    {"nodes": [s02_node("base")]},
    account_scope_id=ACCOUNT,
    epoch_number=claim.target_epoch_number or (manifest.epoch_number + 1),
    epoch_id=str(claim.target_epoch_id),
    incorporated_through_seq=manifest.journal_head_seq,
    publication_kind=wrong_kind,
)

result = {"rejected": False, "error_type": None, "message": None, "published": False}
try:
    publication = asyncio.run(publish_epoch(
        paths=paths,
        claim_id=claim.claim_id,
        basis_epoch_id=manifest.epoch_id,
        candidate_snapshot_path=candidate,
        incorporated_through_seq=manifest.journal_head_seq,
        publication_kind=wrong_kind,
        refresh_cut=None,
        clear_hard_through_generation=None,
        clear_external_through_generation=None,
    ))
    result["published"] = True
except CoherenceSchemaError as exc:
    result["rejected"] = True
    result["error_type"] = "CoherenceSchemaError"
    result["message"] = str(exc)
write_result(root / f"kind_{mode}_result.json", result)
"""
)


def _run_kind_binding(root, mode: str, env: dict | None = None):
    paths, _sd = _based_root(root, records=3)
    actor = launch_actor(_KIND_BINDING, str(root), str(REPO), mode, env=env)
    try:
        result = read_result(root / f"kind_{mode}_result.json", timeout=120)
    finally:
        finish_actor(actor, label=f"kind-binding actor ({mode})")
    return paths, result

def test_exact_claim_kind_verification_real_processes(tmp_path):
    """CF-S04A4-03 discharge: on the single RefreshPublicationClaim
    authority, a checkpoint claim can never be borrowed for a compat-patch
    publication and a compat claim can never be borrowed for a checkpoint
    publication — each rejected with CoherenceSchemaError and ZERO epoch
    movement, OBSERVED in real subprocesses."""
    ck_root = tmp_path / "checkpoint_case"
    paths_ck, r_ck = _run_kind_binding(ck_root, "checkpoint")
    assert r_ck["rejected"] is True
    assert r_ck["error_type"] == "CoherenceSchemaError"
    assert "compat_patch publication requires a compat_patch" in r_ck["message"]
    assert r_ck["published"] is False
    manifest = read_manifest(paths_ck, required=True)
    assert manifest.epoch_number == 0  # no epoch movement
    collect_distinct_pids(ck_root, ["kind_checkpoint"])

    compat_root = tmp_path / "compat_case"
    paths_cp, r_cp = _run_kind_binding(compat_root, "compat")
    assert r_cp["rejected"] is True
    assert r_cp["error_type"] == "CoherenceSchemaError"
    assert "checkpoint publication requires a checkpoint" in r_cp["message"]
    assert r_cp["published"] is False
    manifest = read_manifest(paths_cp, required=True)
    assert manifest.epoch_number == 0  # no epoch movement
    collect_distinct_pids(compat_root, ["kind_compat"])
