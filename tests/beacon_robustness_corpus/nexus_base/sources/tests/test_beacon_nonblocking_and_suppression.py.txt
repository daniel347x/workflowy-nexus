"""Non-Blocking Beacon Writes (NBBW) + global #wfy-suppress for beacon symbol resolution.

Regression locks for program [ap:3ACUQS]:

(i)   a beacon.tags.ensure batch runs its file parsing/writing OFF the event
      loop: a heartbeat coroutine keeps ticking while every parse is slow;
(iii) an outside write between the JIT plan and the commit writes ZERO bytes;
      cancelling a batch mid-commit releases the Q3 lease only AFTER the
      in-flight write has finished;
(iv)  Dan's ruling [ap:7ORCPL]: a symbol present both under a #wfy-suppress'd
      ancestor and elsewhere resolves to the unsuppressed one only (no
      ambiguity) for beacon.resolve, beacon.snippet and a batch write; a
      suppressed-only item is never written; no tab -> loud failure.

These run the REAL client seams (WorkFlowyClientNexus preflight/plan/commit and
the real symbol resolver) over a synthetic cache and real temp files; only the
cache accessor, the suppress-config read and the lease governor are stubbed.
"""

def _fn_ids(prefix, names):
    return {name: f"{prefix}-0000-0000-0000-{i:012d}" for i, name in enumerate(names, start=1)}


class _RealSeamClient(acn.WorkFlowyClientNexus):
    """The REAL beacon seams over an in-memory cache (no network, no __init__)."""

    def __init__(self, nodes):  # noqa: D401 - deliberately skips the real ctor
        self.nodes = nodes

    def _get_nodes_export_cache_nodes(self):
        return list(self.nodes)


def _install(monkeypatch, nodes, *, suppress=()):
    client = _RealSeamClient(nodes)

    async def _export(_client, node_id=None, use_cache=True, force_refresh=False, **_kw):
        return {"nodes": list(nodes)}

    async def _suppress():
        return list(suppress)

    monkeypatch.setattr(acn, "export_nodes_impl", _export)
    monkeypatch.setattr(srv, "get_client", lambda: client)
    monkeypatch.setattr(srv, "_rate_limiter", None)
    monkeypatch.setattr(srv, "_beacon_symbol_suppress_tags", _suppress)
    return client


def _observe_leases(monkeypatch, events=None):
    state = SimpleNamespace(acquisitions=[], releases=[], held=False)

    async def acquire(targets, **_kw):
        state.acquisitions.append(list(targets))
        state.held = True
        return [{"lease_id": "nbbw-test"}]

    async def release(leases, *, primary_error=None):
        if events is not None:
            events.append("release")
        state.releases.append(primary_error)
        state.held = False

    monkeypatch.setattr(srv, "acquire_q3_leases_async", acquire)
    monkeypatch.setattr(srv, "_release_q3_leases_preserving_primary", release)
    return state


def _spec(symbols, tags=("nbbw",), roots=(_ROOT,), mode="ensure_beacons"):

        root_uuids=list(roots),
        root_uuid=None,

    return spec

# ---------------------------------------------------------------------------
# (i) the event loop keeps running while every parse is slow
# ---------------------------------------------------------------------------


def test_batch_parsing_runs_off_the_event_loop(tmp_path, monkeypatch):
    path, ids, nodes = _one_file_fixture(tmp_path)
    _install(monkeypatch, nodes)
    leases = _observe_leases(monkeypatch)
    real_parse = nmc._parse_disk_beacons_for_language
    parse_threads = []

    def slow_parse(language, source_path):
        # Stand-in for a multi-MB file: every whole-file parse takes 0.35 s.
        parse_threads.append(threading.current_thread() is threading.main_thread())
        time.sleep(0.35)
        return real_parse(language, source_path)

    monkeypatch.setattr(nmc, "_parse_disk_beacons_for_language", slow_parse)

    async def main():
        gaps = []
        done = asyncio.Event()

        async def heartbeat():
            last = time.perf_counter()
            while not done.is_set():
                await asyncio.sleep(0.01)
                now = time.perf_counter()
                gaps.append(now - last)
                last = now

        hb = asyncio.create_task(heartbeat())
        await asyncio.sleep(0.03)
        res = await srv._lightning_rod_mutate_beacons(spec=_spec(list(ids)))
        done.set()
        await hb
        return res, gaps

    res, gaps = asyncio.run(main())
    assert res["success"] is True, res
    assert [r["status"] for r in res["results"]] == ["committed"] * 3
    # preflight + plan + commit parse per item, all in worker threads.
    assert len(parse_threads) >= 9 and not any(parse_threads)
    # 9+ blocking parses of 0.35 s each; the loop never stalled for one.
    assert max(gaps) < 0.3, f"event loop stalled for {max(gaps):.3f}s"
    assert leases.releases == [None]
    text = path.read_text(encoding="utf-8")
    assert text.count("slice_labels=old,nbbw,") == 3

# ---------------------------------------------------------------------------
# cancellation mid-commit: the lease outlives the in-flight write
# ---------------------------------------------------------------------------


def test_cancel_mid_commit_releases_lease_only_after_the_write(tmp_path, monkeypatch):
    path, ids, nodes = _one_file_fixture(tmp_path, names=("alpha", "beta"))
    _install(monkeypatch, nodes)
    events = []
    leases = _observe_leases(monkeypatch, events)
    real_helper = nmc.update_beacon_from_node_python
    entered = threading.Event()
    proceed = threading.Event()

    def gated_helper(*args, **kwargs):
        entered.set()
        assert proceed.wait(10), "test never released the gated writer"
        result = real_helper(*args, **kwargs)
        events.append("write_done")
        return result

    monkeypatch.setattr(nmc, "update_beacon_from_node_python", gated_helper)

    async def main():
        task = asyncio.create_task(srv._lightning_rod_mutate_beacons(spec=_spec(["alpha", "beta"])))
        while not entered.is_set():
            await asyncio.sleep(0.01)
        task.cancel()
        await asyncio.sleep(0.2)
        # Cancelled, but the write is still in flight: the lease is NOT released.
        assert events == [] and leases.held
        proceed.set()
        with pytest.raises(asyncio.CancelledError):
            await task

    asyncio.run(main())
    assert events == ["write_done", "release"]
    assert len(leases.releases) == 1 and isinstance(leases.releases[0], asyncio.CancelledError)
    text = path.read_text(encoding="utf-8")
    # alpha's write completed atomically; beta never started.
    assert text.count("slice_labels=old,nbbw,") == 1

def test_phase_b_resolver_thread_sees_the_callers_operation_cache_snapshot(tmp_path, monkeypatch):
    """The Phase-B worker thread (private loop) must read the SAME pinned
    operation snapshot as the calling task, or every symbol would silently
    resolve against no cache (the resolver swallows cache errors as 'unresolved')."""
    from workflowy_mcp.coherence import operation_cache as oc

    _path, ids, nodes = _one_file_fixture(tmp_path, names=("alpha",))
    _install(monkeypatch, nodes)
    _observe_leases(monkeypatch)
    seen = {}

    async def recording_resolver(**kwargs):
        seen["snapshot"] = oc.current_operation_cache(required=False)
        seen["main_thread"] = threading.current_thread() is threading.main_thread()
        seen["suppress_tags"] = kwargs.get("suppress_tags")
        return {"ok": True, "records": [{"item": "alpha", "status": "unresolved", "candidates": []}],
                "resolution_report": {}}

    monkeypatch.setattr(srv, "_resolve_lightning_rod_symbol_items_from_cache", recording_resolver)
    sentinel = object()

    async def main():
        token = oc._OPERATION_CACHE_SNAPSHOT.set(sentinel)
        try:
            return await srv._lightning_rod_mutate_beacons(spec=_spec(["alpha"]))
        finally:
            oc._OPERATION_CACHE_SNAPSHOT.reset(token)

    res = asyncio.run(main())
    assert seen["snapshot"] is sentinel
    assert seen["main_thread"] is False
    assert seen["suppress_tags"] == []
    assert [r["status"] for r in res["results"]] == ["unresolved"]

def test_client_engine_scopes_candidates_before_counting(tmp_path, monkeypatch):
    """The shared engine drops out-of-scope candidates BEFORE the 0/1/N count:
    the same duplicate symbol is ambiguous without a scope and unique with one."""
    _a, _b, ids, nodes = _scope_fixture(tmp_path)
    client = _install(monkeypatch, nodes)
    ambiguous = asyncio.run(client.read_uuid_from_symbol(None, "shared2"))
    assert ambiguous["match_count"] == 2
    scoped = asyncio.run(
        client.read_uuid_from_symbol(None, "shared2", scope_root_ids=[_REPO_A])
    )
    assert scoped["match_count"] == 1
    assert [c["uuid"] for c in scoped["candidates"]] == [ids["a.shared2"]]
    # A candidate matched only OUTSIDE the scope is reported, not silently
    # counted as a plain zero: match_count 0 + scope_filtered_count 1.
    outside = asyncio.run(
        client.read_uuid_from_symbol(None, "only_b", scope_root_ids=[_REPO_A])
    )
    assert outside["match_count"] == 0
    assert outside["scope_filtered_count"] == 1

def _pin_cache(nodes):
    """Pin the nodes as the task's operation-cache snapshot (plain mapping).

    ``_resolve_beacon_scope_roots`` resolves roots through the family's ONE
    graph-handle resolver, which reads the pinned operation cache; in
    production every dispatch runs inside the cache_operation_boundary that
    pins it. These direct-dispatch tests pin the same nodes by hand.
    """
    from workflowy_mcp.coherence import operation_cache as oc

    class _Pin:
        def __enter__(self):
            self.token = oc._OPERATION_CACHE_SNAPSHOT.set(
                {str(n["id"]): n for n in nodes}
            )
            return self

        def __exit__(self, *exc):
            oc._OPERATION_CACHE_SNAPSHOT.reset(self.token)
            return False

    return _Pin()


def test_beacon_resolve_root_scopes_the_symbol_leg(tmp_path, monkeypatch):
    """beacon.resolve: root_handles (one handle or a list) disambiguates a
    duplicate symbol; an unresolvable root fails loud; the scope is echoed."""
    _a, _b, ids, nodes = _scope_fixture(tmp_path)
    _install(monkeypatch, nodes, suppress=[])
    registry = srv.build_beacon_registry()

    with _pin_cache(nodes):
        ambiguous = asyncio.run(
            dispatch_operation(registry, "resolve", {"source_handle": "shared2"})
        )
        assert ambiguous["match_count"] == 2

        scoped = asyncio.run(
            dispatch_operation(
                registry,
                "resolve",
                {"source_handle": "shared2", "root_handles": _REPO_B},
            )
        )
        assert scoped["match_count"] == 1
        assert [c["uuid"] for c in scoped["candidates"]] == [ids["b.shared2"]]

        listed = asyncio.run(
            dispatch_operation(
                registry,
                "resolve",
                {"source_handle": "shared2", "root_handles": [_REPO_A, _REPO_B]},
            )
        )
        assert listed["match_count"] == 2

        outside = asyncio.run(
            dispatch_operation(
                registry,
                "resolve",
                {"source_handle": "only_b", "root_handles": _REPO_A},
            )
        )
        assert outside["match_count"] == 0
        assert outside["scope_filtered_count"] == 1

        bad = asyncio.run(
            dispatch_operation(
                registry,
                "resolve",
                {"source_handle": "shared2", "root_handles": "nope-nope"},
            )
        )
    assert bad["error"]["code"] == "invalid_parameter_value"
    assert "not present in the /nodes-export cache" in bad["error"]["message"]


def test_beacon_snippet_root_scopes_the_symbol_leg(tmp_path, monkeypatch):
    """beacon.snippet: root_handles resolves the duplicate to the in-scope
    side; an explicit UUID source_handle ignores the scope."""
    a, b, ids, nodes = _scope_fixture(tmp_path)
    _install(monkeypatch, nodes, suppress=[])
    registry = srv.build_beacon_registry()

    with _pin_cache(nodes):
        out = asyncio.run(
            dispatch_operation(
                registry,
                "snippet",
                {"source_handle": "shared2", "root_handles": [_REPO_A]},
            )
        )
        assert isinstance(out, str), out
        assert "return 'a'" in out and "return 'b'" not in out

        # The scope governs symbol resolution only: an explicit UUID is read
        # as given, even when it lies outside every passed root.
        out_uuid = asyncio.run(
            dispatch_operation(
                registry,
                "snippet",
                {"source_handle": ids["b.only_b"], "root_handles": _REPO_A},
            )
        )
        assert isinstance(out_uuid, str), out_uuid
        assert "return 2" in out_uuid

        # A mistyped root is a loud parameter error, never a silent no-scope.
        bad = asyncio.run(
            dispatch_operation(
                registry,
                "snippet",
                {"source_handle": "shared2", "root_handles": "zzzzzzzz"},
            )
        )
        assert bad["error"]["code"] == "invalid_parameter_value"
        assert "not present in the /nodes-export cache" in bad["error"]["message"]

        # An AMBIGUOUS root prefix is a did-you-mean, never a guess.
        amb = asyncio.run(
            dispatch_operation(
                registry,
                "snippet",
                {"source_handle": "shared2", "root_handles": "aaaaaaaa"},
            )
        )
    assert amb["error"]["code"] == "invalid_parameter_value"
    assert amb["error"]["extra"]["candidates"]

def _hex_symbol_fixture(tmp_path):
    """real.py + a #wfy-suppress'd copy.py, both holding a HEX-NAMED symbol.

    ``deadbeefcafe`` is pure hex of >= 8 chars, so beacon.snippet routes it
    through the UUID leg first (the shared gate) and only reaches it as the
    symbol FALLBACK when no node id carries that prefix -- the exact path
    Gap 1 left unfiltered.
    """
    real = tmp_path / "real.py"
    copy = tmp_path / "copy.py"
    real.write_text("def deadbeefcafe():\n    return 'real'\n", encoding="utf-8")
    copy.write_text("def deadbeefcafe():\n    return 'copy'\n", encoding="utf-8")
    ids = {
        "real.deadbeefcafe": "eeeeeeee-0000-0000-0000-000000000001",
        "copy.deadbeefcafe": "eeeeeeee-0000-0000-0000-000000000002",
    }
    nodes = [
        {"id": _ROOT, "name": "Root", "note": "", "parent_id": None},
        {"id": _FILE_A, "name": "real.py", "note": f"Path: {real}", "parent_id": _ROOT},
        {
            "id": ids["real.deadbeefcafe"],
            "name": "deadbeefcafe",
            "note": "AST_QUALNAME: deadbeefcafe\n",
            "parent_id": _FILE_A,
        },
        {"id": _HIDE, "name": "Old copy #<u>wfy-suppress</u>", "note": "", "parent_id": _ROOT},
        {"id": _FILE_B, "name": "copy.py", "note": f"Path: {copy}", "parent_id": _HIDE},
        {
            "id": ids["copy.deadbeefcafe"],
            "name": "deadbeefcafe",
            "note": "AST_QUALNAME: deadbeefcafe\n",
            "parent_id": _FILE_B,
        },
    ]
    return real, copy, ids, nodes


def test_uuid_fallback_symbol_leg_drops_suppressed_duplicate(tmp_path, monkeypatch):
    """Gap 1: a UUID-shaped handle that misses falls back to the symbol leg,
    and that fallback filters suppressed candidates (the describe promises
    it). Pre-fix the suppressed duplicate makes the fallback ambiguous."""
    _real, _copy, _ids, nodes = _hex_symbol_fixture(tmp_path)
    _install(monkeypatch, nodes, suppress=["wfy-suppress"])
    out = asyncio.run(
        dispatch_operation(
            srv.build_beacon_registry(), "snippet", {"source_handle": "deadbeefcafe"}

    assert isinstance(out, str), out
    assert "return 'real'" in out and "return 'copy'" not in out


def test_uuid_fallback_suppressed_only_symbol_is_not_found(tmp_path, monkeypatch):
    """A hex-named symbol that exists ONLY under suppression stays invisible
    to the UUID fallback: a UUID-framed not-found, never a suppressed hit."""
    real = tmp_path / "real.py"

    real.write_text("def plain():\n    return 1\n", encoding="utf-8")
    copy.write_text("def deadbeefcafe():\n    return 'copy'\n", encoding="utf-8")
    nodes = [
        {"id": _ROOT, "name": "Root", "note": "", "parent_id": None},

        {"id": _HIDE, "name": "Old copy #<u>wfy-suppress</u>", "note": "", "parent_id": _ROOT},
        {"id": _FILE_B, "name": "copy.py", "note": f"Path: {copy}", "parent_id": _HIDE},

            "id": "eeeeeeee-0000-0000-0000-000000000002",
            "name": "deadbeefcafe",
