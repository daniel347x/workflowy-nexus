_PARSERS = {short: getattr(nmc, fn) for short, fn in bsb.PARSER_FUNCS.items()}
_PARSERS_BY_LANGUAGE = {
    "python": ("python",), "js_ts": ("js",), "markdown": ("markdown",),
    "sql": ("sql",), "shell": ("sh",), "yaml": ("yaml",), "vue": ("js", "markdown"),
}
_SKIP_DIRS = {".git", ".venv", ".venv-wsl", "node_modules", "__pycache__", ".runs"}


def _tracked_beacon_language_files():
    """Every file of the working tree (the runner's clone holds exactly the
    tracked and untracked-unignored files) whose extension the writers'
    table maps to a beacon language."""
    for dirpath, dirnames, filenames in os.walk(_REPO_ROOT):
        dirnames[:] = sorted(d for d in dirnames if d not in _SKIP_DIRS and not d.startswith("."))
        for fn in sorted(filenames):
            language = _BEACON_LANGUAGE_BY_EXT.get(os.path.splitext(fn)[1].lower())
            if language:
                yield os.path.join(dirpath, fn), language


def test_parsers_with_and_without_a_sink_over_tracked_files_corpus_and_fixtures(golden):
    """§9.1 item 1 (+ Ledger CF-11): one loop over every tracked beacon-language
    file, the route-(b) corpus and the fixtures. Default call and
    diagnostics=None equal the base output wherever the baseline has it; a
    sink never changes the result and adds no key."""
    work: list[tuple[str, list[str], tuple[str, ...]]] = []
    for path, language in _tracked_beacon_language_files():
        with open(path, encoding="utf-8") as f:
            work.append((path, f.read().splitlines(), _PARSERS_BY_LANGUAGE[language]))
    for key, (_fname, text) in bsb.all_sources().items():
        work.append((key, text.splitlines(), tuple(_PARSERS)))
    tracked = [w for w in work if not w[0].startswith(("corpus:", "fixture:"))]
    assert len(tracked) > 300 and any(w[0].endswith(".js") for w in tracked)
    with_records = 0
    for name, lines, shorts in work:
        for short in shorts:
            parser = _PARSERS[short]
            plain = parser(lines)
            if name in golden["parsers"]:
                assert plain == golden["parsers"][name][short], (name, short)
                assert parser(lines, diagnostics=None) == golden["parsers"][name][short], (name, short)
            sink: list = []
            with_sink = parser(lines, diagnostics=sink)
            assert with_sink == plain, (name, short)
            for a, b in zip(with_sink, plain):
                assert set(a) == set(b), (name, short)
            with_records += bool(sink)
    assert with_records > 100

def _blk(line, end, outcome, kind_raw, bid, keys, unknown=()):
    return {
        "record": "block", "line": line, "end_line": end, "outcome": outcome,
        "kind_raw": kind_raw, "id": bid, "field_keys": sorted(keys),
        "unknown_keys": list(unknown),
    }


def _close(line, bid):
    return {"record": "close", "line": line, "id": bid}


def _match(line, opener):
    return {"record": "close_match", "line": line, "opener_line": opener}
