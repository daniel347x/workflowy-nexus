"""Regression lock: ETCH forest preflight rejects unknown/malformed node fields
with ZERO writes (the `content`-vs-`note` silent-drop incident, 2026).

Background: ``workflowy_etch`` read known keys off each node dict with
``.get(...)`` and ignored everything else, so ``{"name": "x", "content": "..."}``
created a name-only node and reported success -- data silently swallowed.
The recursive preflight inside ``workflowy_etch`` is the ONE ETCH-forest
grammar authority (every ingress -- node.etch sync/async, point.capture's
``nodes`` form, nodes_file -- reaches it), so unknown-key rejection lives
there. These tests lock: index-precise messages, the teaching hint for
``content``, all five legal keys accepted, and NO create_node call on any
rejection, including a malformed node deep in the forest AFTER valid siblings.

Also locks the sibling closure in node.edit ``note_edits``: an unknown key
(e.g. ``replacement``) is rejected instead of being ignored -- which
previously turned into a silent DELETION because omitted ``replace`` means
delete by contract.
"""

from __future__ import annotations

import asyncio
import json

import pytest

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", "src"))

_PARENT = "11111111-2222-3333-4444-555555555555"


def _run(coro):
    return asyncio.run(coro)


def _make_client():
    """Real WorkFlowyClientEtch via __new__ with create_node counted."""
    c = WorkFlowyClientEtch.__new__(WorkFlowyClientEtch)
    calls: list[dict] = []

    async def _fake_create_node(request, _internal_call=False, **kw):
        calls.append(request.model_dump())
        return WorkFlowyNode(
            id=f"created-{len(calls)}", name=request.name, note=request.note
        )

    c.create_node = _fake_create_node  # type: ignore[assignment]
    c._log_to_file = lambda *a, **k: None  # type: ignore[assignment]
    c._mark_nodes_export_dirty = lambda ids: None  # type: ignore[assignment]
    return c, calls

def _literal_values(annotation) -> frozenset:
    """Unwrap `Literal[...] | None` (or bare Literal) into its value set."""
    import typing

    args = typing.get_args(annotation)
    for arg in args:
        if typing.get_origin(arg) is typing.Literal:
            return frozenset(typing.get_args(arg))
    if typing.get_origin(annotation) is typing.Literal:
        return frozenset(args)
    raise AssertionError(f"no Literal found in {annotation!r}")


def test_allowed_keys_match_create_tree_consumption():
    assert ETCH_NODE_ALLOWED_KEYS == frozenset(
        {"name", "note", "children", "layout_mode", "position"}
    )


def test_enum_constants_are_derived_from_the_creation_model_not_hardcoded_twice():
    """REAL drift detector: the preflight enums must equal the Literal types on
    NodeCreateRequest (the authority create_tree hands each node to). If someone
    adds 'h4' to the model, this fails until ETCH_LAYOUT_MODES follows."""
    fields = NodeCreateRequest.model_fields
    assert ETCH_LAYOUT_MODES == _literal_values(fields["layoutMode"].annotation)
    assert ETCH_POSITIONS == _literal_values(fields["position"].annotation)


# ---------------------------------------------------------------------------
# The incident: `content` instead of `note`
# ---------------------------------------------------------------------------

def test_content_key_is_rejected_with_teaching_hint_and_zero_writes():
    c, calls = _make_client()
    result = _run(c.workflowy_etch(
        _PARENT, [{"name": "RENDER TEST", "content": "Test note body."}]
    ))
    assert result["success"] is False
    assert result["nodes_created"] == 0
    assert result["root_node_ids"] == []
    assert calls == []  # ZERO create_node calls
    msg = result["errors"][0]
    assert "nodes[0] has unknown field(s): ['content']" in msg
    assert "'note'" in msg
    assert "point.capture" in msg
    assert "NO nodes were created" in msg


def test_unknown_key_deep_in_forest_after_valid_siblings_means_zero_writes():
    """A bad child at depth 2, AFTER two perfectly valid roots: nothing is created."""
    c, calls = _make_client()
    forest = [
        {"name": "Root A", "note": "fine"},
        {"name": "Root B", "children": [
            {"name": "B.1"},
            {"name": "B.2", "children": [
                {"name": "B.2.a", "body": "wrong key"},
            ]},
        ]},
    ]
    result = _run(c.workflowy_etch(_PARENT, forest))
    assert result["success"] is False
    assert calls == []
    assert "nodes[1].children[1].children[0] has unknown field(s): ['body']" in result["errors"][0]

# note_edits sibling closure (node.edit): unknown key rejected, not ignored
# ---------------------------------------------------------------------------

def _patch_edit_boundary(monkeypatch):
    calls: list = []

    class _Client:
        async def get_node(self, node_id):
            calls.append(("get", node_id))
            return WorkFlowyNode(id=node_id, name="T", note="keep important text here")

        async def update_node(self, *a, **k):
            calls.append(("update", a, k))
            raise AssertionError("update_node must not be reached")

    monkeypatch.setattr(server, "get_client", lambda: _Client())
    monkeypatch.setattr(server, "_rate_limiter", None)

    async def _resolve(node_id, *, label, required=True):
        return node_id

    monkeypatch.setattr(server, "_resolve_mutator_node_id", _resolve)
    monkeypatch.setattr(server, "_resolve_generic_node_handle", _resolve)
    return calls


def test_note_edits_replacement_key_is_rejected_not_treated_as_deletion(monkeypatch):
    """`replacement` used to be ignored => omitted `replace` => DELETE the match."""
    calls = _patch_edit_boundary(monkeypatch)
    with pytest.raises(ValueError, match=r"note_edits\[0\] has unknown field\(s\): \['replacement'\]") as ei:
        _run(server.edit_node(
            "uuid-1",
            note_edits=[{"search": "important text", "replacement": "new text"}],
        ))
    assert "'replace'" in str(ei.value)
    assert "NO edit was applied" in str(ei.value)
    assert calls == []  # raised before any client interaction


def test_note_edits_unknown_key_in_second_item_rejects_whole_batch(monkeypatch):
    calls = _patch_edit_boundary(monkeypatch)
    with pytest.raises(ValueError, match=r"note_edits\[1\] has unknown field\(s\): \['find'\]") as ei:
        _run(server.edit_node(
            "uuid-1",
            note_edits=[
                {"search": "a", "replace": "b"},
                {"find": "c", "replace": "d"},
            ],
        ))
    assert "'search'" in str(ei.value)
    assert calls == []


def test_note_edits_legal_grammar_still_accepted_including_omitted_replace(monkeypatch):
    """Omitted replace == deletion remains the contract; only unknown keys are new errors."""
    seen: dict = {}

    class _Client:
        async def get_node(self, node_id):
            return WorkFlowyNode(id=node_id, name="T", note="alpha beta gamma")

        async def update_node(self, node_id, request, *a, **k):
            seen["note"] = request.note
            return WorkFlowyNode(id=node_id, name="T", note=request.note)

    monkeypatch.setattr(server, "get_client", lambda: _Client())
    monkeypatch.setattr(server, "_rate_limiter", None)

    async def _resolve(node_id, *, label, required=True):
        return node_id

    monkeypatch.setattr(server, "_resolve_mutator_node_id", _resolve)
    monkeypatch.setattr(server, "_resolve_generic_node_handle", _resolve)

    _run(server.edit_node(
        "uuid-1",
        note_edits=[{"search": "alpha ", "replace": "ALPHA "}, {"search": " gamma"}],
    ))
    assert seen["note"] == "ALPHA beta"
