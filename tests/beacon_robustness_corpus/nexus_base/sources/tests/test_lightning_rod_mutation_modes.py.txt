def _spec_for(
    mode, *, operations=None, symbols=None, tags=None, comment=None, roots=None, favor="uuid",
    atomic=False
):
    spec, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode=mode,
        symbols=symbols,
        tags=tags,
        operations=operations,
        comment=comment,
        atomic=atomic,
        root_uuids=(roots if roots is not None else [_ROOT]),
        root_uuid=None,
        favor=favor,
    )
    assert err is None, err
    assert spec is not None
    return spec

def test_mutation_normalization_comment_modes():
    # 2R4E1P: comment_mode rides top-level (shorthand) and per-op (granular).
    spec, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=["foo"],
        tags=["a"],
        operations=None,
        comment="c",
        comment_mode="replace",
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err is None and spec is not None
    assert spec.operations[0].comment.mode == "replace"

    spec, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=None,
        tags=None,
        operations=[{"symbol": "foo", "tags": ["a"], "comment": "c", "comment_mode": "append"}],
        comment=None,
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err is None and spec is not None
    assert spec.operations[0].comment.mode == "append"

    # default (absent) stays fill_if_empty
    spec, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=["foo"],
        tags=["a"],
        operations=None,
        comment="c",
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err is None and spec.operations[0].comment.mode == "fill_if_empty"

    # an explicit comment_mode without a comment is a loud error
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=["foo"],
        tags=["a"],
        operations=None,
        comment=None,
        comment_mode="replace",
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err and "comment_mode requires comment" in err

    # unknown comment_mode
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=["foo"],
        tags=["a"],
        operations=None,
        comment="c",
        comment_mode="upsert",
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err and "upsert" in err

    # remove mode refuses comment_mode (top-level and per-op)
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="remove_beacon_tags",
        symbols=["foo"],
        tags=["a"],
        operations=None,
        comment=None,
        comment_mode="replace",
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err and "comment_mode" in err and "remove_beacon_tags" in err
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="remove_beacon_tags",
        symbols=None,
        tags=None,
        operations=[{"symbol": "foo", "tags": ["a"], "comment": "c", "comment_mode": "fill"}],
        comment=None,
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err and "comment_mode" in err

    # per-op comment_mode without a per-op comment is a loud error
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=None,
        tags=None,
        operations=[{"symbol": "foo", "tags": ["a"], "comment_mode": "append"}],
        comment=None,
        root_uuids=["root"],
        root_uuid=None,
        favor="uuid",
    )
    assert err and "comment_mode requires comment" in err

    # top-level comment_mode with granular operations follows comment's
    # mixing rule: rejected
    _, err = LightningRodMutationSpec.normalize_from_tool_args(
        mode="ensure_beacons",
        symbols=None,
        tags=None,

def _resolve(mode, value, note, disk, exists=True):
    if mode == "preserve":
        intent = bmc.BeaconCommentIntent.preserve()
    elif mode == "replace":
        intent = bmc.BeaconCommentIntent.replace(value)
    elif mode == "append":
        intent = bmc.BeaconCommentIntent.append(value)
    else:
        intent = bmc.BeaconCommentIntent.fill_if_empty(value)
    return bmc.resolve_beacon_comment(
        intent=intent,
        note_comment=note,
        disk_comment=disk,
        beacon_exists_on_disk=exists,
    )


def test_resolve_preserve_keeps_disk_then_note_and_never_conflicts():
    r = _resolve("preserve", None, note="n", disk="d")
    assert r.outcome == "preserve" and r.resolved == "d"
    r = _resolve("preserve", None, note="n", disk="")
    assert r.outcome == "preserve" and r.resolved == "n"
    r = _resolve("preserve", None, note="", disk="")
    assert r.outcome == "preserve" and r.resolved is None
    # divergent note/disk preserved WITHOUT error, disk value wins
    r = _resolve("preserve", None, note="n1", disk="d1")
    assert r.outcome == "preserve" and r.resolved == "d1"


def test_resolve_fill_idempotent_and_conflicts():
    # fill on missing/empty disk
    r = _resolve("fill", "req", note="", disk="")
    assert r.outcome == "fill" and r.resolved == "req"
    # note == requested + disk EMPTY -> FILL (materialize to disk; rerun noops)
    r = _resolve("fill", "req", note="req", disk="")
    assert r.outcome == "fill" and r.resolved == "req"
    # idempotent on equal (disk == requested, note either state)
    r = _resolve("fill", "req", note="", disk="req")
    assert r.outcome == "idempotent" and r.resolved == "req"
    r = _resolve("fill", "req", note="req", disk="req")
    assert r.outcome == "idempotent"
    # 2R4E1P: a fill-mode conflict is KEPT, never an error -- labels still
    # ensured, the existing comment wins, warning carries BOTH texts.
    r = _resolve("fill", "req", note="", disk="other")
    assert r.outcome == "kept_existing_conflict" and r.resolved == "other"
    assert r.existing_comment == "other"
    assert "other" in (r.warning or "") and "req" in (r.warning or "")
    assert "replace" in (r.warning or "") and "append" in (r.warning or "")
    # both sources agree on a DIFFERENT comment -> kept (existing wins)
    r = _resolve("fill", "req", note="other", disk="other")
    assert r.outcome == "kept_existing_conflict" and r.resolved == "other"
    # disk empty but the note disagrees -> kept (the note's claim wins)
    r = _resolve("fill", "req", note="other", disk="")
    assert r.outcome == "kept_existing_conflict" and r.resolved == "other"
    assert r.existing_comment == "other"
    # note and disk disagree with each other -> kept, DISK WINS
    r = _resolve("fill", "req", note="n1", disk="d1")
    assert r.outcome == "kept_existing_conflict" and r.resolved == "d1"
    assert "d1" in (r.warning or "") and "req" in (r.warning or "")


def test_resolve_replace_and_append_modes():
    # replace: explicit opt-in overwrite
    r = _resolve("replace", "req", note="whatever", disk="other")
    assert r.outcome == "replace" and r.resolved == "req"
    assert r.warning is None  # an explicit overwrite is not a conflict
    # replace on equal text stays idempotent (re-runs are noops)
    r = _resolve("replace", "req", note="", disk="req")
    assert r.outcome == "idempotent" and r.resolved == "req"
    # replace with nothing on disk = a fill (create carries the comment)
    r = _resolve("replace", "req", note="", disk="", exists=False)
    assert r.outcome == "replace" and r.resolved == "req"
    # append: empty disk fills
    r = _resolve("append", "req", note="", disk="")
    assert r.outcome == "fill" and r.resolved == "req"
    # append extends the existing comment with one separating space
    r = _resolve("append", "req", note="n", disk="other")
    assert r.outcome == "append" and r.resolved == "other req"
    # append re-runs are idempotent (exact text, or already-appended tail)
    r = _resolve("append", "req", note="", disk="req")
    assert r.outcome == "idempotent" and r.resolved == "req"
    r = _resolve("append", "req", note="", disk="other req")
    assert r.outcome == "idempotent" and r.resolved == "other req"


def test_resolve_compares_in_parser_normalized_form():
    # NORMALIZATION LOCK (Finding A): a trailing-comma request vs a stripped
    # disk parse converges (idempotent), never a false kept-conflict.
    r = _resolve("fill", "req,", note="", disk="req")
    assert r.outcome == "idempotent"
    # a comma-carrying cached note vs a stripped disk parse does NOT produce
    # kept_existing_conflict.
    r = _resolve("fill", "req", note="req,", disk="req")
    assert r.outcome == "idempotent"
    r = _resolve("fill", "req", note="req]", disk="req")
    assert r.outcome == "idempotent"

class _PipelineClient:
    """Controlled client for the commit-6 pipeline.

    Delegates preflight/plan to the REAL ``_compute_beacon_write_plan`` and
    commit to the REAL language helpers (so the same-file CAS behavior is
    genuinely exercised). Every Workflowy-mutating method RAISES -- the
    mutation path must perform ZERO Workflowy API calls (invariant 9).
    """

    def __init__(
        self, *, nodes, candidates, jit_fail=(), commit_fail=(), force_preflight_actionable=False
    ):
        self.nodes = list(nodes)
        self.cache_by_id = {str(n["id"]): n for n in self.nodes}
        self.candidates = dict(candidates)
        self.jit_fail = set(jit_fail)
        self.commit_fail = set(commit_fail)
        self.force_preflight_actionable = force_preflight_actionable
        self.preflight_calls = []
        self.plan_calls = []
        self.commit_calls = []

    def _get_nodes_export_cache_nodes(self):
        return list(self.nodes)

    async def read_uuid_from_symbol(self, file_path, symbol, symbol_kind):
        uid = self.candidates.get(symbol)
        return {"candidates": ([{"uuid": uid, "name": symbol}] if uid else [])}

    # -- ZERO-Workflowy-API lock (invariant 9) --
    async def workflowy_add_tag(self, *a, **k):
        raise AssertionError("mutation path called workflowy_add_tag")

    async def workflowy_remove_tag(self, *a, **k):
        raise AssertionError("mutation path called workflowy_remove_tag")

    async def update_beacon_from_node(self, *a, **k):
        raise AssertionError("mutation path called update_beacon_from_node")

    async def refresh_file_node_beacons(self, *a, **k):
        raise AssertionError("mutation path called refresh_file_node_beacons")

    async def get_node(self, *a, **k):
        raise AssertionError("mutation path called get_node")

    async def preflight_beacon_write_from_node(self, node_id, operation, mode):
        self.preflight_calls.append(node_id)
        if self.force_preflight_actionable:
            return {"outcome": "actionable", "target_key": None}
        plan = acn._compute_beacon_write_plan(
            self.cache_by_id, node_id, operation, mode, include_fingerprint=False
        )
        if not plan.success:
            return {
                "outcome": "failure",
                "error": plan.error,
                "error_kind": plan.error_kind,
                "target_key": plan.target_key,
            }
        if plan.planned_operation == "noop":
            return {
                "outcome": "noop",
                "target_key": plan.target_key,
                # mirror the real method's noop shape (2R4E1P: the kept
                # fill-mode conflict warning rides the noop row too)
                "noop_reason": getattr(plan, "noop_reason", None),
                "comment_outcome": getattr(plan, "comment_outcome", None),
                "comment_warning": getattr(plan, "comment_warning", None),
            }
        return {
            "outcome": "actionable",
            "planned_operation": plan.planned_operation,
            "target_key": plan.target_key,
        }

    async def plan_beacon_write_from_node(self, node_id, operation, mode):
        self.plan_calls.append(node_id)
        if node_id in self.jit_fail:
            return acn.BeaconWritePlan(
                success=False,
                planned_operation=None,
                node_id=node_id,
                target_key=None,
                cached_name="",
                intended_name="",
                write_intent=None,
                occurrence=None,
                file_node_id=None,
                source_path=None,
                language=None,
                beacon_id=None,
                note_comment=None,
                disk_comment=None,
                comment_outcome=None,
                resolved_comment=None,
                intended_slice_labels=(),
                source_sha256=None,
                error="injected JIT failure",
                error_kind="beacon_identity_conflict",
            )
        return acn._compute_beacon_write_plan(
            self.cache_by_id, node_id, operation, mode, include_fingerprint=True
        )

    async def commit_beacon_write_plan(self, *, plan, note):
        self.commit_calls.append(plan.node_id)
        if plan.node_id in self.commit_fail:
            return {
                "success": False,
                "error": "injected commit failure",
                "error_kind": "failed_to_write_file",
                "disk_changed": False,
            }
        ext = os.path.splitext(str(plan.source_path))[1].lower()
        lang = acn._BEACON_LANGUAGE_BY_EXT.get(ext)
        carto = acn._import_cartographer_module()
        helper = getattr(carto, acn._BEACON_LANGUAGE_HELPER_NAME[lang])
        if lang == "markdown":
            res = helper(
                str(plan.source_path),
                plan.intended_name,
                note,
                occurrence=plan.occurrence,
                write_intent=plan.write_intent,
                write_plan=plan,
            )
        else:
            res = helper(
                str(plan.source_path),
                plan.intended_name,
                note,
                write_intent=plan.write_intent,
                write_plan=plan,
            )
        res.setdefault("source_path", str(plan.source_path))
        res.setdefault("file_node_id", plan.file_node_id)
        res.setdefault("language", lang)
        res.setdefault(
            "disk_changed",
            bool(
                res.get("success")
                and res.get("operation") in {"created_beacon", "updated_beacon", "deleted_beacon"}
            ),
        )
        return res


def _pipeline_nodes(source_path, entries):
    """Build a synthetic cache: ROOT -> FILE(source_path) -> child nodes."""
    nodes = [
        {"id": _ROOT, "name": "Root", "note": "", "parent_id": None, "priority": 0},
        {
            "id": _FILE,
            "name": os.path.basename(str(source_path)),
            "note": f"Path: {source_path}",
            "parent_id": _ROOT,
            "priority": 0,
        },
    ]
    for nid, name, note in entries:
        nodes.append({"id": nid, "name": name, "note": note, "parent_id": _FILE, "priority": 100})
    return nodes

def _observe_batch_lease_boundary(monkeypatch):
    """Observe caller sequencing only; the governor itself is not reimplemented."""
    state = SimpleNamespace(acquisitions=[], releases=[], held=False, refusal=None)
    tokens = [{"lease_id": "test-only-token"}]

    async def acquire(targets, **kwargs):
        assert not state.held
        state.acquisitions.append(list(targets))
        if state.refusal is not None:
            raise state.refusal
        state.held = True
        return tokens

    async def release(leases, *, primary_error=None):
        assert state.held and leases is tokens
        state.releases.append(primary_error)
        state.held = False

    monkeypatch.setattr(srv, "acquire_q3_leases_async", acquire)
    monkeypatch.setattr(srv, "_release_q3_leases_preserving_primary", release)
    return state
