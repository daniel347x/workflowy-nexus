from __future__ import annotations

import ast
import asyncio
import subprocess
import sys
import time
from pathlib import Path

import pytest

from workflowy_mcp import carto_job_governance as gov

ACCOUNT = "a" * 32
REPO = Path(__file__).resolve().parents[1]


def _cache() -> dict[str, dict]:
    return {
        "vault": {"id": "vault"},
        "repo": {"id": "repo", "parent_id": "vault"},
        "file": {"id": "file", "parent_id": "repo"},
    }


def _footprint(tmp_path: Path) -> dict:
    target = tmp_path / "repo" / "a.py"
    target.parent.mkdir(exist_ok=True)
    target.write_text("x = 1\n", encoding="utf-8")
    return gov.compute_footprint(target, "file", _cache(), kind="file")


def _lock_holder(lock_path: Path, tmp_path: Path) -> tuple[subprocess.Popen, Path, Path]:
    ready = tmp_path / "lock-ready"
    release = tmp_path / "lock-release"
    code = r"""
import sys, time
from pathlib import Path
from workflowy_mcp.carto_job_governance import acquire_os_lock
lock_path, ready, release = map(Path, sys.argv[1:])
held = acquire_os_lock(lock_path, timeout_seconds=5.0, poll_seconds=0.01)
ready.write_text("ready", encoding="utf-8")
try:
    deadline = time.monotonic() + 10.0
    while not release.exists() and time.monotonic() < deadline:
        time.sleep(0.01)
finally:
    held.release()
"""
    process = subprocess.Popen(
        [sys.executable, "-c", code, str(lock_path), str(ready), str(release)],
        cwd=REPO,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )
    deadline = time.monotonic() + 5.0
    while not ready.exists() and time.monotonic() < deadline:
        if process.poll() is not None:
            out, err = process.communicate()
            raise AssertionError(f"lock holder exited early: {out}\n{err}")
        time.sleep(0.01)
    assert ready.exists()
    return process, ready, release


def _finish_holder(process: subprocess.Popen, release: Path) -> None:
    release.write_text("release", encoding="utf-8")
    out, err = process.communicate(timeout=5)
    assert process.returncode == 0, f"{out}\n{err}"


def test_async_q3_lock_wait_keeps_event_loop_responsive(tmp_path):
    jobs = tmp_path / "jobs"
    paths = gov.governance_paths(jobs)
    process, _ready, release = _lock_holder(paths.admission_mutex, tmp_path)

    async def scenario() -> int:
        ticks = 0
        stop = asyncio.Event()

        async def ticker() -> None:
            nonlocal ticks
            while not stop.is_set():
                ticks += 1
                await asyncio.sleep(0.01)

        ticker_task = asyncio.create_task(ticker())
        acquire_task = asyncio.create_task(
            gov.acquire_q3_leases_async(
                [_footprint(tmp_path)],
                account_scope_id=ACCOUNT,
                jobs_root=jobs,
            )
        )
        await asyncio.sleep(0.12)
        assert ticks >= 5
        release.write_text("release", encoding="utf-8")
        leases = await acquire_task
        await gov.release_q3_leases_async(leases, jobs_root=jobs)
        stop.set()
        await ticker_task
        return ticks

    try:
        assert asyncio.run(scenario()) >= 5
    finally:
        if process.poll() is None:
            _finish_holder(process, release)
        else:
            out, err = process.communicate()
            assert process.returncode == 0, f"{out}\n{err}"


def test_async_lock_wait_cancellation_stops_before_transaction(tmp_path):
    jobs = tmp_path / "jobs"
    paths = gov.governance_paths(jobs)
    process, _ready, release = _lock_holder(paths.admission_mutex, tmp_path)

    async def scenario() -> None:
        task = asyncio.create_task(
            gov.acquire_q3_leases_async(
                [_footprint(tmp_path)],
                account_scope_id=ACCOUNT,
                jobs_root=jobs,
            )
        )
        await asyncio.sleep(0.08)
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task
        assert not list(paths.leases_dir.glob("*.json"))
        release.write_text("release", encoding="utf-8")
        await asyncio.sleep(0.08)
        assert not list(paths.leases_dir.glob("*.json"))

    try:
        asyncio.run(scenario())
    finally:
        if process.poll() is None:
            _finish_holder(process, release)
        else:
            out, err = process.communicate()
            assert process.returncode == 0, f"{out}\n{err}"


def _context_calls_async_os_lock(node: ast.AsyncWith) -> bool:
    for item in node.items:
        call = item.context_expr
        if not isinstance(call, ast.Call):
            continue
        if isinstance(call.func, ast.Name) and call.func.id == "async_os_lock":
            return True
        if isinstance(call.func, ast.Attribute) and call.func.attr == "async_os_lock":
            return True
    return False


def test_no_await_inside_admission_mutex_transactions():
    offenders: list[tuple[str, str, int]] = []
    for path in (REPO / "src/workflowy_mcp").rglob("*.py"):
        tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
        for function in [node for node in ast.walk(tree) if isinstance(node, ast.AsyncFunctionDef)]:
            for node in ast.walk(function):
                if not isinstance(node, ast.AsyncWith) or not _context_calls_async_os_lock(node):
                    continue
                if any(
                    isinstance(child, ast.Await) for stmt in node.body for child in ast.walk(stmt)
                ):
                    offenders.append((path.name, function.name, node.lineno))
    assert offenders == []


def test_async_functions_do_not_call_sync_admission_wrappers_directly():
    forbidden = {
        "acquire_q3_leases",
        "release_q3_leases",
        "finalize_claim",
        "finalize_cancelled_job_claim",
        "cleanup_carto_auxiliary_artifacts",
        "sweep_region_governance",
        "_delete_carto_job_json_if_unclaimed",
    }
    offenders: list[tuple[str, str, str]] = []
    for path in (REPO / "src/workflowy_mcp").rglob("*.py"):
        # Detached worker owns a dedicated process/event loop and intentionally
        # retains synchronous finalization; the MCP server async paths may not.
        if path.name == "weave_worker.py":
            continue
        tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
        for function in [node for node in ast.walk(tree) if isinstance(node, ast.AsyncFunctionDef)]:
            for call in [node for node in ast.walk(function) if isinstance(node, ast.Call)]:
                name = None
                if isinstance(call.func, ast.Name):
                    name = call.func.id
                elif isinstance(call.func, ast.Attribute):
                    name = call.func.attr
                if name in forbidden:
                    offenders.append((path.name, function.name, name))
    assert offenders == []
