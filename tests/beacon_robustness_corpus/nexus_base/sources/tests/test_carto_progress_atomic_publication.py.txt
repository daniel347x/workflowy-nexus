"""Invoke BOTH actual nested closures; no synthetic progress implementation."""
from __future__ import annotations
import ast
import datetime
import json
import threading
from pathlib import Path

import pytest
from workflowy_mcp import config
from workflowy_mcp import server as srv
import workflowy_mcp.client.api_client_nexus as nexus
from test_carto_final_cache_publication import final_env, write_job


def actual_closure(path, name):
    tree = ast.parse(Path(nexus.__file__).read_text())
    definition = next(n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name == name)
    prefix = ('_carto_ops = 0\n_carto_pending = {"nodes_created": 0, "nodes_updated": 0, "nodes_moved": 0, "nodes_deleted": 0}\n_carto_retry_counts = {}\n'
              if name == "_carto_flush_progress" else
              '_carto_pending_nodes_created = 0\n_carto_bump_counter = 0\n_carto_retry_created = None\n')
    setup = ast.parse(prefix).body
    tail = ast.parse('return closure, locals()').body
    tail[0].value.elts[0] = ast.Name(id=name, ctx=ast.Load())
    factory = ast.FunctionDef(name="factory", args=ast.arguments(posonlyargs=[], args=[ast.arg(arg="carto_job_file")],
                              kwonlyargs=[], kw_defaults=[], defaults=[]),
                              body=setup + [definition] + tail, decorator_list=[])
    module = ast.fix_missing_locations(ast.Module(body=[factory], type_ignores=[]))
    namespace = {"__package__": "workflowy_mcp.client", "json_module": json, "datetime": datetime}
    exec(compile(module, str(nexus.__file__), "exec"), namespace)
    return namespace["factory"](str(path))


def call_progress(closure, state, name, *, increment=1):
    if name == "_carto_flush_progress":
        state["_carto_pending"]["nodes_created"] += increment
        closure(force=True, phase="applying")
    else:
        closure(nodes_created=increment, force=True, phase="creating")


@pytest.mark.parametrize("name", ["_carto_flush_progress", "_carto_bump"])
def test_actual_writer_keeps_complete_old_record_until_replace(final_env, monkeypatch, name):
    jobs, _ = final_env
    path, payload = write_job(jobs, progress={"nodes_created": 0}, warnings=[{"message": "old"}])
    closure, state = actual_closure(path, name)
    started, release = threading.Event(), threading.Event()
    original = config.os.replace

    def paused(src, dst):
        if Path(dst) == path:
            started.set()
            assert release.wait(5)
        return original(src, dst)
    monkeypatch.setattr(config.os, "replace", paused)
    thread = threading.Thread(target=call_progress, args=(closure, state, name))
    thread.start()
    try:
        assert started.wait(5), "actual closure did not use atomic replacement"
        for _ in range(5):
            inventory = srv._scan_carto_job_files(str(jobs), include_inventory=True)
            assert not inventory["uncertain"]
            assert len(inventory["rows"]) == 1
            assert json.loads(path.read_text())["progress"]["nodes_created"] == 0
    finally:
        release.set()
        thread.join(5)
    updated = json.loads(path.read_text())
    assert updated["progress"]["nodes_created"] == 1
    assert updated["warnings"] == payload["warnings"]


@pytest.mark.parametrize("name", ["_carto_flush_progress", "_carto_bump"])
@pytest.mark.parametrize("failure", ["false", "exception", "false_after_replace"])
def test_failed_atomic_flush_retries_without_losing_or_double_counting(final_env, monkeypatch, name, failure):
    jobs, _ = final_env
    path, _ = write_job(jobs, progress={"nodes_created": 0})
    closure, state = actual_closure(path, name)
    original = config.merge_write_carto_job_status
    def fail(path, payload):
        if failure == "exception":
            raise OSError("sharing violation")
        if failure == "false_after_replace":
            assert original(path, payload)
        return False
    monkeypatch.setattr(config, "merge_write_carto_job_status", fail)
    call_progress(closure, state, name)
    monkeypatch.setattr(config, "merge_write_carto_job_status", original)
    call_progress(closure, state, name, increment=0)
    assert json.loads(path.read_text())["progress"]["nodes_created"] == 1
    call_progress(closure, state, name)
    assert json.loads(path.read_text())["progress"]["nodes_created"] == 2


@pytest.mark.parametrize("name", ["_carto_flush_progress", "_carto_bump"])
@pytest.mark.parametrize("status", ["cancelling", "completed", "completed_with_warnings", "failed", "cancelled"])
def test_stale_actual_progress_preserves_status_and_warnings(final_env, monkeypatch, name, status):
    jobs, _ = final_env
    path, payload = write_job(jobs, progress={"nodes_created": 0}, warnings=[{"message": "phase1"}])
    closure, state = actual_closure(path, name)
    original = config.merge_write_carto_job_status
    def late(path, stale):
        # The real status authority receives a stale payload after another
        # phase finalized/cancelled; its precedence/warning table owns this.
        gov_payload = {**payload, "status": status, "warnings": [{"message": "phase2"}]}
        assert original(path, gov_payload)
        return original(path, stale)
    monkeypatch.setattr(config, "merge_write_carto_job_status", late)
    call_progress(closure, state, name)
    result = json.loads(path.read_text())
    assert result["status"] == status
    assert result["warnings"] == [{"message": "phase1"}, {"message": "phase2"}]
