def _cache() -> dict[str, dict]:
    return {
        "vault": {"id": "vault"},
        "repo": {"id": "repo", "parent_id": "vault"},
        "file": {"id": "file", "parent_id": "repo"},
    }


def _footprint(tmp_path: Path) -> dict:
    target = tmp_path / "repo" / "a.py"
    target.parent.mkdir(exist_ok=True)
    target.write_text("x = 1\n", encoding="utf-8")
    return gov.compute_footprint(target, "file", _cache(), kind="file")


def _lease(tmp_path: Path, jobs: Path) -> dict:
    return gov.acquire_q3_leases([_footprint(tmp_path)], account_scope_id=ACCOUNT, jobs_root=jobs)[
        0
    ]


def _claim(tmp_path: Path, jobs: Path, *, job_id: str = "gov04-recovery") -> dict:
    process = psutil.Process(os.getpid())
    claim = gov.build_claim(
        job_id=job_id,
        job_type="CARTO_BULK_APPLY",
        dedupe_key=hashlib.sha256(job_id.encode()).hexdigest(),
        account_scope_id=ACCOUNT,
        footprint=_footprint(tmp_path),
        launcher="system",
        parent_job_id=None,
        launcher_pid=os.getpid(),
        pid=os.getpid(),
        pid_create_time=process.create_time(),
        handoff_state="running",
        spawn_started_at=None,
        execution_owner="launcher",
    )
    gov.write_claim(claim, jobs)
    return claim


def _lock_holder(lock_path: Path, tmp_path: Path) -> tuple[subprocess.Popen, Path]:
    ready = tmp_path / "lock-ready"
    release = tmp_path / "lock-release"
    code = r"""
import sys, time
from pathlib import Path
from workflowy_mcp.carto_job_governance import acquire_os_lock
lock_path, ready, release = map(Path, sys.argv[1:])
held = acquire_os_lock(lock_path, timeout_seconds=5.0, poll_seconds=0.01)
ready.write_text("ready", encoding="utf-8")
try:
    deadline = time.monotonic() + 10.0
    while not release.exists() and time.monotonic() < deadline:
        time.sleep(0.01)
finally:
    held.release()
"""
    process = subprocess.Popen(
        [sys.executable, "-c", code, str(lock_path), str(ready), str(release)],
        cwd=REPO,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )
    deadline = time.monotonic() + 5.0
    while not ready.exists() and time.monotonic() < deadline:
        if process.poll() is not None:
            out, err = process.communicate()
            raise AssertionError(f"lock holder exited early: {out}\n{err}")
        time.sleep(0.01)
    assert ready.exists()
    return process, release


def _finish_holder(process: subprocess.Popen, release: Path) -> None:
    release.write_text("release", encoding="utf-8")
    out, err = process.communicate(timeout=5)
    assert process.returncode == 0, f"{out}\n{err}"


async def _cancel_supervisors() -> None:
    tasks = list(gov._CLEANUP_TASKS.values())
    for task in tasks:
        task.cancel()
    if tasks:
        await asyncio.gather(*tasks, return_exceptions=True)
    await asyncio.sleep(0)


async def _wait_until(predicate, *, timeout: float = 2.0) -> None:
    deadline = asyncio.get_running_loop().time() + timeout
    while not predicate():
        if asyncio.get_running_loop().time() >= deadline:
            raise AssertionError("condition did not become true before timeout")
        await asyncio.sleep(0.005)

def test_persistent_supervisor_retries_beyond_first_attempt_without_duplicates(tmp_path):
    jobs = tmp_path / "jobs"
    lease = _lease(tmp_path, jobs)
    lease_path = gov.q3_lease_path_for(lease, jobs)
    blocked = True
    attempts = 0
    attempt_event: asyncio.Event | None = None
    released_event: asyncio.Event | None = None

    def controlled_unlink(path: Path) -> None:
        nonlocal attempts
        attempts += 1
        assert attempt_event is not None and released_event is not None
        attempt_event.set()
        if blocked:
            raise PermissionError("scanner still holds lease")
        path.unlink()
        released_event.set()

    async def scenario() -> None:
        nonlocal blocked, attempt_event, released_event
        attempt_event = asyncio.Event()
        released_event = asyncio.Event()
        ticks = 0
        stop = asyncio.Event()

        async def ticker() -> None:
            nonlocal ticks
            while not stop.is_set():
                ticks += 1
                await asyncio.sleep(0)

        ticker_task = asyncio.create_task(ticker())
        with pytest.raises(gov.GovernanceCleanupError):
            await gov.release_q3_leases_async(
                [lease],
                jobs_root=jobs,
                unlink_fn=controlled_unlink,
                immediate_attempts=1,
            )
        key = f"q3:{lease_path.name}"
        supervisor = gov._CLEANUP_TASKS[key]
        while attempts < 3:
            attempt_event.clear()
            await asyncio.wait_for(attempt_event.wait(), timeout=1)
        assert gov._CLEANUP_TASKS[key] is supervisor
        assert len(gov._CLEANUP_TASKS) == 1

        with pytest.raises(gov.GovernanceCleanupError):
            await gov.release_q3_leases_async(
                [lease],
                jobs_root=jobs,
                unlink_fn=controlled_unlink,
                immediate_attempts=1,
            )
        assert gov._CLEANUP_TASKS[key] is supervisor
        assert len(gov._CLEANUP_TASKS) == 1

        blocked = False
        await asyncio.wait_for(released_event.wait(), timeout=1)
        await _wait_until(lambda: key not in gov._CLEANUP_TASKS)
        stop.set()
        await ticker_task
        assert ticks > attempts
        assert not lease_path.exists()
        assert key not in gov._CLEANUP_OBLIGATIONS

    asyncio.run(scenario())


def test_cancelled_q3_lock_wait_queues_obligation_without_late_transaction(tmp_path):
    jobs = tmp_path / "jobs"
    lease = _lease(tmp_path, jobs)
    lease_path = gov.q3_lease_path_for(lease, jobs)
    process, release = _lock_holder(gov.governance_paths(jobs).admission_mutex, tmp_path)

    async def scenario() -> None:
        waiting = asyncio.Event()
        unlink_calls = 0

        def cancel_check() -> bool:
            waiting.set()
            return False

        def unlink(path: Path) -> None:
            nonlocal unlink_calls
            unlink_calls += 1
            path.unlink()

        task = asyncio.create_task(
            gov.release_q3_leases_async(
                [lease],
                jobs_root=jobs,
                cancel_check=cancel_check,
                unlink_fn=unlink,
            )
        )
        await waiting.wait()
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task
        key = f"q3:{lease_path.name}"
        assert unlink_calls == 0
        assert lease_path.exists()
        assert key in gov._CLEANUP_OBLIGATIONS
        assert key in gov._CLEANUP_TASKS
        await _cancel_supervisors()
        release.write_text("release", encoding="utf-8")
        await asyncio.sleep(0.05)
        assert unlink_calls == 0
        assert lease_path.exists()

    try:
        asyncio.run(scenario())
    finally:
        if process.poll() is None:
            _finish_holder(process, release)
        else:
            out, err = process.communicate()
            assert process.returncode == 0, f"{out}\n{err}"
