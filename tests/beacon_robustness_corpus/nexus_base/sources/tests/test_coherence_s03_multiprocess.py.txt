BASE_NODES = [s02_node("A", name="alpha"), s02_node("B", name="beta")]


def _producer() -> JournalProducer:
    return JournalProducer(
        pid=os.getpid(), pid_create_time=1.0, owner_kind="process", ownership_kind="none"
    )


def _append(paths, effects):
    return append_journal_transaction(
        paths, operation_kind="test", effects=effects, producer=_producer()
    )


def _setup_base(paths, tmp_path: Path, nodes, *, watermark: int = 0):
    snapshot_dir = tmp_path / "snapshots"
    snapshot_dir.mkdir(parents=True, exist_ok=True)
    manifest = read_manifest(paths, required=True)
    candidate = write_snapshot_candidate(
        new_snapshot_candidate_path(snapshot_dir),
        {"nodes": nodes},
        account_scope_id=paths.account_root.name,
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        incorporated_through_seq=watermark,
        publication_kind="full_refresh",
    )
    write_manifest(
        paths,
        manifest_with_updates(
            manifest,
            base_snapshot_path=candidate,
            coherence_mode="transition",
            base_incorporated_through_seq=watermark,
            journal_floor_seq=watermark + 1,
        ),
        expected_epoch_id=manifest.epoch_id,
        expected_head_seq=manifest.journal_head_seq,
    )
    return read_manifest(paths, required=True)

def test_two_processes_replay_same_tail_and_converge(tmp_path):
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    manifest = _setup_base(paths, tmp_path, BASE_NODES)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])
    _append(paths, [SemanticEffect.node_fields("A", ["name"], {"name": "alpha-2"})])

    first = _adopt_in_subprocess(paths, tmp_path, "first.json")
    second = _adopt_in_subprocess(paths, tmp_path, "second.json")

    # Both processes converged on the identical adopted state.
    assert first["epoch_id"] == manifest.epoch_id == second["epoch_id"]
    assert first["adopted_sequence"] == 2 == second["adopted_sequence"]
    assert first["nodes"] == second["nodes"]
    assert first["nodes"]["C"]["name"] == "gamma"
    assert first["nodes"]["A"]["name"] == "alpha-2"
    assert first["disposition"] == "epoch_reloaded" == second["disposition"]


def test_process_appends_then_another_adopts_incrementally(tmp_path):
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)

    # Process 1 adopts the base lineage.
    before = _adopt_in_subprocess(paths, tmp_path, "before.json")
    assert before["adopted_sequence"] == 0

    # This process appends two records; process 2 must adopt the tail.
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])
    _append(paths, [SemanticEffect.node_create(s02_node("D", name="delta"))])

    after = _adopt_in_subprocess(paths, tmp_path, "after.json")
    assert after["adopted_sequence"] == 2
    assert after["nodes"]["C"]["name"] == "gamma"
    assert after["nodes"]["D"]["name"] == "delta"
    # The earlier process's state is a strict prefix subset of the new one.
    assert set(before["nodes"]) <= set(after["nodes"])


def test_epoch_cut_in_one_process_then_reload_in_another(tmp_path):
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    manifest = _setup_base(paths, tmp_path, BASE_NODES)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])

    before = _adopt_in_subprocess(paths, tmp_path, "before.json")
    assert before["epoch_id"] == manifest.epoch_id

    # Simulate the S02 epoch cut (new base incorporating through seq 1).
    snapshot_dir = tmp_path / "snapshots"
    current = read_manifest(paths, required=True)
    new_id = "f" * 32
    candidate = write_snapshot_candidate(
        new_snapshot_candidate_path(snapshot_dir),
        {"nodes": BASE_NODES + [s02_node("C", name="gamma")]},
        account_scope_id=ACCOUNT_A,
        epoch_number=current.epoch_number + 1,
        epoch_id=new_id,
        incorporated_through_seq=1,
        publication_kind="full_refresh",
    )
    write_manifest(
        paths,
        manifest_with_updates(
            current,
            epoch_number=current.epoch_number + 1,
            epoch_id=new_id,
            previous_epoch_id=current.epoch_id,
            base_snapshot_path=candidate,
            base_incorporated_through_seq=1,
            journal_floor_seq=2,
        ),
        expected_epoch_id=current.epoch_id,
        expected_head_seq=1,
    )
    # A post-cut record in the NEW epoch lineage.
    record = JournalTransaction(
        schema_version=1,
        account_scope_id=ACCOUNT_A,
        sequence=2,
        previous_sequence=1,
        transaction_id="postcut" + "0" * 24,
        published_epoch_id=new_id,
        operation_kind="test",
        committed_at=NOW,
        producer=_producer(),
        verification=(),
        effects=(SemanticEffect.node_create(s02_node("E", name="echo")),),
    )
    atomic_write_json_durable(
        paths.journal_dir / journal_record_filename(2), record.to_dict()
    )
    cut_manifest = read_manifest(paths, required=True)
    write_manifest(
        paths,
        manifest_with_updates(cut_manifest, journal_head_seq=2),
        expected_epoch_id=new_id,
        expected_head_seq=1,
    )

    after = _adopt_in_subprocess(paths, tmp_path, "after.json")
    assert after["epoch_id"] == new_id
    assert after["epoch_number"] == current.epoch_number + 1
    assert after["adopted_sequence"] == 2
    assert after["nodes"]["C"]["name"] == "gamma"  # from the new base
    assert after["nodes"]["E"]["name"] == "echo"  # from the post-cut tail

    CacheViewPublisher, adopt_manifest_head, coherence_paths, unpopulated_cache_view,

    payload = {

        "ids": sorted(
            str(n.get("id"))
            for n in view.nodes_export.get("nodes", [])

def test_s03b_cross_process_generation_pinning(tmp_path):
    """S03-B multiprocess dimension of the one-generation invariant: process
    A pins a view at sequence 0; the PARENT appends a journal record;
    process B adopts and sees sequence 1; process A — still inside its
    operation — never re-adopts and its pinned view is byte-stable.
    """
    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    _setup_base(paths, tmp_path, BASE_NODES)

    holder_out = tmp_path / "holder.json"
    ready = tmp_path / "ready.sentinel"
    go = tmp_path / "go.sentinel"
    proc = launch_subprocess(
        _HOLD_AND_REDUMP,
        ACCOUNT_A,
        str(tmp_path / "cartographer_jobs"),
        str(tmp_path / "snapshots"),
        str(holder_out),
        str(ready),
        str(go),
    )
    try:
        assert wait_for_file(ready, timeout=30.0), "holder process never signaled ready"
        assert wait_for_file(holder_out, timeout=10.0)

        # The parent (a second writer) appends a record; process B adopts it.
        _append(paths, [SemanticEffect.node_create(s02_node("D", name="delta"))])
        adopted_b = _adopt_in_subprocess(paths, tmp_path, "b.json")
        assert adopted_b["adopted_sequence"] == 1
        assert "D" in adopted_b["nodes"]

        # Release the holder; it re-dumps WITHOUT re-adopting.
        touch(go)
        assert wait_for_file(Path(str(holder_out) + ".second"), timeout=30.0)
    finally:
        proc.wait(timeout=30)
    assert proc.returncode == 0

    first = json.loads(holder_out.read_text(encoding="utf-8"))
    second = json.loads(
        Path(str(holder_out) + ".second").read_text(encoding="utf-8")
    )
    assert first["adopted_sequence"] == 0
    # The pinned operation view is byte-stable across the external advance.
    assert second == first
    assert "D" not in second["ids"]

def _launch_barriered_adopter(tmp_path: Path, name: str):
    out_file = tmp_path / f"{name}.json"
    ready_file = tmp_path / f"ready-{name}"
    proc = launch_subprocess(
        _ADOPT_AND_DUMP_BARRIER,
        ACCOUNT_A,
        str(tmp_path / "cartographer_jobs"),
        str(tmp_path / "snapshots"),
        str(out_file),
        str(ready_file),
        str(tmp_path / "go"),
    )
    return proc, out_file, ready_file


def _collect_barriered(proc, out_file: Path) -> dict:
    stdout, stderr = proc.communicate(timeout=90)
    assert proc.returncode == 0, (
        f"concurrent adopter failed:\nSTDOUT:\n{stdout}\nSTDERR:\n{stderr}"
    )
    return json.loads(out_file.read_text(encoding="utf-8"))


def _run_two_barriered_adopters(tmp_path: Path):
    """Launch two adopters that can ONLY proceed once BOTH are staged
    immediately before the contested adoption."""

    proc_a, out_a, ready_a = _launch_barriered_adopter(tmp_path, "A")
    try:
        proc_b, out_b, ready_b = _launch_barriered_adopter(tmp_path, "B")
        try:
            assert wait_for_file(ready_a, timeout=30), "adopter A never became ready"
            assert wait_for_file(ready_b, timeout=30), "adopter B never became ready"
            touch(tmp_path / "go")
            return _collect_barriered(proc_a, out_a), _collect_barriered(proc_b, out_b)
        finally:
            if proc_b.poll() is None:
                proc_b.kill()
    finally:
        if proc_a.poll() is None:
            proc_a.kill()


def test_concurrent_processes_orphan_recovery_exactly_once(tmp_path):
    """Two real subprocesses, released simultaneously against one orphaned
    head: the coherence mutex serializes recovery, the head advances
    EXACTLY once, and both converge on identical state."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    manifest = _setup_base(paths, tmp_path, BASE_NODES)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])
    orphan = JournalTransaction(
        schema_version=1,

        sequence=2,
        previous_sequence=1,
        transaction_id="orphanrace" + "0" * 23,
        published_epoch_id=manifest.epoch_id,
        operation_kind="test",
        committed_at=NOW,
        producer=_producer(),
        verification=(),
        effects=(SemanticEffect.node_create(s02_node("D", name="delta")),),
    )
    atomic_write_json_durable(
        paths.journal_dir / journal_record_filename(2), orphan.to_dict()

    assert result_b["disposition"] in ("epoch_reloaded", "tail_replayed", "already_current")

    real_recover = adoption_module._recover_orphans_locked
    def _parked_recover(paths_arg, manifest_arg, *, fs_ops=None):
        (work / "a-entered-recovery").write_text("1", encoding="utf-8")

    # invocation 2 fires only after an attempt found A's mutex busy.

        calls["n"] += 1

def test_orphan_recovery_serializes_on_the_coherence_mutex(tmp_path):
    """POSITIVE mutex proof (seal-3 item 1): child A parks INSIDE the held
    coherence mutex mid-recovery; child B's cancel_check fires its SECOND
    invocation (which only happens after a lock attempt found the mutex
    busy) before the parent releases A.  B's contention on A's held mutex
    is therefore witnessed directly, never inferred from timing.

    This test guards SUCCESSFUL exclusion/serialization: with the mutex
    working, BOTH processes adopt cleanly.  An independent second wall
    survives its removal: the recovery head advance is a CAS-guarded
    write_manifest(expected_head_seq=...), so a real double recovery
    would produce one winner and one loud manifest conflict — never a
    silent double-advance.  Both walls are intentional.
    """

    paths = _orphan_race_fixture(tmp_path)

    proc_a, out_a = _launch_recovery_child(tmp_path, "A")
    try:
        assert wait_for_file(tmp_path / "a-entered-recovery", timeout=30), (
            "A never entered its parked recovery"
        )
        proc_b, out_b = _launch_recovery_child(tmp_path, "B")
        try:
            # THE positive proof: B's second cancel_check invocation only
            # fires after a lock attempt found A's mutex busy.
            assert wait_for_file(tmp_path / "b-contended", timeout=30), (
                "B never contended on A's held coherence mutex"
            )
            touch(tmp_path / "release-a")
            result_a = _collect_barriered(proc_a, out_a)
            result_b = _collect_barriered(proc_b, out_b)
        finally:
            if proc_b.poll() is None:
                proc_b.kill()
    finally:
        if proc_a.poll() is None:
            proc_a.kill()

    # Exactly one head advance, and it is A's.
    assert (tmp_path / "advance-A").exists()
    assert not (tmp_path / "advance-B").exists()
    final_manifest = read_manifest(paths, required=True)
    assert final_manifest.journal_head_seq == 2  # advanced exactly once
    # The valid orphan was adopted, never quarantined.
    assert not any(paths.quarantine_dir.iterdir())
    # Both processes converge on identical epoch/head/state.
    assert result_a["adopted_sequence"] == 2 == result_b["adopted_sequence"]
    assert result_a["epoch_id"] == result_b["epoch_id"]
    assert result_a["nodes"] == result_b["nodes"]
    assert result_a["nodes"]["D"]["name"] == "delta"


def test_orphan_recovery_mutex_proof_fails_when_mutex_disabled(tmp_path):
    """MUTATION-SENSITIVITY of the positive proof (seal-3 item 1): with the
    adoption slow-path coherence mutex replaced by a no-op async context
    (injected by env inside the child scripts — no production hook), B's
    lock attempts succeed immediately, its cancel_check never reaches the
    contention-witnessing second invocation, and the b-contended marker is
    NEVER produced.  The parked-A protocol therefore proves the repaired
    serialization test above cannot pass vacuously: without the mutex, the
    evidence it requires cannot exist.  In this mutation world B recovers
    the orphan WITHOUT ever contending (advance-B, no advance-A) — and in a
    genuine simultaneous double-recovery the CAS-guarded manifest write
    remains the independent second wall against a silent double-advance.
    """

    paths = _orphan_race_fixture(tmp_path)

    proc_a, out_a = _launch_recovery_child(tmp_path, "A", mutate_no_mutex=True)
    try:
        assert wait_for_file(tmp_path / "a-entered-recovery", timeout=30)
        proc_b, out_b = _launch_recovery_child(tmp_path, "B", mutate_no_mutex=True)
        try:
            # The marker the serialization test REQUIRES can never appear.
            assert not wait_for_file(tmp_path / "b-contended", timeout=8), (
                "B contended even with the coherence mutex disabled — the "
                "serialization test would be vacuous"
            )
            touch(tmp_path / "release-a")
            _collect_barriered(proc_b, out_b)
            # A's late double-recovery either completes as a no-op or loses
            # loudly via the CAS second wall (ManifestConflictError) — both
            # are documented no-mutex outcomes; the marker assertions below
            # are the point.
            a_stdout, a_stderr = proc_a.communicate(timeout=90)
            if proc_a.returncode != 0:
                assert "ManifestConflict" in a_stderr, (
                    f"A failed for an unexpected reason:\n{a_stderr}"
                )
        finally:
            if proc_b.poll() is None:
                proc_b.kill()
    finally:
        if proc_a.poll() is None:
            proc_a.kill()

    # B recovered WITHOUT contending (and A consequently advanced nothing).
    assert (tmp_path / "advance-B").exists()
    assert not (tmp_path / "advance-A").exists()
    final_manifest = read_manifest(paths, required=True)
    assert final_manifest.journal_head_seq == 2


def test_concurrent_processes_same_tail_converge(tmp_path):
    """Two real subprocesses, released simultaneously (ready/go rendezvous),
    replay the same two-record tail and converge on byte-identical adopted
    state.  This test proves independent two-process CONVERGENCE under
    concurrent release; it deliberately does NOT claim mutex-serialization
    evidence (that contract belongs to the orphan-recovery proof above)."""

    paths = make_coherence_root(tmp_path, ACCOUNT_A)
    manifest = _setup_base(paths, tmp_path, BASE_NODES)
    _append(paths, [SemanticEffect.node_create(s02_node("C", name="gamma"))])
    _append(paths, [SemanticEffect.node_fields("A", ["name"], {"name": "alpha-2"})])

    result_a, result_b = _run_two_barriered_adopters(tmp_path)

    assert result_a["adopted_sequence"] == 2 == result_b["adopted_sequence"]
    assert result_a["epoch_id"] == result_b["epoch_id"] == manifest.epoch_id
    assert result_a["nodes"] == result_b["nodes"]
    assert result_a["nodes"]["A"]["name"] == "alpha-2"
