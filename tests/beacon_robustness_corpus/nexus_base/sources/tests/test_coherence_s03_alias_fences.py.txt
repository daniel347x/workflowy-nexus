def _dotted_function_identities(path: Path) -> set[str]:
    """Every exact dotted function/method identity defined in one source
    file (module-level, class-nested, and genuinely nested), using the same
    stack grammar as the scanner's FunctionMapper."""

    import ast as _ast

    tree = _ast.parse(path.read_text(encoding="utf-8"))
    identities: set[str] = set()
    stack: list[str] = []

    class _V(_ast.NodeVisitor):
        def _fn(self, node):
            stack.append(getattr(node, "name", "<fn>"))
            if isinstance(node, (_ast.FunctionDef, _ast.AsyncFunctionDef)):
                identities.add(".".join(stack))
            for child in _ast.iter_child_nodes(node):
                self.visit(child)
            stack.pop()

        visit_FunctionDef = _fn
        visit_AsyncFunctionDef = _fn
        visit_ClassDef = _fn

    _V().visit(tree)
    return identities


def _registry_identity_failures(registry: dict) -> list[str]:
    """R2-3: every identity-bearing registry entry must resolve to an
    ACTUAL exact dotted AST function/method identity in its named file —
    an unqualified class method (a bare leaf name) fails, module-level
    functions remain valid as module-level names, and a
    leaf-name-somewhere check is NOT sufficient."""

    from coherence_s03_fence_scanner import (
        _allowlist_entries,
        _disposition_functions,
        _spawn_dispositions,
    )

    sections = {
        "consumer_dispositions": sorted(_disposition_functions(registry)),
        "task_spawns": sorted(_spawn_dispositions(registry)),
        "allowlist": sorted(_allowlist_entries(registry)),
    }
    identities_by_file: dict[str, set[str]] = {}
    failures: list[str] = []
    for section, pairs in sections.items():
        for rel, fn in pairs:
            path = REPO / rel
            if not path.is_file():
                failures.append(
                    f"{section}: file missing on disk: {rel} (identity {fn!r})"
                )
                continue
            if rel not in identities_by_file:
                identities_by_file[rel] = _dotted_function_identities(path)
            if fn not in identities_by_file[rel]:
                failures.append(
                    f"{section}: {rel}::{fn} is not an exact dotted "
                    "function/method identity (bare or mismatched key)"
                )
    return failures


def test_registry_identities_exist_exactly() -> None:

def _enclosing_map(path: Path) -> dict[int, str]:
    import ast as _ast

    tree = _ast.parse(path.read_text(encoding="utf-8"))
    mapper: dict[int, str] = {}

    class V(_ast.NodeVisitor):
        def __init__(self) -> None:
            self.stack: list[str] = []

        def _fn(self, node):
            self.stack.append(getattr(node, "name", "<fn>"))
            for child in _ast.iter_child_nodes(node):
                self.visit(child)
            self.stack.pop()

        visit_FunctionDef = _fn
        visit_AsyncFunctionDef = _fn
        visit_ClassDef = _fn

        def generic_visit(self, node):
            lineno = getattr(node, "lineno", None)
            if lineno is not None and self.stack:
                mapper[lineno] = ".".join(self.stack)
            super().generic_visit(node)

    V().visit(tree)
    return mapper

def _build_machine_inventory() -> dict:
    """The §11 machine inventory: allowlist + dispositions + scan summary.

    R3.3: the spawn census uses the ONE shared collector
    (``collect_spawn_sites`` — the same predicate the production fence
    enforces), never a second, weaker grammar copy.
    """

    import ast as _ast

    from coherence_s03_fence_scanner import (
        _allowlist_functions,
        collect_spawn_sites,
    )

    registry = REGISTRY
    accessor_sites: list[str] = []
    spawn_sites: list[str] = []
    for path in production_files():
        rel = str(path.relative_to(REPO)).replace("\\", "/")
        text = path.read_text(encoding="utf-8")
        tree = _ast.parse(text)
        fn_map = _enclosing_map(path)
        for node in _ast.walk(tree):
            if not isinstance(node, _ast.Call):
                continue
            func = node.func
            name = func.id if isinstance(func, _ast.Name) else (
                func.attr if isinstance(func, _ast.Attribute) else ""
            )
            if name == "_get_nodes_export_cache_nodes":
                # LINE-NUMBER-FREE key (was f"{rel}:{node.lineno}::{fn}").
                # This artifact is a CURRENCY lock: its question is "has the
                # SET of callsites changed?", not "where are they?".  Keying on
                # line numbers made it churn on ANY unrelated edit that shifted
                # the file -- a 39-line insertion in server.py tripped it exactly
                # like a real bypass would, which trains alarm fatigue and hides
                # the signal.  The entry stays a LIST (not a set) so a second
                # callsite inside the same function still appears as a repeated
                # element: count sensitivity is preserved, position is not.
                accessor_sites.append(f"{rel}::{fn_map.get(node.lineno, '<module>')}")
        for line, fn, form in collect_spawn_sites(rel, text):
            # LINE-NUMBER-FREE key (was f"{rel}:{line}::{fn} [{form}]"); same
            # reasoning as accessor_sites above.  `line` is still consumed by
            # collect_spawn_sites internally -- only the emitted key drops it.
            spawn_sites.append(f"{rel}::{fn} [{form}]")
    return {
        "manifest": "S03-B machine inventory (§11) — boundary owners, H2 allowlist, consumer dispositions, accessor/spawn sites",
        "baseline_commit": "06cd024",
        "registry_file": "tests/coherence_s03b_registry.json",
        "allowlist_size": len(registry["allowlist"]),
        "allowlist": sorted(f"{f}::{fn}" for f, fn in _allowlist_functions(registry)),
        "consumer_dispositions": registry["consumer_dispositions"],
        "task_spawns": registry["task_spawns"],
        "ws_actions": registry["ws_actions"],
        "cache_free_operations": registry["mcp_dispatch"]["cache_free_operations"],
        "accessor_callsites": sorted(accessor_sites),
        "task_spawn_sites": sorted(spawn_sites),
    }
