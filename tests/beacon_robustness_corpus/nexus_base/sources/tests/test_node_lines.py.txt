def client_for(monkeypatch, note="one\ntwo\nthree\n", *, nodes=None, status=None):
    if nodes is None:
        nodes = [{"id": ROOT, "name": NAME, "note": "STALE CACHE", "parent_id": None}]
    client = make_boundary_capable(object.__new__(server.WorkFlowyClient), nodes=nodes, boundary_status=status)
    client.get_node = AsyncMock(return_value=WorkFlowyNode(id=ROOT, name=NAME, note=html.escape(note)))
    client.update_node = AsyncMock(side_effect=AssertionError("no writes"))
    client.create_node = AsyncMock(side_effect=AssertionError("no writes"))
    client._adopt_coherence_head = AsyncMock(wraps=client._adopt_coherence_head)
    monkeypatch.setattr(server, "get_client", lambda: client)
    monkeypatch.setattr(server, "_rate_limiter", None)
    return client


async def dispatch_async(verb, params):
    return await dispatch_operation(
        server.build_node_registry(), verb, params,
        operation_boundary=server._mcp_operation_boundary,
        mutation_admission=server._mcp_mutation_admission,
        exception_mapper=server._mcp_exception_mapper,
        result_finalizer=server._mcp_operation_result_finalizer,
    )


def dispatch(verb, params):
    return asyncio.run(dispatch_async(verb, params))

def subtree():
    # Intentionally shuffled cache order: traversal must follow priority.
    # B is COMPLETED (and suppressed by name): hidden by the new default.
    return [
        {"id": ROOT, "name": NAME, "note": "hit ROOT &lt;", "parent_id": None},
        {"id": B, "name": "Spec B #wfy-suppress", "note": "hit B", "parent_id": ROOT, "priority": 20, "completed": True},
        {"id": A, "name": "other parent", "note": "hit A", "parent_id": ROOT, "priority": 10},
        {"id": C, "name": "[ap:CHILD01] Spec C #wfy-ns", "note": "hit C", "parent_id": A},
        {"id": OUT, "name": "Spec OUT", "note": "hit OUT", "parent_id": None},
    ]


SUPPRESSED = "55555555-5555-4555-8555-555555555555"


def subtree_deep():
    # A completed PARENT hides its whole subtree by default; a suppressed
    # (non-completed) node stays included -- search is a grep, not a render.
    return [
        {"id": ROOT, "name": NAME, "note": "root note", "parent_id": None},
        {"id": A, "name": "done parent", "note": "hit parent", "parent_id": ROOT, "priority": 10, "completed": True, "completedAt": 123},
        {"id": C, "name": "child of done", "note": "hit child", "parent_id": A},
        {"id": SUPPRESSED, "name": "kept #wfy-suppress", "note": "hit suppressed", "parent_id": ROOT, "priority": 20},
    ]


def test_search_subtree_pinned_read_only_hides_completed_by_default(monkeypatch):
    client = client_for(monkeypatch, nodes=subtree())
    view = client._cache_view_publisher.capture()
    before = copy.deepcopy(view.nodes_export)
    result = dispatch("search", {"root_handle": "[ap:LINES01]", "query": "hit", "context_lines": 0})
    assert result.index("hit ROOT &lt;") < result.index("hit A") < result.index("hit C")
    assert "hit B" not in result
    assert "hit OUT" not in result
    assert "Source: pinned operation cache (root and all descendants; completed nodes hidden by default — 1 completed hidden; suppressed nodes and child Docks included; include_completed=true searches them too)." in result
    assert client._cache_view_publisher.capture() is view
    assert view.nodes_export == before
    client.get_node.assert_not_awaited()
    client.update_node.assert_not_awaited()
    client.create_node.assert_not_awaited()
    client._adopt_coherence_head.assert_awaited_once()

def test_subtree_completed_parent_hides_its_descendants_and_counts_them(monkeypatch):
    client_for(monkeypatch, nodes=subtree_deep())
    result = dispatch("search", {"root_handle": ROOT, "query": "hit", "context_lines": 0})

    full = dispatch("search", {"root_handle": ROOT, "query": "hit", "context_lines": 0, "include_completed": True})
    assert "hit parent" in full and "hit child" in full

def page_metadata(report):
    match = re.search(
        r"^Pagination: offset=(\d+); returned_matches=(\d+); returned_occurrences=(\d+); "
        r"has_more=(true|false); next_offset=(\d+|null)\.$", report, re.MULTILINE,
    )
    assert match, report
    offset, count, _occurrences, more, next_offset = match.groups()
    return int(offset), int(count), more == "true", None if next_offset == "null" else int(next_offset)


def page_occurrences(report):
    match = re.search(r"^Pagination: .*?returned_occurrences=(\d+);", report, re.MULTILINE)
    assert match, report
    return int(match.group(1))


def numbered_rows(report):
    """Displayed grep rows (hits and context) as 1-based numbers."""
    return [
        int(row.split("|")[0]) for row in report.splitlines()
        if re.match(r"^\s*\d+ \| ", row)
    ]


def span_finder(token: str, *, case_sensitive: bool = False):
    pattern = re.compile(re.escape(token), 0 if case_sensitive else re.IGNORECASE)
    return lambda line: [match.span() for match in pattern.finditer(line)]


def regex_span_finder(pattern: str):
    compiled = re.compile(pattern, re.IGNORECASE)
    return lambda line: [match.span() for match in compiled.finditer(line)]


def marked_numbers(report):
    return [int(row.split("|")[0][2:]) for row in report.splitlines() if row.startswith("> ")]


@pytest.mark.parametrize("query,extra", [("hit", {}), (r"^hit .*hit$", {"is_regex": True})])
def test_page_through_more_than_200_live_matches(monkeypatch, query, extra):
    # Repeated occurrences, context and CRLF do not consume match offsets.
    note = "\r\n".join(f"gap {n}\r\nhit {n} 📊 &lt; hit" for n in range(451)) + "\r\n"
    client = client_for(monkeypatch, note)
    view = client._cache_view_publisher.capture()
    offset, numbers, sizes = 0, [], []
    while offset is not None:
        result = dispatch("search", {
            "node_handle": "[ap:LINES01]", "query": query,
            "max_matches": 200, "offset": offset, "context_lines": 1, **extra,
        })
        actual_offset, count, more, next_offset = page_metadata(result)
        assert actual_offset == offset
        page_numbers = marked_numbers(result)
        assert len(page_numbers) == count
        assert more is (next_offset is not None)
        if more:
            assert next_offset == offset + count
            assert f"Repeat the same search with offset={next_offset}" in result
        numbers.extend(page_numbers)
        sizes.append(count)
        assert len(sizes) <= 3  # Fail loudly rather than loop on a broken cursor.
        offset = next_offset
    assert sizes == [200, 200, 51]
    assert numbers == list(range(2, 903, 2))
    assert client.get_node.await_count == client._adopt_coherence_head.await_count == 3
    assert client._cache_view_publisher.capture() is view
    client.update_node.assert_not_awaited()
    client.create_node.assert_not_awaited()
    assert current_operation_cache(required=False) is None

def test_pure_pager_stops_at_one_extra_hit_without_consuming_remaining_documents():
    visited = []
    def documents():
        yield ROOT, NAME, "hit skipped\nno\nhit selected"
        yield A, "next document", "gap\nhit lookahead\nnot scanned"
        raise AssertionError("must stop without consuming another document")
    def matches(line):
        visited.append(line)
        return [(0, 3)] if "hit" in line else []
    result = search_note_documents(documents(), matches, offset=1, max_matches=1, context_lines=0)
    assert visited == ["hit skipped", "no", "hit selected", "gap", "hit lookahead"]
    assert page_metadata(result) == (1, 1, True, 2)
    assert numbered_rows(result) == [3]
    assert f"lines in {A}" not in result
    assert result.splitlines()[0].startswith("INFORMATIONAL, CRITICAL DETAIL: PAGINATION ACTIVE.")

def test_each_subtree_page_adopts_current_view_without_retaining_prior_view(monkeypatch):
    from workflowy_mcp.coherence.cache_view import CacheView
    client = client_for(monkeypatch, nodes=subtree())
    params = {"root_handle": ROOT, "query": "hit", "max_matches": 1, "context_lines": 0}
    assert page_metadata(dispatch("search", params)) == (0, 1, True, 1)
    original = client._cache_view_publisher.capture()
    changed = subtree()
    changed[2]["note"] = "hit NEW A"
    next_view = CacheView.build(
        epoch_number=original.epoch_number, epoch_id=original.epoch_id,
        adopted_sequence=1, nodes_export={"nodes": changed},
    )
    # Simulate another operation's publication, never mutate the pinned map.
    monkeypatch.setattr(client._cache_view_publisher, "capture", lambda: next_view)
    second = dispatch("search", {**params, "offset": 1})
    assert "1 | hit NEW A" in second
    assert page_metadata(second) == (1, 1, True, 2)
    assert client._adopt_coherence_head.await_count == 2
    client.get_node.assert_not_awaited()
    client.update_node.assert_not_awaited()
    assert current_operation_cache(required=False) is None

def test_collect_out_param_reports_page_facts_and_changes_nothing():
    matcher = span_finder("hit")
    documents = [("u1", "N", "hit one\nhit two"), ("u2", "M", "hit three")]
    plain = search_note_documents(
        documents, matcher, context_lines=0, max_matches=2,
    )
    collected: dict = {}
    with_collect = search_note_documents(
        documents, matcher, context_lines=0, max_matches=2, collect=collected,
    )
    assert with_collect == plain  # collect is an out-param, never a shape change
    assert collected == {
        "has_more": True,
        "next_offset": 2,
        "returned_matches": 2,
        "returned_occurrences": 2,
    }
    complete: dict = {}
    search_note_documents(
        documents, matcher, context_lines=0, max_matches=50, collect=complete,
    )
    assert complete == {
        "has_more": False,
        "next_offset": None,
        "returned_matches": 3,
        "returned_occurrences": 3,
    }
