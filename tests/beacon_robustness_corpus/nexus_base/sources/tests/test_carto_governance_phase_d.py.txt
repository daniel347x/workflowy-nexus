ACCOUNT = "d" * 32


def _cache() -> dict[str, dict]:
    return {
        "vault": {"id": "vault"},
        "repo": {"id": "repo", "parent_id": "vault"},
        "file": {"id": "file", "parent_id": "repo"},
    }


def _footprint(tmp_path: Path) -> dict:
    target = tmp_path / "repo" / "file.py"
    target.parent.mkdir(parents=True, exist_ok=True)
    target.touch(exist_ok=True)
    return gov.compute_footprint(target, "file", _cache(), kind="file")


def _claim(
    tmp_path: Path,
    *,
    job_id: str = "job",
    pid: int | None = None,
    pid_create_time: float | None = None,
    pid_start_ticks: int | None = None,
    pid_boot_id: str | None = None,
    state: str = "running",
    owner: str = "detached_worker",
    launched_at: str | None = None,
    spawn_started_at: str | None = None,
) -> dict:
    process = psutil.Process(os.getpid())
    return gov.build_claim(
        job_id=job_id,
        job_type="CARTO_REFRESH",
        dedupe_key="a" * 64,
        account_scope_id=ACCOUNT,
        footprint=_footprint(tmp_path),
        launcher="system",
        parent_job_id=None,
        launcher_pid=os.getpid(),
        pid=os.getpid() if pid is None else pid,
        pid_create_time=(process.create_time() if pid_create_time is None else pid_create_time),
        pid_start_ticks=pid_start_ticks,
        pid_boot_id=pid_boot_id,
        handoff_state=state,
        spawn_started_at=spawn_started_at,
        execution_owner=owner,
        launched_at=launched_at,
    )


def _identity_reader(mapping):
    """U01 ``identity_reader`` double: map pid -> ProcessIdentity (or a
    zero-arg callable producing one); absent pids raise NoSuchProcess."""

    def _read(pid):
        entry = mapping.get(int(pid))
        if entry is None:
            raise psutil.NoSuchProcess(pid)
        return entry() if callable(entry) else entry

    return _read


def _write_job(
    jobs: Path,
    claim: dict | None,
    *,
    job_id: str = "job",
    status: str = "running",
    pid: int | None = None,
    created_at: str | None = None,
    job_type: str = "CARTO_REFRESH",
) -> Path:
    jobs.mkdir(parents=True, exist_ok=True)
    path = jobs / f"{job_id}.json"
    payload = {
        "id": job_id,
        "type": job_type,
        "status": status,
        "pid": pid if pid is not None else (claim or {}).get("pid"),
        "created_at": created_at or gov.utc_now_iso(),
        "updated_at": created_at or gov.utc_now_iso(),
        "cache_refresh_required": True,
    }
    if claim is not None:
        payload["claim_id"] = claim["claim_id"]
    gov.atomic_write_json(path, payload)
    return path


def _load(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


class _UnknownPsutil:
    NoSuchProcess = psutil.NoSuchProcess

    @staticmethod
    def Process(_pid):
        raise psutil.AccessDenied(_pid)


def test_owner_liveness_truth_table_reuse_boot_and_unknown():
    """U01 P2 (probe doubles): every row of the one classifier's truth table —
    reuse and boot mismatch are dead; unreadable, unprobeable and legacy are
    unknown; epoch divergence alone is never death evidence."""

    stored = gov.StoredOwnerIdentity(
        pid=123, pid_create_time=10.0, start_ticks=100, boot_id="boot-a"
    )

    def _reader(identity):
        return lambda _pid: identity

    assert (
        gov.classify_owner_liveness(
            stored,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=100, boot_id="boot-a", create_time=10.0)
            ),
        )
        == "alive"
    )
    # Same pid, different incarnation (start ticks differ) — PID reuse.
    assert (
        gov.classify_owner_liveness(
            stored,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=101, boot_id="boot-a", create_time=10.0)
            ),
        )
        == "dead"
    )
    # Different boot.
    assert (
        gov.classify_owner_liveness(
            stored,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=100, boot_id="boot-b", create_time=10.0)
            ),
        )
        == "dead"
    )
    # Epoch moved by seconds with the incarnation unchanged — alive (the
    # retired epsilon comparison classified this live owner dead).
    assert (
        gov.classify_owner_liveness(
            stored,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=100, boot_id="boot-a", create_time=11.0)
            ),
        )
        == "alive"
    )
    # Unreadable probe, unprobeable current, and legacy artifact: unknown.
    assert gov.classify_owner_liveness(stored, identity_reader=_raise_psutil_access_denied) == "unknown"
    assert (
        gov.classify_owner_liveness(
            stored,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=None, boot_id=None, create_time=10.0)
            ),
        )
        == "unknown"
    )
    legacy = gov.StoredOwnerIdentity(pid=123, pid_create_time=10.0, start_ticks=None, boot_id=None)
    assert (
        gov.classify_owner_liveness(
            legacy,
            identity_reader=_reader(
                gov.ProcessIdentity(pid=123, start_ticks=100, boot_id="boot-a", create_time=10.0)
            ),
        )
        == "unknown"
    )

    def _missing(pid):
        raise psutil.NoSuchProcess(pid)

    assert gov.classify_owner_liveness(stored, identity_reader=_missing) == "dead"
    assert gov.classify_owner_liveness(legacy, identity_reader=_missing) == "dead"

def test_cancel_ordering_already_terminal_and_parent_child_independence(tmp_path, monkeypatch):
    jobs = tmp_path / "jobs"
    claim = _claim(tmp_path, job_id="parent", owner="launcher")
    gov.write_claim(claim, jobs)
    job_file = _write_job(jobs, claim, job_id="parent")
    child = _claim(tmp_path, job_id="child", owner="launcher")
    gov.write_claim(child, jobs)
    _write_job(jobs, child, job_id="child")

    events = []
    original_create = gov.create_cancellation_sentinel
    original_merge = gov.merge_write_carto_job_status

    def create(*args, **kwargs):
        events.append("sentinel")
        return original_create(*args, **kwargs)

    def merge(path, payload):
        if payload.get("status") == "cancelling":
            events.append("cancelling")
        return original_merge(path, payload)

    monkeypatch.setattr(gov, "create_cancellation_sentinel", create)
    monkeypatch.setattr(gov, "merge_write_carto_job_status", merge)
    result = asyncio.run(
        gov.cancel_carto_job(
            job_file,
            requested_by="test",
            task_cancel=lambda: events.append("task_cancel"),
            cooperative_timeout_seconds=0,
        )
    )
    assert events[:3] == ["sentinel", "cancelling", "task_cancel"]
    assert result["outcome"] == "kill_refused_in_process_job"
    assert gov.cancellation_sentinel_path("parent", jobs).exists()
    assert not gov.cancellation_sentinel_path("child", jobs).exists()

    terminal = _load(job_file)
    terminal["status"] = "failed"
    original_merge(job_file, terminal)
    before = job_file.read_bytes()
    gov.delete_cancellation_sentinel("parent", jobs)
    result = asyncio.run(gov.cancel_carto_job(job_file, requested_by="test"))
    assert result["outcome"] == "already_terminal"
    assert job_file.read_bytes() == before
    assert not gov.cancellation_sentinel_path("parent", jobs).exists()


class _FakeProcess:
    def __init__(
        self,
        pid,
        *,
        create_time,
        cmdline,
        mutex_path: Path | None = None,
        timeout_once=False,
    ):
        self.pid = pid
        self._create_time = create_time
        self._cmdline = cmdline
        self._mutex_path = mutex_path
        self._timeout_once = timeout_once
        self.terminated = False
        self.killed = False

    def create_time(self):
        return self._create_time

    def cmdline(self):
        return self._cmdline

    def terminate(self):
        if self._mutex_path is not None:
            with gov.os_lock(self._mutex_path, timeout_seconds=0.2):
                pass
        self.terminated = True

    def wait(self, timeout):
        if self._timeout_once and not self.killed:
            self._timeout_once = False
            raise psutil.TimeoutExpired(timeout, self.pid)
        return 0

    def kill(self):
        self.killed = True

def test_postpoll_ownership_reread_targets_transitioned_child(tmp_path):
    jobs = tmp_path / "jobs"

        tmp_path,

        owner="launcher",
        state="spawning",
        spawn_started_at=gov.utc_now_iso(),
    )
    gov.write_claim(claim, jobs)
    job_file = _write_job(jobs, claim)
    child = _FakeProcess(
        300,
        create_time=3,
        cmdline=["python", "--carto-job-file", str(job_file)],
    )
    launcher_process = _FakeProcess(
        os.getpid(),
        create_time=claim["pid_create_time"],
        cmdline=[],
    )
    process_module = SimpleNamespace(
        Process=lambda pid: child if int(pid) == 300 else launcher_process,
        NoSuchProcess=psutil.NoSuchProcess,
        TimeoutExpired=psutil.TimeoutExpired,
    )
    reader = _identity_reader(
        {
            os.getpid(): gov.ProcessIdentity(
                pid=os.getpid(), start_ticks=None, boot_id=None, create_time=1.0
            ),
            300: gov.ProcessIdentity(pid=300, start_ticks=None, boot_id=None, create_time=3.0),
        }
    )
    clock = iter([0.0, 0.0, 2.0])

    def transition(_seconds):
        claim_path = next(gov.governance_paths(jobs).claims_dir.glob("*.json"))
        current = gov.read_claim(claim_path)
        current["pid"] = 300
        current["pid_create_time"] = 3
        current["execution_owner"] = "detached_worker"
        gov.write_claim(current, jobs)

    result = asyncio.run(
        gov.cancel_carto_job(
            job_file,
            requested_by="test",
            cooperative_timeout_seconds=1,
            sleep_fn=transition,
            monotonic_fn=lambda: next(clock),
            psutil_module=process_module,
            identity_reader=reader,
            sweep_fn=lambda *_args, **_kwargs: None,
        )
    )
    assert result["outcome"] == "forced"
    assert child.terminated


def test_postpoll_incarnation_mismatch_refuses_kill(tmp_path):
    """U01: the cooperative poll sees the stored incarnation alive, then the
    mandatory post-poll ownership reread probes a DIFFERENT incarnation
    (start ticks changed) — exact-kill verification refuses, never fires."""
    jobs = tmp_path / "jobs"
    claim = _claim(tmp_path, pid=400, pid_create_time=1, pid_start_ticks=100)
    gov.write_claim(claim, jobs)
    job_file = _write_job(jobs, claim)
    process = _FakeProcess(
        400,
        create_time=1,
        cmdline=["python", "--carto-job-file", str(job_file)],

def test_job_local_cancel_probe_does_not_cancel_sibling_task():
    async def scenario():
        client = WorkFlowyClientCore.__new__(WorkFlowyClientCore)
        client._operation_cancel_callback = None
        started = asyncio.Event()

        async def cancelled():
            with operation_cancel_scope(lambda: True):
                started.set()
                with pytest.raises(OperationCancelledError):
                    await client._sleep_with_operation_cancel(0.01, "cancelled")

        async def sibling():
            await started.wait()
            with operation_cancel_scope(lambda: False):
                await client._sleep_with_operation_cancel(0.01, "sibling")
                return "ok"

        assert await asyncio.gather(cancelled(), sibling()) == [None, "ok"]

    asyncio.run(scenario())
