# @beacon[
#   id=shim@module,
#   role=stdio SHIM: a thin stdio MCP proxy that finds-or-boots the single HTTP daemon and forwards everything to it,
#   slice_labels=nexus-loading-flow,nexus--daemon,
#   kind=span,
#   show_span=true,
#   comment=What stdio-only hosts (TypingMind connector, Claude Desktop) launch instead of the full server; their JSON blob changes by one word (workflowy-mcp -> workflowy-mcp-shim),
# ]
"""stdio shim for the single Workflowy MCP daemon.

A stdio-only MCP host (TypingMind's connector, Claude Desktop, ...) launches
this instead of the full server. It:

1. calls :func:`workflowy_mcp.daemon.ensure_daemon_running` -- probe the
   daemon's port; if nothing answers, spawn ``python -m workflowy_mcp
   --transport http`` detached (same interpreter, same env) and wait;
2. serves a FastMCP **proxy** over stdio whose every tool/resource/prompt
   forwards to the daemon over Streamable HTTP with the shared bearer token.

It NEVER falls back to running the server in-process: if the daemon cannot be
reached the shim exits non-zero with a clear stderr message. One execution
domain, or a loud failure -- never a second silent domain.

The shim imports nothing from ``server`` (which would build the whole FastMCP
app); it stays a few milliseconds of startup on top of the interpreter.
"""

from __future__ import annotations

import os
import sys

TIMEOUT_ENV_VAR = "WORKFLOWY_MCP_SHIM_CALL_TIMEOUT_S"
DEFAULT_CALL_TIMEOUT_S = 900.0  # long renders / governed mutations can take minutes


def _log(msg: str) -> None:
    print(f"[workflowy-mcp shim] {msg}", file=sys.stderr, flush=True)


# @beacon[
#   id=shim@main,
#   role=main(): ensure daemon, build FastMCPProxy over ProxyClient(daemon_url, bearer), serve stdio,
#   slice_labels=nexus-loading-flow,nexus--daemon,
#   kind=ast,
# ]
def main() -> None:
    """Console-script entrypoint (``workflowy-mcp-shim``)."""
    from workflowy_mcp import daemon as _daemon

    if not _daemon.ensure_daemon_running():
        _log(
            f"FATAL: the Workflowy MCP daemon is not reachable at {_daemon.daemon_url()} "
            "and could not be started. Refusing to run a second in-process server. "
            "Start it manually with `workflowy-mcp --transport http` and check its console/log."
        )
        raise SystemExit(2)

    from fastmcp.server.providers.proxy import FastMCPProxy, ProxyClient

    url = _daemon.daemon_url()
    token = _daemon.shared_token()
    timeout = float(os.getenv(TIMEOUT_ENV_VAR) or DEFAULT_CALL_TIMEOUT_S)

    def _client_factory() -> ProxyClient:
        # Self-healing: re-check the daemon on EVERY call (a ~1 ms TCP probe when
        # it is up). If the daemon died or was closed for a redeploy, the next
        # tool call respawns it with this shim's inherited env -- the human never
        # has to start it by hand while a shim host is running.
        if not _daemon.ensure_daemon_running():
            raise RuntimeError(
                f"Workflowy MCP daemon unreachable at {url} and could not be restarted"
            )
        # A string `auth` is treated as a bearer token by the FastMCP client.
        return ProxyClient(url, auth=token, timeout=timeout)

    proxy = FastMCPProxy(
        client_factory=_client_factory,
        # Present the DAEMON's identity (name/version/instructions) to the host
        # so a TypingMind/Claude Desktop user sees the same server they always did.
        identity="upstream",
        name="NEXUS \u2014 Workflowy MCP Server (shim)",
    )
    _log(f"proxying stdio <-> {url}")
    proxy.run(transport="stdio")


if __name__ == "__main__":
    main()
# @beacon-close[
#   id=shim@module,
# ]
