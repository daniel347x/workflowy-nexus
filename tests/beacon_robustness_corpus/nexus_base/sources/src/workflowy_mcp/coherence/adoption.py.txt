"""Journal adoption engine (S03 §5).

One process-local adoption authority: every cache-backed operation adopts
the current manifest epoch/head through THIS engine before reading cache
state.  Adoption is incremental whenever possible:

- **Fast path (§5.1):** the locally published view already matches the
  manifest ``(epoch_id, journal_head_seq)`` → return the SAME view object;
  no replay, no base reload, no coherence mutex beyond the strict
  lock-free manifest read (S01/S02 durable replacement makes a lock-free
  read complete-or-old).
- **Same-epoch tail replay (§5.3):** clone nothing globally — apply only
  the missing contiguous records ``adopted+1 .. captured_head`` onto a
  private payload and CAS-publish.
- **Epoch reload (§5.4):** discard the old basis, load ONLY the
  manifest-referenced base, replay the active range through the captured
  head, publish one new view.  Timestamp order is never consulted and
  old-epoch node objects are never reused.
- **Failure recovery (§5.5):** a failed incremental apply publishes
  nothing; one clean-base reload + complete active replay to the same
  captured head may recover (``recovered_from_base``); repeated failure is
  a loud typed ``CacheAdoptionError`` with the last good view untouched.

Markers (``adopted_sequence`` / epoch identity) live ONLY on the built
``CacheView``; they become visible with the completed local swap, never
before (§5.7).  An operation linearizes at the manifest epoch/head
captured for its adoption — a concurrently advancing shared head is
picked up by the NEXT operation, never by spinning (§5.6).

Lock doctrine (§5.2 / invariant 11): coherence mutex → publisher mutex,
never the reverse; journal record READS are bounded local reads performed
under the coherence mutex; full cache copying/replay and CAS publication
happen OUTSIDE it.  No network I/O under either.
"""

# ---------------------------------------------------------------------------
# The unpopulated ("no cache") sentinel view
# ---------------------------------------------------------------------------

#: Sentinel epoch identity of the UNPOPULATED view.  A real epoch id is a
#: ``secrets.token_hex(16)`` product; the all-zero form is a reserved,
#: never-published sentinel meaning "this process holds no adopted cache".
UNPOPULATED_EPOCH_ID = "0" * 32


def unpopulated_cache_view(*, account_scope_id: str | None = None) -> CacheView:
    """Transfer a fresh empty sentinel: no adopted cache/lineage, sequence 0."""

    return CacheView.build(
        epoch_number=0,
        epoch_id=UNPOPULATED_EPOCH_ID,
        adopted_sequence=0,
        nodes_export={"nodes": []},
        account_scope_id=account_scope_id,
        ownership="transferred",
    )


def is_unpopulated_view(view: CacheView) -> bool:
    """True iff ``view`` is the no-cache sentinel (never a real lineage)."""

    return view.epoch_id == UNPOPULATED_EPOCH_ID


# ---------------------------------------------------------------------------
# Result
# ---------------------------------------------------------------------------


@dataclass(frozen=True)
# @beacon[
#   id=auto-beacon@CacheAdoptionResult-wqs5,
#   role=CacheAdoptionResult,
#   slice_labels=mci-journal-adoption,
#   kind=ast,
# ]
class CacheAdoptionResult:
    """The outcome of one ``adopt_manifest_head`` call (S03 §4).

    ``to_epoch_id`` / ``to_sequence`` are ALWAYS the markers of ``view``
    (the view actually returned), derived in one place — never the
    captured manifest target, which a lost CAS may leave behind (seal
    item 9).  ``from_*`` describe the basis this adoption attempt began
    from (``None`` for the unpopulated sentinel).  ``records_applied`` on
    an epoch reload / clean-base recovery is the active journal-range
    width replayed over the base (seal O3).
    """

    view: CacheView
    disposition: Literal[
        "already_current", "tail_replayed", "epoch_reloaded", "recovered_from_base"
    ]
    from_epoch_id: str | None
    from_sequence: int | None
    to_epoch_id: str
    to_sequence: int
    records_applied: int
    diagnostics: tuple[dict[str, Any], ...]


class _LocalEpochAdvanced(Exception):
    """Internal control flow: the LOCAL publisher moved to another epoch
    while this adoption was building — restart from the new state."""


# ---------------------------------------------------------------------------
# Small helpers
# ---------------------------------------------------------------------------


def _markers(view: CacheView) -> tuple[int, int]:
    return (view.epoch_number, view.adopted_sequence)


def _result(
    *,
    view: CacheView,
    disposition: str,
    basis: CacheView | None,
    records_applied: int,
    diagnostics: list[dict[str, Any]],
) -> CacheAdoptionResult:
    # to_* describe the view ACTUALLY returned (seal item 9); from_*
    # describe the basis (None/None for the unpopulated sentinel).
    populated_basis = basis is not None and not is_unpopulated_view(basis)
    return CacheAdoptionResult(
        view=view,
        disposition=disposition,  # type: ignore[arg-type]
        from_epoch_id=basis.epoch_id if populated_basis else None,
        from_sequence=basis.adopted_sequence if populated_basis else None,
        to_epoch_id=view.epoch_id,
        to_sequence=view.adopted_sequence,
        records_applied=records_applied,
        diagnostics=tuple(diagnostics),
    )


def _check_local_manifest_compatibility(
    local: CacheView, manifest: CoherenceManifest, *, paths: CoherencePaths
) -> None:
    """Fail-closed account/epoch identity binding for a POPULATED local view
    (seal item 1).

    Staleness-immune by construction: a lock-free manifest read may observe
    an OLDER complete artifact, but it can never fabricate a different
    account, a same-id/different-number epoch, or a same-number/different-id
    epoch — so all three violations fail closed immediately, on the fast
    path and again against the locked manifest on the slow path.
    """

    if is_unpopulated_view(local):
        return
    if local.account_scope_id is None:
        # Seal-2 item 1: a POPULATED view without an account binding is
        # never acceptable — an unbound payload must not be treated as
        # current or replayed as though it belonged to this account.
        raise CoherenceAccountMismatchError(
            "populated local cache view carries no account binding",
            path=paths.manifest_path,
            account_scope_id=manifest.account_scope_id,
            epoch_id=local.epoch_id,
            sequence=local.adopted_sequence,
        )
    if local.account_scope_id != manifest.account_scope_id:
        raise CoherenceAccountMismatchError(
            "local cache view belongs to a different account scope than the manifest",
            path=paths.manifest_path,
            account_scope_id=manifest.account_scope_id,
            epoch_id=local.epoch_id,
            sequence=local.adopted_sequence,
        )
    if local.epoch_id == manifest.epoch_id and local.epoch_number != manifest.epoch_number:
        raise JournalIntegrityError(
            "local view epoch id matches the manifest but epoch numbers differ",
            path=paths.manifest_path,
            epoch_id=local.epoch_id,
            sequence=local.adopted_sequence,
        )
    if local.epoch_id != manifest.epoch_id and local.epoch_number == manifest.epoch_number:
        raise JournalIntegrityError(
            "local view epoch number collides with a different manifest epoch identity",
            path=paths.manifest_path,
            epoch_id=local.epoch_id,
            sequence=local.adopted_sequence,
        )


def _has_above_head_records(paths: CoherencePaths, head: int) -> bool:
    """Cheap lock-free scan for ANY parseable journal artifact above the
    observed manifest head (seal item 7).

    A record written immediately AFTER this scan is legitimately invisible
    to this operation (it linearizes at the earlier head); the next
    operation detects it.  Unparseable filenames are ignored per the
    journal filename grammar; the locked orphan recovery owns all
    classification.
    """

    try:
        entries = list(paths.journal_dir.iterdir())
    except FileNotFoundError:
        return False
    for entry in entries:
        sequence = parse_journal_record_filename(entry.name)
        if sequence is not None and sequence > head:
            return True
    return False


def _cas_publish(
    publisher: CacheViewPublisher,
    build,
    *,
    context: str,
    max_attempts: int = 4,
) -> tuple[CacheView, bool]:
    """Bounded compare-and-swap publication loop.

    ``build(current)`` returns the candidate successor, or ``None`` when
    the current view already satisfies the goal (no publication needed).
    S03-CR-02: ``build`` may instead return a ``(candidate, commit)``
    tuple — the commit then runs under the SAME publisher critical section
    as the swap (``publish_with_commit``), welding per-attempt sparse
    transition-effect bookkeeping to the publication.  ``build`` runs
    OUTSIDE the publisher mutex; the swap is identity-CAS.
    """

    last_error: Exception | None = None
    for _attempt in range(max_attempts):
        current = publisher.capture()
        built = build(current)
        if built is None:
            return current, False
        candidate, commit = built if isinstance(built, tuple) else (built, None)
        try:
            if commit is not None:
                publisher.publish_with_commit(
                    candidate, expected_current=current, commit=commit
                )
            else:
                publisher.publish(candidate, expected_current=current)
            return candidate, True
        except CacheViewError as exc:
            last_error = exc
    raise CacheAdoptionError(
        f"adoption publication raced beyond {max_attempts} attempts ({context})"
    ) from last_error


# ---------------------------------------------------------------------------
# Same-epoch incremental tail replay (§5.3)
# ---------------------------------------------------------------------------


def _replay_tail_onto(
    payload: Mapping[str, Any],
    records: list[JournalTransaction],
    *,
    account_scope_id: str,
    diagnostics: list[dict[str, Any]],
) -> dict[str, Any]:
    """Apply the captured tail records in order (replay_authoritative)."""

    working: dict[str, Any] = dict(payload)
    for record in records:
        working = apply_journal_transaction(
            working,
            record,
            expected_sequence=record.sequence,
            mode=ApplicationMode.REPLAY_AUTHORITATIVE,
            expected_account=account_scope_id,
            diagnostics=diagnostics,
        )
    return working


def _publish_tail_replay(
    publisher: CacheViewPublisher,
    *,
    basis: CacheView,
    records: list[JournalTransaction],
    manifest: CoherenceManifest,
) -> CacheAdoptionResult:
    """Publish ``basis + records`` with rebase-on-race semantics (§5.3).

    - current already at/beyond target (same epoch) → return it;
    - current same epoch, behind target (incl. a transition COW patch at
      the SAME markers or a mid-way adopter) → rebase the remaining tail
      onto that newer locally-patched view and retry the CAS;
    - current same epoch, BELOW the basis → impossible by construction;
      fail closed (never a blind last-writer-wins swap);
    - current epoch changed → the caller restarts from the new state.

    Each attempt transfers the COW tail result to its terminal build; the
    unchanged S1 loop still pays k shallow preparations, never k deep copies.
    """

    target_head = manifest.journal_head_seq
    target_epoch_id = manifest.epoch_id
    records_by_seq = {record.sequence: record for record in records}
    diag_holder: dict[str, list[dict[str, Any]]] = {"diagnostics": []}
    applied_holder: dict[str, int] = {"applied": 0}

    def build(current: CacheView) -> CacheView | None:
        if current.epoch_id == target_epoch_id and current.adopted_sequence >= target_head:
            return None
        if current.epoch_id != basis.epoch_id:
            raise _LocalEpochAdvanced()
        if current.adopted_sequence < basis.adopted_sequence:
            raise CacheAdoptionError(
                "local view regressed below the adoption basis in the same epoch",
                epoch_id=current.epoch_id,
                sequence=current.adopted_sequence,
            )
        attempt_diagnostics: list[dict[str, Any]] = []
        working = _replay_tail_onto(
            current.nodes_export,
            [
                records_by_seq[seq]
                for seq in range(current.adopted_sequence + 1, target_head + 1)
            ],
            account_scope_id=manifest.account_scope_id,
            diagnostics=attempt_diagnostics,
        )
        diag_holder["diagnostics"] = attempt_diagnostics
        applied_holder["applied"] = target_head - current.adopted_sequence
        return CacheView.build(
            epoch_number=manifest.epoch_number,
            epoch_id=target_epoch_id,
            adopted_sequence=target_head,
            nodes_export=working,
            account_scope_id=manifest.account_scope_id,
            ownership="transferred",
        )

    published, did_publish = _cas_publish(
        publisher, build, context="same-epoch tail replay"
    )
    return _result(
        view=published,
        disposition="tail_replayed" if did_publish else "already_current",
        basis=basis,
        records_applied=applied_holder["applied"] if did_publish else 0,
        diagnostics=diag_holder["diagnostics"],
    )


# ---------------------------------------------------------------------------
# Epoch reload (§5.4) and clean-base recovery (§5.5)
# ---------------------------------------------------------------------------


def _materialize_reload(
    paths: CoherencePaths,
    manifest: CoherenceManifest,
    *,
    diagnostics: list[dict[str, Any]] | None = None,
) -> MaterializedView:
    """Load ONLY the manifest-referenced base + active range (never newest)."""

    return materialize_current_payload(paths, manifest, diagnostics=diagnostics)


def _publish_epoch_view(
    publisher: CacheViewPublisher,
    *,
    new_view: CacheView,
    publication_kind: str | None,
    disposition: str,
    basis: CacheView | None,
    records_applied: int,
    diagnostics: list[dict[str, Any]],
) -> CacheAdoptionResult:
    """Regression-guarded publication of a reloaded/recovered view.

    S03-CR-02 (Contract D): when the publisher carries a transition-effect
    policy (every real client publisher — coherence-internal/test
    publishers leave it ``None`` and keep the sealed no-overlay behavior),
    the offered view is rebased through the ONE sparse merge authority
    (``merge_offer_with_transition_effects``) and the surviving-effect /
    legacy-dirty bookkeeping commits atomically with the swap.  This is
    what makes ordinary cross-process epoch adoption (WO-1) and clean-base
    recovery (CF-S03CR-10) converge with — never silently erase — a
    process-local verified COW write: a COW patch racing recovery
    materialization is already composed in the policy store, so the merge
    picks it up on the winning CAS attempt whichever side lands first.
    """

    policy = getattr(publisher, "transition_policy", None)

    def build(current: CacheView) -> CacheView | tuple[CacheView, Any] | None:
        if not is_unpopulated_view(current) and _markers(current) >= _markers(new_view):
            # A newer/equivalent local view already exists — never regress.
            return None
        if policy is None:
            return new_view
        effects = policy.snapshot_effects()
        if not effects:
            return new_view
        merged = merge_offer_with_transition_effects(
            new_view, effects, publication_kind=publication_kind
        )

        def _commit(_merged=merged) -> None:
            # Inside the same critical section as the swap: replace the
            # effect store with the merge's surviving effects and mark the
            # merge's legacy-dirty evidence (carry/conflict — Contract E).
            policy.commit_merge(_merged.surviving_effects, _merged.dirty_ids)

        return merged.rebased, _commit

    published, did_publish = _cas_publish(publisher, build, context=disposition)
    final_disposition = disposition if did_publish else "already_current"
    return _result(
        view=published,
        disposition=final_disposition,
        basis=basis,
        records_applied=records_applied if did_publish else 0,
        diagnostics=diagnostics,
    )


def _reload_from_manifest(
    publisher: CacheViewPublisher,
    *,
    paths: CoherencePaths,
    manifest: CoherenceManifest,
    basis: CacheView | None,
    disposition: str,
    prior_diagnostics: list[dict[str, Any]] | None = None,
) -> CacheAdoptionResult:
    """§5.4: full epoch-base reload + complete active replay to the head.

    Replay-authoritative diagnostics from the active-record replay are
    collected and returned (seal item 6).
    S2's owned materialized payload is explicitly transferred into the view.
    """

    diagnostics: list[dict[str, Any]] = list(prior_diagnostics or [])
    materialized = _materialize_reload(paths, manifest, diagnostics=diagnostics)
    # The transition CREATE retirement discriminator is envelope provenance,
    # not wall-clock time.  materialize_current_payload already validated this
    # immutable manifest base; re-read its strict envelope here to thread the
    # publication kind into the merge without widening CacheView's identity.
    publication_kind = load_manifest_base_envelope(paths, manifest).publication_kind
    new_view = CacheView.build(
        epoch_number=manifest.epoch_number,
        epoch_id=manifest.epoch_id,
        adopted_sequence=manifest.journal_head_seq,
        nodes_export=materialized.payload,
        account_scope_id=manifest.account_scope_id,
        ownership="transferred",
    )
    records_applied = max(
        0, manifest.journal_head_seq - manifest.journal_floor_seq + 1
    )
    return _publish_epoch_view(
        publisher,
        new_view=new_view,
        publication_kind=publication_kind,
        disposition=disposition,
        basis=basis,
        records_applied=records_applied,
        diagnostics=diagnostics,
    )


# ---------------------------------------------------------------------------
# The one adoption authority
# ---------------------------------------------------------------------------


# @beacon[
#   id=coherence@adopt-manifest-head,
#   slice_labels=ra-coherence,ra-coherence-journal,ra-workflowy-cache,mci-journal-adoption,
#   kind=ast,
#   comment=S03 journal adoption engine (§5): lock-free fast head check; slow-path snapshot capture under the coherence mutex (orphan recovery, contiguous record reads); same-epoch tail replay with rebase-on-race CAS (never discards a local patch, never regresses); epoch reload from the manifest base only; one clean-base recovery; markers publish only with a completed view; coherence mutex -> publisher mutex, replay always outside. S03-CR-02: epoch-reload/recovery publication rebases through the ONE sparse transition-effect merge when the publisher carries a policy (WO-1 cross-process overlay bypass closed; CF-S03CR-10 recovery/COW race converges). WO-S03-CLOSURE-F5-1: epoch reload threads the strict snapshot-envelope publication kind into that merge so only successive full-refresh absences can retire an orphaned CREATE effect; compat-patch/checkpoint/job-basis offers remain unconditional carries.,
# ]
async def adopt_manifest_head(
    *,
    paths: CoherencePaths,
    publisher: CacheViewPublisher,
    snapshot_dir: str | Path,
    cancel_check: Any = None,
) -> CacheAdoptionResult:
    """Adopt the manifest epoch/head into the process-local cache publisher.

    Exactly one contiguous prefix through one observed manifest head is
    adopted per call; the published view's markers advance only with a
    completed local apply (§5.7).  See the module docstring for the four
    adoption dispositions and the failure-recovery contract.
    """

    restarts = 0
    while True:
        try:
            return await _adopt_once(
                paths=paths,
                publisher=publisher,
                snapshot_dir=snapshot_dir,
                cancel_check=cancel_check,
            )
        except _LocalEpochAdvanced:
            restarts += 1
            if restarts > 2:
                raise CacheAdoptionError(
                    "local epoch advanced repeatedly during adoption",
                    path=paths.manifest_path,
                ) from None


async def _adopt_once(
    *,
    paths: CoherencePaths,
    publisher: CacheViewPublisher,
    snapshot_dir: str | Path,
    cancel_check: Any,
) -> CacheAdoptionResult:
    from .refresh_authority import bootstrap_coherence_from_legacy

    # ------------------------------------------------------------------
    # §5.1 fast head check (lock-free manifest read: complete-or-old).
    # ------------------------------------------------------------------
    local = publisher.capture()
    manifest = read_manifest(paths, required=False)
    if manifest is None:
        manifest = bootstrap_coherence_from_legacy(paths, snapshot_dir=snapshot_dir)

    if is_unpopulated_view(local):
        if manifest.base_snapshot_path is None and manifest.journal_head_seq == 0:
            # Seal-2 item 7: even the sentinel fast path must check for
            # above-head journal evidence before accepting — a durable
            # orphan at sequence 1 must divert to the locked authority,
            # never sit invisible behind an empty-lineage fast path.
            if not _has_above_head_records(paths, manifest.journal_head_seq):
                # Genuinely empty lineage (bootstrap state): nothing to
                # adopt; the empty sentinel IS the current state.
                return _result(
                    view=local,
                    disposition="already_current",
                    basis=local,
                    records_applied=0,
                    diagnostics=[],
                )
    else:
        # Seal item 1: fail-closed account/epoch identity binding.  These
        # violations are staleness-immune, so they fail immediately.
        _check_local_manifest_compatibility(local, manifest, paths=paths)
        if (
            local.epoch_id == manifest.epoch_id
            and local.adopted_sequence == manifest.journal_head_seq
        ):
            # Seal item 7: never accept the fast path while ANY above-head
            # journal artifact exists (incl. gapped artifacts beyond a
            # missing head+1); divert to the locked slow path instead.  A
            # LOCAL-NEWER appearance is NOT an error here — the lock-free
            # read may simply be stale; the slow path re-reads under the
            # mutex before judging (seal item 1 rule 5).
            if not _has_above_head_records(paths, manifest.journal_head_seq):
                return _result(
                    view=local,
                    disposition="already_current",
                    basis=local,
                    records_applied=0,
                    diagnostics=[],
                )

    # ------------------------------------------------------------------
    # §5.2 slow adoption snapshot capture (bounded reads under the mutex).
    # ------------------------------------------------------------------
    async with coherence_lock_async(paths, cancel_check=cancel_check):
        manifest = read_manifest(paths, required=True)
        assert manifest is not None
        manifest = _recover_orphans_locked(paths, manifest, fs_ops=None)
        captured_head = manifest.journal_head_seq
        populated_local = not is_unpopulated_view(local)
        same_epoch = populated_local and local.epoch_id == manifest.epoch_id

        if populated_local:
            # Seal item 1: re-check identity binding against the LOCKED manifest.
            _check_local_manifest_compatibility(local, manifest, paths=paths)

        if same_epoch and local.adopted_sequence == captured_head:
            return _result(
                view=local,
                disposition="already_current",
                basis=local,
                records_applied=0,
                diagnostics=[],
            )

        tail_records: list[JournalTransaction] = []
        if same_epoch:
            if local.adopted_sequence > captured_head:
                # The LOCKED manifest still lies behind the local view —
                # genuinely invalid (no stale-read explanation remains).
                raise JournalIntegrityError(
                    "local adopted sequence is ahead of the manifest journal head "
                    "(confirmed under the coherence mutex)",
                    path=paths.manifest_path,
                    epoch_id=manifest.epoch_id,
                    sequence=local.adopted_sequence,
                )
            tail_records = read_journal_records(
                paths, local.adopted_sequence + 1, captured_head
            )
            for record in tail_records:
                _validate_record_against_manifest(
                    record,
                    manifest,
                    path=paths.journal_dir / journal_record_filename(record.sequence),
                )
        elif populated_local and local.epoch_number > manifest.epoch_number:
            # Locked re-read confirms the local epoch is NEWER than the
            # shared lineage: genuinely invalid (tampered/reset manifest).
            raise JournalIntegrityError(
                "local view epoch is newer than the locked manifest epoch",
                path=paths.manifest_path,
                epoch_id=manifest.epoch_id,
                sequence=local.adopted_sequence,
            )
    # -- coherence mutex released BEFORE full cache copying/replay -------

    # ------------------------------------------------------------------
    # §5.3 same-epoch incremental replay (private, outside the mutex).
    # ------------------------------------------------------------------
    if same_epoch:
        try:
            return _publish_tail_replay(
                publisher, basis=local, records=tail_records, manifest=manifest
            )
        except (EffectApplicationError, CacheViewError) as exc:
            # §5.5: local cache-state divergence/application failure → ONE
            # clean-base recovery to the same captured head.  Sequence,
            # account, and schema failures are NOT divergence and propagate
            # loudly without a futile retry.
            return _recover_from_base(
                publisher,
                paths=paths,
                manifest=manifest,
                basis=local,
                cause=exc,
            )

    # ------------------------------------------------------------------
    # §5.4 epoch reload (no view, unpopulated, or epoch differs).
    # ------------------------------------------------------------------
    if manifest.base_snapshot_path is None:
        if is_unpopulated_view(local) and captured_head == 0:
            # The locked manifest confirms the genuinely empty lineage.
            return _result(
                view=local,
                disposition="already_current",
                basis=local,
                records_applied=0,
                diagnostics=[],
            )
        raise CoherenceStateMissingError(
            "manifest lineage carries no base snapshot; a full refresh is "
            "required before cache adoption",
            path=paths.manifest_path,
            epoch_id=manifest.epoch_id,
        )

    return _reload_with_rotation_restart(
        publisher,
        paths=paths,
        manifest=manifest,
        basis=local,
        disposition="epoch_reloaded",
    )


def _manifest_advanced_past(
    fresh: CoherenceManifest | None, captured: CoherenceManifest
) -> bool:
    """Complete epoch-identity comparison for the rotation restart (item 5)."""

    return fresh is not None and (
        fresh.epoch_id != captured.epoch_id or fresh.epoch_number != captured.epoch_number
    )


def _reload_with_rotation_restart(
    publisher: CacheViewPublisher,
    *,
    paths: CoherencePaths,
    manifest: CoherenceManifest,
    basis: CacheView | None,
    disposition: str,
    prior_diagnostics: list[dict[str, Any]] | None = None,
) -> CacheAdoptionResult:
    """Seal item 5: bounded invalidated-basis restart.

    The captured epoch's base/records are materialized OUTSIDE the mutex.
    If the basis disappears mid-flight in a rotation-consistent way
    (missing base / missing active record), re-read the manifest: an
    ADVANCED manifest restarts adoption from the new lineage (bounded by
    the existing restart budget); an UNCHANGED manifest propagates the
    typed failure loudly.  No partial materialization is ever published.
    """

    try:
        return _reload_from_manifest(
            publisher,
            paths=paths,
            manifest=manifest,
            basis=basis,
            disposition=disposition,
            prior_diagnostics=prior_diagnostics,
        )
    except (CoherenceStateMissingError, JournalIntegrityError) as exc:
        fresh = read_manifest(paths, required=False)
        if _manifest_advanced_past(fresh, manifest):
            raise _LocalEpochAdvanced() from exc
        raise


def _recover_from_base(
    publisher: CacheViewPublisher,
    *,
    paths: CoherencePaths,
    manifest: CoherenceManifest,
    basis: CacheView,
    cause: CoherenceError,
) -> CacheAdoptionResult:
    """§5.5: one clean epoch-base reload + complete active replay.

    Publishes ``recovered_from_base`` on success; a repeated failure is a
    loud typed ``CacheAdoptionError`` and the last good view stays
    published, markers unchanged.
    """

    if manifest.base_snapshot_path is None:
        raise CacheAdoptionError(
            f"clean-base recovery impossible: manifest carries no base "
            f"(incremental failure: {cause})",
            path=paths.manifest_path,
            epoch_id=manifest.epoch_id,
        ) from cause
    prior: list[dict[str, Any]] = [
        {
            "kind": "incremental_replay_failed",
            "detail": str(cause)[:500],
            "exception": type(cause).__name__,
            "recovery": "clean_base_recovery_attempted",
        }
    ]
    try:
        result = _reload_with_rotation_restart(
            publisher,
            paths=paths,
            manifest=manifest,
            basis=basis,
            disposition="recovered_from_base",
            prior_diagnostics=prior,
        )
    except _LocalEpochAdvanced:
        raise
    except CoherenceError as exc:
        raise CacheAdoptionError(
            f"journal adoption failed and clean-base recovery also failed: {exc}",
            path=paths.manifest_path,
            epoch_id=manifest.epoch_id,
            sequence=manifest.journal_head_seq,
        ) from exc
    if result.disposition == "recovered_from_base":
        # Intentional shallow-reference update (seal-2 optional cleanup):
        # the result's diagnostics tuple shares the ``prior`` dict by
        # reference, so this final status flip is visible in the returned
        # result without reconstructing it.  The tuple itself stays fixed.
        prior[0]["recovery"] = "clean_base_succeeded"
    return result
