            def _is_structural_beacon_close(raw_line: str) -> bool:
                t = raw_line.strip()
                if t.startswith("<!--"):
                    t = t[4:].lstrip()
                if t.startswith("#"):
                    t = t.lstrip("#").lstrip()
                elif t.startswith("//"):
                    t = t[2:].lstrip()
                return t.startswith("]")
