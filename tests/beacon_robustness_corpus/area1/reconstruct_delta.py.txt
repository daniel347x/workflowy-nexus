"""Reconstruct the EXACT C1+C2 20-key golden delta list against a345dc7.

BASE = os.path.join(WORK, "base_a345dc7")
BASE_REV = "a345dc780f0a6fed4b47a63b3fd13c914b15309f"

# --- the base root must be FIRST on sys.path -------------------------------
for _p in (os.path.join(BASE, "src"), BASE, os.path.join(BASE, "tests"), WORK):
    while _p in sys.path:
        sys.path.remove(_p)
sys.path.insert(0, WORK)
sys.path.insert(0, os.path.join(BASE, "tests"))
sys.path.insert(0, os.path.join(BASE, "src"))
sys.path.insert(0, BASE)
for _m in ("nexus_map_codebase", "beacon_mutation_contract", "beacon_check_fixtures", "a1lib_base"):
    sys.modules.pop(_m, None)

# a1lib copy with SNAP pointed at the verified base snapshot
src = open(os.path.join(A1, "a1lib.py"), encoding="utf-8").read()
src2 = src.replace(
    'SNAP = "/mnt/c/tmp/beacon_robustness/probes/snapshot"',
    f'SNAP = {BASE!r}',

assert src2 != src
with open(os.path.join(WORK, "a1lib_base.py"), "w", encoding="utf-8", newline="\n") as f:
    f.write(src2)

import a1lib_base as L  # noqa: E402
import beacon_check_fixtures as fx  # noqa: E402

n = L.n
assert os.path.dirname(os.path.abspath(n.__file__)) == BASE, n.__file__
assert os.path.dirname(os.path.abspath(fx.__file__)) == os.path.join(BASE, "tests"), fx.__file__

TREE_TESTS = os.path.join(BASE, "tests")
golden = json.load(open(os.path.join(TREE_TESTS, "beacon_seams_baseline.json"), encoding="utf-8"))
SHORT = {"python": "python", "js": "js_ts", "markdown": "markdown", "sql": "sql", "sh": "shell", "yaml": "yaml"}
STOP_ORDER = ["C1_opener_or_close", "C1a_opener_only", "C2_no_prefix_line", "C3a_blank_line",
              "C1+C2", "C1+C2+C3a", "C3b_prefix_only_line"]

sources = {}
for fn in sorted(os.listdir(os.path.join(TREE_TESTS, "beacon_seams_corpus"))):
    if fn.endswith(".txt"):
        sources[f"corpus:{fn[:-4]}"] = (fn[:-4], open(os.path.join(TREE_TESTS, "beacon_seams_corpus", fn), "rb").read().decode("utf-8"))
for name, (fname, text) in fx.all_text_fixtures().items():
    sources[f"fixture:{name}"] = (fname, text)


def sha256_text(t: str) -> str:
    return hashlib.sha256(t.encode("utf-8")).hexdigest()


def sha256_file(p: str) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


summary = {"sources": len(sources), "pinned_sources": len(golden["parsers"]),
           "current_equals_baseline_parsers": 0, "current_differs_baseline_parsers": [],
           "pinned_baseline_note": ("beacon_seams_baseline.json pins base_rev 0c2ad8312be620a7b8cb397e271461dc9d50b3c2; "
                                    "on all 54 sources x 6 parsers (324 records) the a345dc7 parsers still equal the pinned "
                                    "lists, so 'before' below is both the pinned golden record and the a345dc7 behaviour"),
           "base_snapshot": BASE_REV}

keys_detail: dict[tuple[str, str], dict] = {}
for key, (fname, text) in sources.items():
    lines = text.splitlines()
    pinned = golden["parsers"].get(key)
    for short, lang in SHORT.items():
        parser = getattr(n, L.PARSER_NAME[lang])
        diag = []
        cur = parser(lines, diagnostics=diag)
        if pinned is not None:
            if json.loads(json.dumps(cur)) == pinned.get(short):
                summary["current_equals_baseline_parsers"] += 1
            else:
                summary["current_differs_baseline_parsers"].append([key, short])
        srows = L.python_string_rows(text) if lang == "python" else None
        recs = {}
        for rec in diag:
            if rec.get("record") != "block":
                continue
            c = L.classify_block(lang, lines, rec, srows)
            recs[rec["line"]] = c
        cand_diffs: dict[str, dict[int, dict]] = {}
        for cn in L.CANDIDATES:
            if cn.startswith("MD_") and lang != "markdown":
                continue
            after = L.patched_parser(lang, cn)(lines)
            d = L.diff_parses(cur, after)
            cand_diffs[cn] = {item["line"]: item for item in d}
        if cand_diffs.get("C1+C2"):
            own = L.LANG_BY_EXT.get(os.path.splitext(fname)[1].lower())
            own = "js_ts" if own == "vue" else own
            tag = "own_language" if own == lang else "cross_language"
            after12 = L.patched_parser(lang, "C1+C2")(lines)
            deltas = []
            for line, item in sorted(cand_diffs["C1+C2"].items()):
                r = recs.get(line)
                cls = "new_block_surfaced" if r is None else (
                    "anomalous_block_changed" if r["anomalous"] else "CLEAN_block_changed")
                attribution = {cn: (line in cand_diffs[cn]) for cn in STOP_ORDER}
                for cn in sorted(cand_diffs):
                    if cn.startswith("MD_"):
                        attribution[cn] = line in cand_diffs[cn]
                entry = {
                    "line": line,
                    "class": cls,
                    "before": item["before"],
                    "after": item["after"],
                    "changed_by": attribution,
                }
                if r is not None:
                    entry["before_record"] = {
                        "outcome": r["outcome"], "unterminated": r["unterminated"],
                        "interior_openers": r["interior_openers"],
                        "interior_close_openers": r["interior_close_openers"],
                        "interior_noprefix": r["interior_noprefix"],
                        "one_line_opener": r["one_line_opener"],
                        "fires_first": r["fires"],
                        "end_line": r["end_line"], "id": r["id"], "kind_raw": r["kind_raw"],
                    }
                deltas.append(entry)
            keys_detail[(key, short)] = {
                "source": key, "filename": fname, "parser": short, "tag": tag,
                "test_case": f"tests/beacon_seams_baseline.json :: parsers[{key!r}][{short!r}]",
                "clause": "U01 §6.3",
                "pinned_beacons": len(cur), "patched_beacons": len(after12),
                "delta_counts": {
                    "new_block_surfaced": sum(1 for d in deltas if d["class"] == "new_block_surfaced"),
                    "anomalous_block_changed": sum(1 for d in deltas if d["class"] == "anomalous_block_changed"),
                    "CLEAN_block_changed": sum(1 for d in deltas if d["class"] == "CLEAN_block_changed"),
                    "total": len(deltas),
                },
                "before_beacons": json.loads(json.dumps(cur)),
                "after_beacons": json.loads(json.dumps(after12)),
                "deltas": deltas,
            }

# ---- extra keys that only appear when the blank-line stop is added --------
c3a_only: dict[str, dict] = {}
for key, (fname, text) in sources.items():
    lines = text.splitlines()
    for short, lang in SHORT.items():
        parser = getattr(n, L.PARSER_NAME[lang])
        cur = parser(lines)
        d12 = {i["line"]: i for i in L.diff_parses(cur, L.patched_parser(lang, "C1+C2")(lines))}
        d123a = {i["line"]: i for i in L.diff_parses(cur, L.patched_parser(lang, "C1+C2+C3a")(lines))}
        extra = sorted(set(d123a) - set(d12))
        changed = sorted(l for l in set(d123a) & set(d12) if d123a[l] != d12[l])
        if extra or changed:
            c3a_only[f"{key}|{short}"] = {
                "records_only_with_c3a": extra,
                "records_differing_between_c1c2_and_c1c2c3a": changed,
            }

own = [k for k, v in keys_detail.items() if v["tag"] == "own_language"]
cross = [k for k, v in keys_detail.items() if v["tag"] == "cross_language"]
cls_by = collections.Counter()
cls_by_tag = collections.Counter()
for v in keys_detail.values():
    for d in v["deltas"]:
        cls_by[d["class"]] += 1
        cls_by_tag[f"{d['class']}|{v['tag']}"] += 1

# ---- separation from the later combined (lexical/pairing/wrapper) changes --

    "id_values_in_deltas_top": ids_seen.most_common(8),
    "note": ("None of the five re-paired whole-file span ids (nor any non-null id) occurs in these deltas: "
             "the C1+C2 stop-rule delta is cleanly separate from the later §6.7 pairing / §6.8 lexical / §6.5 "

    "clause_6_3_verbatim": (

        {
            "test": "tests/beacon_seams_baseline.json",
